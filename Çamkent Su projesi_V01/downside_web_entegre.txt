<script>
  const tel = document.getElementById("tel");
  const sifre = document.getElementById("sifre");
  const err = document.getElementById("err");
  const sp = document.getElementById("sp");
  const btn = document.getElementById("btn");

  function hata(msg) { err.textContent = msg; err.style.display = "block"; }

  function gir() {
    err.style.display = "none";
    const t = tel.value.trim();
    const s = sifre.value.trim();
    if (t.length !== 11) { hata("11 haneli telefon numarası girin (05XX XXX XX XX)."); return; }
    if (s.length < 4) { hata("Şifre girin."); return; }
    sp.style.display = "block";
    btn.disabled = true;

    fetch("/login?tel=" + encodeURIComponent(t) + "&sifre=" + encodeURIComponent(s),
          { cache: "no-store" })
      .then(r => r.json().catch(() => null))
      .then(d => {
        if (d && d.ok) { location.href = "/"; return; }
        sp.style.display = "none";
        btn.disabled = false;
        hata((d && d.msg) || "Giriş başarısız.");
      })
      .catch(() => {
        sp.style.display = "none";
        btn.disabled = false;
        hata("Bağlantı kurulamadı. Tekrar deneyin.");
      });
  }

  btn.addEventListener("click", gir);
  [tel, sifre].forEach(i => i.addEventListener("keydown", e => { if (e.key === "Enter") gir(); }));

  const btnTalep = document.getElementById("btnTalep");
  const st = document.getElementById("st");
  const ad = document.getElementById("ad");
  const hane = document.getElementById("hane");
  const eskiSel = document.getElementById("eski");
  const eskiLabel = document.getElementById("eskiLabel");
  let token = null;
  let pollTimer = null;
  let bekleDegisim = false;

  function durum(msg, cls) {
    st.textContent = msg;
    st.className = "status" + (cls ? " " + cls : " neu");
    st.style.display = "block";
  }

  function talepGonder(degisim) {
    const t = tel.value.trim();
    fetch("/talep?tel=" + encodeURIComponent(t) +
          "&ad=" + encodeURIComponent(ad.value.trim()) +
          "&hane=" + encodeURIComponent(hane.value.trim()) +
          "&degisim=" + (degisim ? "1" : "0") +
          (degisim && eskiSel.value ? "&eski=" + encodeURIComponent(eskiSel.value) : ""),
          { cache: "no-store" })
      .then(r => r.json())
      .then(d => {
        if (d.durum === "onaylandi") {
          sifre.value = d.sifre;
          durum("Şifreniz hazır. GİRİŞ YAP butonuyla devam edin.", "ok");
          btnTalep.style.display = "none";
          sifre.focus();
          return;
        }
        if (d.durum === "isleniyor") {
          eskiLabel.style.display = "none";
          eskiSel.style.display = "none";
          bekleDegisim = false;
          token = d.token;
          durum("Talebiniz alındı. Uygun ise onaylanacaktır; şifreniz geldiğinde burada görünecektir.", "neu");
          pollTimer = setInterval(poll, 2000);
        } else if (d.durum === "mesgul") {
          durum("Talep kuyruğu şu an dolu. Yönetici onayladıkça yeniden deneyin.", "warn");
          btnTalep.disabled = false;
        } else if (d.durum === "degisim_onay") {
          eskiSel.innerHTML = "";
          (d.eski || "").split("|").filter(x => x).forEach(ent => {
            const p = ent.split(";");
            const o = document.createElement("option");
            o.value = p[0];
            o.textContent = p[0] + " (" + (p[1] || "?") + ")";
            eskiSel.appendChild(o);
          });
          eskiLabel.style.display = "block";
          eskiSel.style.display = "block";
          bekleDegisim = true;
          durum("Bu hanede en fazla 3 numara kaydı olabilir. Listeden silinecek eski kaydı seçip talebi tekrar gönderin.", "warn");
          btnTalep.disabled = false;
        } else {
          durum("Talep alınamadı. Tekrar deneyin.", "bad");
          btnTalep.disabled = false;
        }
      })
      .catch(() => {
        durum("Bağlantı kurulamadı. Tekrar deneyin.", "bad");
        btnTalep.disabled = false;
      });
  }

  function talep() {
    err.style.display = "none";
    const t = tel.value.trim();
    if (t.length !== 11) { hata("11 haneli telefon numarası girin (05XX XXX XX XX)."); return; }
    if (ad.value.trim().length < 2) { hata("Ad soyad girin."); return; }
    if (hane.value.trim().length < 1) { hata("Hane no girin."); return; }
    btnTalep.disabled = true;
    durum("Talebiniz işleniyor…", "neu");
    if (bekleDegisim) { bekleDegisim = false; talepGonder(1); }
    else talepGonder(0);
  }

  function poll() {
    if (!token) return;
    fetch("/durum?tel=" + encodeURIComponent(tel.value.trim()) +
          "&token=" + encodeURIComponent(token), { cache: "no-store" })
      .then(r => r.json())
      .then(d => {
        if (d.durum === "onaylandi") {
          clearInterval(pollTimer); pollTimer = null;
          sifre.value = d.sifre;
          durum("Şifreniz hazır. GİRİŞ YAP butonuyla devam edin.", "ok");

// CRC16 (Modbus polysiyle aynı, A001/FFFF) — paket bütünlük doğrulaması için.
// crc16 alanı struct'ın SON elemanı olduğundan, CRC hesabı crc16 alanına kadar
// (offsetof kadar) yapılır; karşı tarafın da aynı düzeni kullandığı varsayılır.
uint16_t crc16(const uint8_t* data, size_t len) {
  uint16_t crc = 0xFFFF;
  for (size_t i = 0; i < len; i++) {
    crc ^= data[i];
    for (uint8_t j = 0; j < 8; j++) {
      if (crc & 1) crc = (crc >> 1) ^ 0xA001;
      else crc >>= 1;
    }
  }
  return crc;
}

// Gizli anahtarı üretir/kaydeder; ilk açılışta rastgele oluşturup Preferences'a yazar.
void webSecretInit() {                     // gizli anahtar: ilk açılışta üret
  prefs.begin("settings", false);
  if (!prefs.isKey("web_secret")) {
    char hex[33];
    for (int i = 0; i < 16; i++) snprintf(hex + i * 2, 3, "%02x", (uint8_t)esp_random());
    prefs.putString("web_secret", hex);
  }
  webSecret = prefs.getString("web_secret", "");
  prefs.end();
}

// Telefon numarası için 6 haneli şifre üretir (HMAC-SHA256; gizli anahtar + "camkent:"+numara).
String webSifreUret(const String &telNorm) {
  if (webSecret.length() == 0) return "";
  String msg = "camkent:" + telNorm;
  unsigned char out[32] = {0};
  mbedtls_md_context_t ctx;
  const mbedtls_md_info_t *info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
  mbedtls_md_init(&ctx);
  if (mbedtls_md_setup(&ctx, info, 1) != 0) { mbedtls_md_free(&ctx); return ""; }
  mbedtls_md_hmac_starts(&ctx, (const unsigned char *)webSecret.c_str(), webSecret.length());
  mbedtls_md_hmac_update(&ctx, (const unsigned char *)msg.c_str(), msg.length());
  mbedtls_md_hmac_finish(&ctx, out);
  mbedtls_md_free(&ctx);
  uint32_t v = ((uint32_t)out[0] << 16) | ((uint32_t)out[1] << 8) | out[2];
  char buf[8];
  snprintf(buf, sizeof(buf), "%06u", (unsigned)(v % 1000000));
  return String(buf);
}

// Numaranın engelli listesinde (web_engelli, virgülle ayrılmış) olup olmadığını kontrol eder.
bool webEngelliMi(const String &telNorm) { // web_engelli: "0532...,0533..." (virgülle)
  String list = web_engelli;
  int start = 0;
  while (start <= (int)list.length()) {
    int i = list.indexOf(',', start);
    String one = (i < 0) ? list.substring(start) : list.substring(start, i);
    if (webNorm(one) == telNorm) return true;
    if (i < 0) break;
    start = i + 1;
  }
  return false;
}

// Yeni oturum kimliği üretir (16 haneli rastgele) ve oturum tablosuna kaydeder.
String webYeniSid(const String &telNorm) { // 16 hanelik rastgele oturum kimliği
  static uint32_t seq = 0;
  char id[17];
  for (int i = 0; i < 16; i++) {
    uint32_t r = esp_random();
    id[i] = "0123456789abcdef"[r % 16];
  }
  id[16] = 0;
  int slot = -1;
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0) { slot = i; break; }
  }
  if (slot < 0) { slot = (int)(seq++ % MAX_SES); } // doluysa en eskiyi bul
  memcpy(webOturum[slot].sid, id, 17);
  webOturum[slot].tel = telNorm;
  webOturum[slot].son = millis();
  webSesKaydet();
  return String(id);
}

// Tarayıcının gönderdiği Cookie başlığından oturum kimliğini (sid) çeker.
String webSidFromCookie() {                // Cookie başlığından sid çeker
  String c = webSrv.header("Cookie");
  int p = c.indexOf("sid=");
  if (p < 0) return "";
  p += 4;
  int e = c.indexOf(';', p);
  String v = (e < 0) ? c.substring(p) : c.substring(p, e);
  v.trim();
  return v;
}

// Geçerli (süresi dolmamış) bir oturum var mı diye kontrol eder.
bool webOturumVar() {                      // geçerli oturum var mı
  String sid = webSidFromCookie();
  if (sid.length() != 16) return false;
  uint32_t now = millis();
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] != 0 && strcmp(webOturum[i].sid, sid.c_str()) == 0) {
      if (now - webOturum[i].son > SES_TTL_MS) { webOturum[i].sid[0] = 0; return false; }
      webOturum[i].son = now;
      return true;
    }
  }
  return false;
}

// Oturum tablosunu NVS'e kaydeder (reboot sonrası oturumlar korunur).
void webSesKaydet() {
  String s = "";
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0) continue;
    if (s.length() > 0) s += "|";
    s += String(webOturum[i].sid) + ";" + webOturum[i].tel;
  }
  Preferences p;
  p.begin("settings", false);
  p.putString("web_ses", s);
  p.end();
}

// Kayıtlı oturumları geri yükler (son etkinlik şimdi kabul edilir; TTL yine işler).
void webSesYukle() {
  Preferences p;
  p.begin("settings", true);
  String s = p.getString("web_ses", "");
  p.end();
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    int a = kayit.indexOf(';');
    if (kayit.length() > 16 && a == 16) {
      String sid = kayit.substring(0, 16);
      String tel = kayit.substring(a + 1);
      for (int i = 0; i < MAX_SES; i++) {
        if (webOturum[i].sid[0] == 0) {
          memcpy(webOturum[i].sid, sid.c_str(), 17);
          webOturum[i].tel = tel;
          webOturum[i].son = millis();
          break;
        }
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
}

// Bir numaranın tüm aktif oturumlarını düşürür (kullanıcı silme/engelleme).
void webOturumTelSil(const String &telNorm) {
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] != 0 && webOturum[i].tel == telNorm) {
      webOturum[i].sid[0] = 0;
      webOturum[i].tel = "";
    }
  }
  webSesKaydet();
}

// IP'nin hata sayacını ve kilidini sıfırlar (başarılı giriş sonrası).
void webKilitTemiz(uint32_t ip) {
  for (int i = 0; i < MAX_KAZAN; i++) if (webKilit[i].ip == ip) { webKilit[i].fail = 0; webKilit[i].kilid = 0; }
}

// IP şu anda kilitli mi diye kontrol eder (10 dakikalık süre içinde).
bool webKilitli(uint32_t ip) {
  for (int i = 0; i < MAX_KAZAN; i++) {
    if (webKilit[i].ip == ip) return (millis() - webKilit[i].kilid < LOCK_SURESI);
  }
  return false;
}

// IP'nin hata sayacını artırır; 5'e ulaşırsa IP'yi 10 dakika kilitler.
void webKilitArttir(uint32_t ip) {
  for (int i = 0; i < MAX_KAZAN; i++) {
    if (webKilit[i].ip == ip) {
      if (millis() - webKilit[i].kilid >= LOCK_SURESI) webKilit[i].fail = 0;
      webKilit[i].fail++;
      if (webKilit[i].fail >= LOCK_SINIR) webKilit[i].kilid = millis();
      return;
    }
  }
  for (int i = 0; i < MAX_KAZAN; i++) {
    if (webKilit[i].ip == 0) { webKilit[i].ip = ip; webKilit[i].fail = 1; return; }
  }
}

// Gömülü (PROGMEM) web dosyasını istemciye gönderir; bilinmeyen yol 404 döner.
void webSrvFile(const char *path, const char *type) {
  const char *veri = 0;
  if      (strcmp(path, "/index.html") == 0) veri = WEB_INDEX_HTML;
  else if (strcmp(path, "/login.html") == 0) veri = WEB_LOGIN_HTML;
  else if (strcmp(path, "/style.css")  == 0) veri = WEB_STYLE_CSS;
  else if (strcmp(path, "/app.js")     == 0) veri = WEB_APP_JS;
  if (!veri) { webSrv.send(404, "text/plain", "not found"); return; }
  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.send_P(200, type, veri);
}

// Oturumu olmayan isteklere 401 (yetkisiz) yanıtı gönderir.
void webSrvYasak() {
  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.send(401, "application/json", "{\"err\":\"auth\"}");
}

// Giriş isteğini işler: IP kilidini kontrol eder, telefon+şifre doğrular,
// doğruysa oturum cookie'si kurup ana sayfaya yönlendirir, yanlışsa 403 verir.
void webLogin() {
  uint32_t ip = (uint32_t)webSrv.client().remoteIP();
  if (webKilitli(ip)) {
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(429, "application/json", "{\"ok\":false,\"msg\":\"Cok fazla deneme. 10 dk sonra tekrar deneyin.\"}");
    return;
  }
  String tel = webNorm(webSrv.arg("tel"));
  String sif = webSrv.arg("sifre");
  if (tel.length() != 11 || sif.length() < 4) {
    webKilitArttir(ip);
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(403, "application/json", "{\"ok\":false,\"msg\":\"Hatali giris\"}");
    return;
  }
  bool engelli = webEngelliMi(tel);
  bool kayitli = webTelKayitli(tel);
  bool dogru   = (webSifreUret(tel) == sif);
  if (engelli || !kayitli || !dogru) {
    webKilitArttir(ip);
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(403, "application/json", "{\"ok\":false,\"msg\":\"Hatali giris\"}");
    return;
  }
  webKilitTemiz(ip);
  for (int i = MAX_TALEP - 1; i >= 0; i--)
    if (webTalep[i].durum != WT_BOS && String(webTalep[i].tel) == tel) webTalepTemiz(i);
  String sid = webYeniSid(tel);
  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.sendHeader("Set-Cookie", "sid=" + sid + "; Path=/; Max-Age=2592000");
  webSrv.send(200, "application/json", "{\"ok\":true}");
}

// Oturumu sonlandırır: sid'i oturum tablosundan siler, cookie'yi temizler.
void webCikis() {
  String sid = webSidFromCookie();
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] != 0 && strcmp(webOturum[i].sid, sid.c_str()) == 0) {
      webOturum[i].sid[0] = 0;
      webOturum[i].tel = "";
    }
  }
  webSesKaydet();
  webSrv.sendHeader("Set-Cookie", "sid=; Path=/; Max-Age=0");
  webSrv.sendHeader("Location", "/");
  webSrv.send(302, "text/plain", "ok");
}

// --- Onaylı hane kayıt defteri (Preferences/NVS) ---
// Format: "hane;ad;tel|hane;ad;tel|..."  (';' ve '|' ad içinde temizlenir)
// Bir haneye EN FAZLA 3 kullanıcı (numara) tanımlanabilir.
#define KAYIT_ANAHTAR "webkayit"
#define MAX_HANE_KULLANICI 3
String webKayitOku() {
  Preferences p;
  p.begin("settings", true);
  String s = p.getString(KAYIT_ANAHTAR, "");
  p.end();
  return s;
}

void webKayitYaz(const String &s) {
  Preferences p;
  p.begin("settings", false);
  p.putString(KAYIT_ANAHTAR, s);
  p.end();
}

// Hanedeki kullanıcı (kayıt) sayısını döner.
int webHaneKac(const String &hane) {
  String s = webKayitOku();
  int n = 0, pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      if (a > 0 && kayit.substring(0, a) == hane) n++;
    }
    if (e < 0) break;
    pos = e + 1;
  }
  return n;
}

// Ad Soyad'dan soyadı (son sözcük) döner — yalnızca panel listesinde gösterim için.
String webSoyad(const String &ad) {
  int sp = ad.lastIndexOf(' ');
  return (sp < 0) ? ad : ad.substring(sp + 1);
}

// 11 haneli numarayı "0532 xxx xx xx" biçiminde gösterir (kısa ise olduğu gibi).
String webTelGoster(const String &tel) {
  if (tel.length() == 11)
    return tel.substring(0, 4) + " " + tel.substring(4, 7) + " " +
           tel.substring(7, 9) + " " + tel.substring(9, 11);
  return tel;
}

// Hanedeki kayıtlı numaraları panel listesi için "tel;soyad|tel;soyad|..." döner.
String webHaneListe(const String &hane) {
  String s = webKayitOku();
  String r = "";
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a && kayit.substring(0, a) == hane) {
        if (r.length() > 0) r += "|";
        r += kayit.substring(b + 1) + ";" + webSoyad(kayit.substring(a + 1, b));
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
  return r;
}

// Kullanıcı ekler: aynı hane+tel varsa adı günceller; yeni kullanıcı için
// hane doluysa (3) false döner (defter değişmez). 11 haneli olmayan tel reddedilir.
bool webHaneEkle(const String &hane, const String &ad, const String &tel) {
  if (tel.length() != 11) return false;
  String s = webKayitOku();
  String yeni = "";
  bool vardi = false;
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a && kayit.substring(0, a) == hane && kayit.substring(b + 1) == tel) {
        yeni += hane + ";" + ad + ";" + tel + "|";   // ad güncelle
        vardi = true;
      } else {
        yeni += kayit + "|";
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
  if (!vardi) {
    if (webHaneKac(hane) >= MAX_HANE_KULLANICI) return false;  // hane dolu
    yeni += hane + ";" + ad + ";" + tel + "|";
  }
  webKayitYaz(yeni);
  return true;
}

  }
  return "KAYITLI (" + String(n) + ")\r\n" + satir;
}

// Belirli bir hanedeki kayıtları "hane | ad | tel" satırlarıyla döndürür.
// Messenger'a "liste 27" gibi yazınca yalnızca o hane gösterilir.
String webHaneKayitListesi(const String &hane) {
  String s = webKayitOku();
  String satir = "";
  int n = 0;
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a && kayit.substring(0, a) == hane) {
        n++;
        satir += kayit.substring(0, a) + " | " +
                 kayit.substring(a + 1, b) + " | " + webTelGoster(kayit.substring(b + 1)) + "\r\n";
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
  if (n == 0) return "HANE " + hane + ": kayit yok";
  return "HANE " + hane + " (" + String(n) + " kayit)\r\n" + satir;
}

// Telefon numarasının kayıtlı olduğu hane; kayıtlı değilse boş döner.
String webTelHane(const String &tel) {
  String s = webKayitOku();
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a && kayit.substring(b + 1) == tel) return kayit.substring(0, a);
    }
    if (e < 0) break;
    pos = e + 1;
  }
  return "";
}

// Şu an paneli açık olan (anlık aktif) kullanıcıların listesi.
// Panel her 2 sn'de /data çeker ve her istekte webOturumVar() "son"u tazeler;
// son etkinlik 10 sn içindeyse o kullanıcı ANLIK AKTİF sayılır.
// haneFiltre doluysa yalnızca o haneye kayıtlı aktif numaralar gösterilir ("aktif 27").
String webAktiflerListesi(const String &haneFiltre) {
  #define AKTIF_ESIK_MS 10000UL
  String unique[16];
  bool online[16];
  uint8_t uc = 0;
  uint32_t now = millis();
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0 || webOturum[i].tel.length() < 7) continue;
    bool var = false;
    for (uint8_t j = 0; j < uc; j++) if (unique[j] == webOturum[i].tel) { var = true; break; }
    if (!var && uc < 16) { unique[uc] = webOturum[i].tel; online[uc] = false; uc++; }
  }
  int onSay = 0;
  for (uint8_t j = 0; j < uc; j++) {
    uint32_t sonGiris = 0;
    for (int i = 0; i < MAX_SES; i++)
      if (webOturum[i].sid[0] != 0 && webOturum[i].tel == unique[j] && webOturum[i].son > sonGiris)
        sonGiris = webOturum[i].son;
    online[j] = ((uint32_t)(now - sonGiris) <= AKTIF_ESIK_MS);
    if (online[j] && (haneFiltre.length() == 0 || webTelHane(unique[j]) == haneFiltre)) onSay++;
  }
  if (onSay == 0) {
    if (haneFiltre.length() > 0) return "HANE " + haneFiltre + ": AKTIF KULLANICI YOK";
    return "SU AN AKTIF KULLANICI YOK";
  }
  String r = "ANLIK AKTIF KULLANICILAR (" + String(onSay) + ")\r\n";
  uint8_t no = 0;
  for (uint8_t j = 0; j < uc; j++)
    if (online[j] && (haneFiltre.length() == 0 || webTelHane(unique[j]) == haneFiltre)) {
      no++;
      r += String(no) + ". " + webTelGoster(unique[j]) + " - ONLINE\r\n";
    }
  return r;
}

// Kuyruğu NVS'e yazar (talepler reboot'ta korunur). Format her kayıt:
//   ad;hane;tel;eski;token;durum;degisim   (kayıtlar | ile ayrılır)
void webTalepKaydet() {
  String s;
  for (int i = 0; i < MAX_TALEP; i++) {
    if (webTalep[i].durum == WT_BOS) continue;
    if (s.length() > 0) s += "|";
    s += String(webTalep[i].ad) + ";" + String(webTalep[i].hane) + ";" +
         String(webTalep[i].tel) + ";" + String(webTalep[i].eski) + ";" +
         String(webTalep[i].token) + ";" + String((int)webTalep[i].durum) + ";" +
         String(webTalep[i].degisim ? "1" : "0");
  }
  Preferences p;
  p.begin("settings", false);
  if (s.length() == 0) p.remove(TALEP_ANAHTAR);
  else p.putString(TALEP_ANAHTAR, s);
  p.end();
}

// Kuyruğu NVS'ten okur (setup sırasında çağrılır).
void webTalepYukle() {
  memset(webTalep, 0, sizeof(webTalep));
  Preferences p;
  p.begin("settings", true);
  String s = p.getString(TALEP_ANAHTAR, "");
  p.end();
  int idx = 0, pos = 0;
  while (pos <= (int)s.length() && idx < MAX_TALEP) {
    int e = s.indexOf('|', pos);
    String rec = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (rec.length() > 0) {
      String alan[7];
      int n = 0, q = 0;
      while (q <= (int)rec.length() && n < 7) {
        int se = rec.indexOf(';', q);
        alan[n++] = (se < 0) ? rec.substring(q) : rec.substring(q, se);
        if (se < 0) break;
        q = se + 1;
      }
      if (n >= 6 && alan[2].length() == 11) {
        strncpy(webTalep[idx].ad, alan[0].c_str(), sizeof(webTalep[idx].ad) - 1);
        strncpy(webTalep[idx].hane, alan[1].c_str(), sizeof(webTalep[idx].hane) - 1);
        strncpy(webTalep[idx].tel, alan[2].c_str(), sizeof(webTalep[idx].tel) - 1);
        strncpy(webTalep[idx].eski, alan[3].c_str(), sizeof(webTalep[idx].eski) - 1);
        strncpy(webTalep[idx].token, alan[4].c_str(), sizeof(webTalep[idx].token) - 1);
        webTalep[idx].durum = (uint8_t)alan[5].toInt();
        webTalep[idx].degisim = (n >= 7 && alan[6] == "1");
        idx++;
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
  webTalepGuncelle();
}

// Cloud göstergelerini kuyruğa göre günceller: web_talep_bekliyor bayrağı ve
// onay için sıradaki (ilk bekleyen) talebin bilgisi web_talep_bilgi'ye yazılır.
void webTalepGuncelle() {
  int bekliyor = 0, ilk = -1;
  for (int i = 0; i < MAX_TALEP; i++) {
    if (webTalep[i].durum == WT_BEKLIYOR) {
      if (ilk < 0) ilk = i;
      bekliyor++;
    }
  }
  web_talep_bekliyor = (bekliyor > 0);
  if (ilk < 0) {
    if (web_talep_bilgi.length() > 0) web_talep_bilgi = "";
  } else {
    String b = "BEKLEYEN TALEP: " + String(bekliyor) + "\r\n";
    b += "Sira: " + String(ilk + 1) + "/" + String(bekliyor) + "\r\n";
    b += "Ad Soyad : " + String(webTalep[ilk].ad) + "\r\n";
    b += "Hane No : " + String(webTalep[ilk].hane) + "\r\n";
    b += "Telefon No : " + webTelGoster(String(webTalep[ilk].tel)) + "\r\n";
    if (webTalep[ilk].degisim)
      b += "DEGISIM: eski " + webTelGoster(String(webTalep[ilk].eski)) +
           " -> " + webTelGoster(String(webTalep[ilk].tel)) + "\r\n";
    web_talep_bilgi = b;
  }
}

// i. talebi kuyruktan çıkarır (arkadakileri kaydırır), NVS + cloud'u günceller.
void webTalepTemiz(int i) {
  if (i < 0 || i >= MAX_TALEP) return;
  for (int k = i; k < MAX_TALEP - 1; k++) webTalep[k] = webTalep[k + 1];
  memset(&webTalep[MAX_TALEP - 1], 0, sizeof(WebTalep));
  webTalepKaydet();
  webTalepGuncelle();
}

// token+tel eşleşen talebin indeksini döner, yoksa -1.
int webTalepBulToken(const String &tel, const String &token) {
  for (int i = 0; i < MAX_TALEP; i++) {
    if (webTalep[i].durum == WT_BOS) continue;
    if (tel == String(webTalep[i].tel) && token == String(webTalep[i].token)) return i;
  }
  return -1;
}

// Aynı telefona ait ilk talebi (durum fark etmez) döner, yoksa -1.
int webTalepBulTel(const String &tel) {
  for (int i = 0; i < MAX_TALEP; i++) {
    if (webTalep[i].durum == WT_BOS) continue;
    if (tel == String(webTalep[i].tel)) return i;
  }
  return -1;
}

// /talep: ad+hane+tel ile talep oluşturur. Talepler süresiz kuyrukta bekler.
// Kural (soyad fark etmez): aynı haneye en fazla 3 numara kaydı.
//  - numara zaten hanede → şifre yenileme
//  - hane < 3 kayıt → normal ekleme (aynı soyad + yeni numara da dahil)
//  - hane dolu (3) → "degisim_onay": panel hangi eski kaydın silineceğini
//    sorar, degisim=1 + eski ile tekrar çağırır.
void webTalepIste() {
  String tel = webNorm(webSrv.arg("tel"));
  String hane = webNorm(webSrv.arg("hane"));
  String ad = webSrv.arg("ad");
  ad.replace(';', ' '); ad.replace('|', ' '); ad.trim();
  bool degisim = (webSrv.arg("degisim") == "1");
  if (tel.length() != 11 || hane.length() < 1 || ad.length() < 2) {
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(400, "application/json", "{\"durum\":\"hatali\"}");
    return;
  }

  // Kayıtlı (onaylı) numara → onay gerekmez, şifre anında verilir (ad güncellenir).
  if (webHaneTelVar(hane, tel)) {
    webHaneEkle(hane, ad, tel);
    int t = webTalepBulTel(tel);
    if (t >= 0) webTalepTemiz(t);   // artık gereksiz talep kaydını kaldır
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(200, "application/json",
      "{\"durum\":\"onaylandi\",\"sifre\":\"" + webSifreUret(tel) + "\"}");
    return;
  }

  // Aynı numaranın önceki talebini kontrol et (süresiz kuyruk).
  for (int i = 0; i < MAX_TALEP; i++) {
    if (webTalep[i].durum == WT_BOS) continue;
    if (String(webTalep[i].tel) != tel) continue;
    if (webTalep[i].durum == WT_BEKLIYOR) {
      // Hâlâ bekliyor → aynı token'ı dön (mükerrer talep oluşmaz).
      webSrv.sendHeader("Cache-Control", "no-store");
      webSrv.send(200, "application/json",
        "{\"durum\":\"isleniyor\",\"token\":\"" + String(webTalep[i].token) + "\"}");
      return;
    }
    if (webTalep[i].durum == WT_ONAYLANDI) {
      // Onaylanmış ama şifre alınmamış (kullanıcı çıkmıştı) → şifreyi anında ver.
      webHaneEkle(String(webTalep[i].hane), String(webTalep[i].ad), String(webTalep[i].tel));
      webSrv.sendHeader("Cache-Control", "no-store");
      webSrv.send(200, "application/json",
        "{\"durum\":\"onaylandi\",\"sifre\":\"" + webSifreUret(tel) + "\"}");
      return;
    }
    // WT_RED → kullanıcı yeniden başvuruyor; eski red talebi yerine yenisi eklenir.
    webTalepTemiz(i);
    break;
  }

  bool degisiklik = false;
  String eski = webNorm(webSrv.arg("eski"));

  if (webHaneTelVar(hane, tel)) {
    // Numara zaten hanede kayıtlı → şifre yenileme (ad güncellenir).
  } else if (webHaneKac(hane) < MAX_HANE_KULLANICI) {
    // Haneye yeni numara ekleniyor; hane dolu değilse (3) normal ekleme.
    // Soyad fark etmez: aynı soyad + aynı hane + yeni numara da kayıt olabilir.
  } else {
    // Hane dolu (3 kayıt): yeni numara için silinecek eski kayıt seçilmeli.
    if (!degisim || !webHaneTelVar(hane, eski)) {
      String liste = webHaneListe(hane);
      webSrv.sendHeader("Cache-Control", "no-store");
      webSrv.send(200, "application/json",
        "{\"durum\":\"degisim_onay\",\"eski\":\"" + liste + "\"}");
      return;
    }
    degisiklik = true;
  }

  // Boş yuva bul; kuyruk doluysa önce RED, sonra ONAYLANDI atılır (BEKLIYOR asla atılmaz).
  int yuv = -1;
  for (int i = 0; i < MAX_TALEP; i++)
    if (webTalep[i].durum == WT_BOS) { yuv = i; break; }
  if (yuv < 0) {
    for (int i = 0; i < MAX_TALEP; i++)
      if (webTalep[i].durum == WT_RED) { yuv = i; break; }
    if (yuv < 0)
      for (int i = 0; i < MAX_TALEP; i++)
        if (webTalep[i].durum == WT_ONAYLANDI) { yuv = i; break; }
    if (yuv < 0) {
      webSrv.sendHeader("Cache-Control", "no-store");
      webSrv.send(200, "application/json", "{\"durum\":\"mesgul\"}");
      return;
    }
    webTalepTemiz(yuv);
  }

  strncpy(webTalep[yuv].ad, ad.c_str(), sizeof(webTalep[yuv].ad) - 1);
  webTalep[yuv].ad[sizeof(webTalep[yuv].ad) - 1] = 0;
  strncpy(webTalep[yuv].hane, hane.c_str(), sizeof(webTalep[yuv].hane) - 1);
  webTalep[yuv].hane[sizeof(webTalep[yuv].hane) - 1] = 0;
  strncpy(webTalep[yuv].tel, tel.c_str(), sizeof(webTalep[yuv].tel) - 1);
  webTalep[yuv].tel[sizeof(webTalep[yuv].tel) - 1] = 0;
  strncpy(webTalep[yuv].eski, eski.c_str(), sizeof(webTalep[yuv].eski) - 1);
  webTalep[yuv].eski[sizeof(webTalep[yuv].eski) - 1] = 0;
  String token = webTalepTokenUret();
  strncpy(webTalep[yuv].token, token.c_str(), sizeof(webTalep[yuv].token) - 1);
  webTalep[yuv].token[sizeof(webTalep[yuv].token) - 1] = 0;
  webTalep[yuv].durum = WT_BEKLIYOR;
  webTalep[yuv].degisim = degisiklik;

  webTalepKaydet();
  webTalepGuncelle();

  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.send(200, "application/json",
    "{\"durum\":\"isleniyor\",\"token\":\"" + token + "\"}");
}

// /durum: login sayfasının poll'u. Şifre yalnızca token+tel eşleşirse döner.
// Talepler süresiz olduğundan zaman aşımı yok: kullanıcı çıksa bile onaylandığında
// şifresini alabilir; reddedildiyse "red", hâlâ bekliyorsa "bekliyor" görür.
void webTalepDurum() {
  String tel = webNorm(webSrv.arg("tel"));
  String token = webSrv.arg("token");
  int i = webTalepBulToken(tel, token);
  String cevap;
  if (i < 0) {
    cevap = "{\"durum\":\"yok\"}";
  } else if (webTalep[i].durum == WT_BEKLIYOR) {
    cevap = "{\"durum\":\"bekliyor\"}";
  } else if (webTalep[i].durum == WT_ONAYLANDI) {
    cevap = "{\"durum\":\"onaylandi\",\"sifre\":\"" + webSifreUret(tel) + "\"}";
  } else {
    cevap = "{\"durum\":\"red\"}";
  }
  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.send(200, "application/json", cevap);
}

// Alarm listesinden mesajı kaldırır (birebir eşleşen ilk kayıt).
void clearAlarm(const char* msg) {
  for (int i = 0; i < alarmCount; i++) {
    if (strcmp(alarmList[i], msg) == 0) {
      for (int j = i; j < alarmCount - 1; j++)
        strcpy(alarmList[j], alarmList[j + 1]);
      alarmCount--;
      return;
    }
  }
}

// Değeri [a, b] aralığına sıkıştırır.
float clampf(float x, float a, float b) {
  if (x < a) return a;
  if (x > b) return b;
  return x;
}

// Analog voltajı bar'a çevirir (4-20 mA: V4=V(4mA), V20=V(20mA), ölçek 0..SENSOR_MAX_BAR).
float voltsToBar(float V, float V4, float V20) {
  if (fabs(V20 - V4) < 0.0001f)
    return 0;
  float ratio = (V - V4) / (V20 - V4);
  float bar = ratio * SENSOR_MAX_BAR;
  return clampf(bar, 0, SENSOR_MAX_BAR);
}

// WiFi sinyal gücünü (RSSI) yüzdeye çevirir (-100..-50 dBm → %0..%100).
int wifiToPercent(int rssi) {
  if (rssi <= -100) return 0;
  if (rssi >= -50)  return 100;
  return 2 * (rssi + 100);
}

// Ayarlar değiştiyse ve 1 saniye beklenildiyse Preferences'a yazar (debounce).
void saveToPrefsDelayed() {
  if (prefsNeedsSave && (millis() - lastPrefsChange > 1000)) {
    prefs.begin("settings", false);
    prefs.putFloat("siteAlt", site_alt_limit);
    prefs.putFloat("siteUst", site_ust_limit);

    prefs.putBool("kstart", kuyu_start);
    prefs.putBool("hstart", hidr_start);
    prefs.end();
    prefsNeedsSave = false;
  }
}

  snprintf(buffer2, sizeof(buffer2), "page1.h1.val=%d", int(site_ust_limit * 10.0f));

  sistem_start = sys_start;

// Nextion'dan gelen veri akışı: 0xFF (end marker/gürültü) ve 0x65
// (dokunmatik olay, 7 byte) yok sayılır, tanınan kontrol ID'leri
// (0x01..0x06) için 4 byte little-endian değer okunur.
void readNextion() {
  static uint32_t lastRead = 0;
  if (millis() - lastRead < 50) return;
  lastRead = millis();

  while (nex.available() > 0) {
    int first = nex.peek();
    if (first < 0) break;

    if (first == 0xFF) {         // end marker / gürültü
      nex.read();
      continue;
    }

    if (first == 0x65) {         // dokunmatik olay → 7 byte atla
      if (nex.available() < 7) break;
      for (int i = 0; i < 7; i++) nex.read();
      continue;
    }

    if (first < 0x01 || first > 0x06) {  // tanınmayan byte → yeniden senkron
      nex.read();
      continue;
    }

    if (nex.available() < 5) break;      // eksik frame, bir sonraki tura bırak

    uint8_t controlId = nex.read();
    uint32_t value = 0;
    for (int i = 0; i < 4; i++) {
      value |= ((uint32_t)nex.read()) << (i * 8);
    }

    switch (controlId) {
      case 0x01:  // site_alt_limit progress bar
        site_alt_limit = value / 10.0f;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x02:  // site_ust_limit progress bar
        site_ust_limit = value / 10.0f;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x03:  // page1 yüklendi → değerleri gönder
        buffer1[0] = '\0';
        buffer2[0] = '\0';
        snprintf(buffer1, sizeof(buffer1), "page1.x0.val=%d", int(site_alt_limit * 10.0f));
        snprintf(buffer2, sizeof(buffer2), "page1.x1.val=%d", int(site_ust_limit * 10.0f));
        nextionSend(buffer1);
        nextionSend(buffer2);
        buffer1[0] = '\0';
        buffer2[0] = '\0';
        snprintf(buffer1, sizeof(buffer1), "page1.h0.val=%d", int(site_alt_limit * 10.0f));
        snprintf(buffer2, sizeof(buffer2), "page1.h1.val=%d", int(site_ust_limit * 10.0f));
        nextionSend(buffer1);
        nextionSend(buffer2);
        nextionSend(kuyu_start ? "page1.c0.val=1" : "page1.c0.val=0");
        nextionSend(hidr_start ? "page1.c1.val=1" : "page1.c1.val=0");
        break;

      case 0x04:
        sys_start = bool(value);
        sistem_start = sys_start;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x05:
        kuyu_start = bool(value);
        ckuyu_start = kuyu_start;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x06:
        hidr_start = bool(value);
        chidr_start = hidr_start;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      default:
        break;
    }
  }
}

// MCP çıkış genişletici üzerindeki 6 dijital girişi okur (yangın alt/üst, site alt, faz, inverter, kuyu).
// [EKLENDI 07/08] Debounce: şamandıra/kontak girişleri yeni duruma ancak DEBOUNCE_MS süre boyunca
// stabil kaldıktan sonra geçer (kontak zıplaması / su yüzeyi dalgalanması çırpınmayı önler).
void readInputs() {
  const uint32_t DB_MS = 300UL;
  static bool raw[6] = {false, false, false, false, false, false};
  static bool dbOut[6] = {false, false, false, false, false, false};
  static uint32_t chg[6] = {0, 0, 0, 0, 0, 0};

  bool r[6];
  r[0] = !mcp.digitalRead(0);  // fireAlt
  r[1] = !mcp.digitalRead(1);  // fireUst
  r[2] = !mcp.digitalRead(2);  // siteAlt
  r[3] = !mcp.digitalRead(3);  // fazOK
  r[4] = !mcp.digitalRead(4);  // inverterOK
  r[5] = !mcp.digitalRead(5);  // kuyu

  for (int i = 0; i < 6; i++) {
    if (r[i] != raw[i]) {
      raw[i] = r[i];
      chg[i] = millis();
    } else if (r[i] != dbOut[i] && (millis() - chg[i]) >= DB_MS) {
      dbOut[i] = r[i];
    }
  }

  io.fireAlt    = dbOut[0];
  io.fireUst    = dbOut[1];
  io.siteAlt    = dbOut[2];
  io.fazOK      = dbOut[3];
  io.inverterOK = dbOut[4];
  io.kuyu       = dbOut[5];
}

// Tüm çıkışları beklenen duruma göre zorla yazar. mcpSetup recovery
// sırasında çağrıldığında diğer kanalların (kuyu/hidrant) OFF'ta
// kilitlenmesini önler.
void forceWriteOutputs() {
  mcp.digitalWrite(inv_en, expected_inv);
  mcp.digitalWrite(yedek_en, expected_yedek);
  mcp.digitalWrite(kuyu_en, expected_kuyu);
  mcp.digitalWrite(hidr_en, expected_hidr);
  last_inv   = expected_inv;
  last_yedek = expected_yedek;
  last_kuyu  = expected_kuyu;
  last_hidr  = expected_hidr;
}

// MCP23017'yi I2C üzerinden başlatır; pin modlarını kurar, çıkışları güvenli duruma (kapalı) çeker.
void mcpSetup() {
  if (!mcp.begin_I2C(MCP_ADDR)) {
    addAlarm("MCP Baglanti Hatasi");
    return;
  }

  for (int i = 0; i <= 5; i++) {
    mcp.pinMode(i, INPUT);
  }

  mcp.pinMode(inv_en, OUTPUT);
  mcp.pinMode(yedek_en, OUTPUT);
  mcp.pinMode(kuyu_en, OUTPUT);
  mcp.pinMode(hidr_en, OUTPUT);

  // Önce güvenli durum (hepsi kapalı)
  mcp.digitalWrite(inv_en, HIGH);
  mcp.digitalWrite(yedek_en, HIGH);
  mcp.digitalWrite(kuyu_en, HIGH);
  mcp.digitalWrite(hidr_en, HIGH);

  forceWriteOutputs();
}

// Site su basınç kontrolü: histerezis ile inverter/yedek pompa seçimi yapar,
// çıkışları yazar ve 1 saniyelik okuma ile MCP kilidi algılar (10 sn → recovery).
void siteControl() {
  static bool pumpRunning = false;
  static uint32_t mismatchTimer = 0;
  static uint32_t lastReadback = 0;
  static bool recoveryActive = false;
  static uint32_t recoveryStart = 0;

  if (!hard_test) {

    // Madde 1: site basınç sensörü 10 sn'dir geçerli örnek vermediyse bayat sayılır.
    // (Kablo kopması / 4mA altı zaten ADC bloğunda siteSensorOK=false yapar.)
    if ((millis() - lastSiteBarUpdate) > 10000UL) {
      siteSensorOK = false;
    }
    if (!siteSensorOK) {
      addAlarm("Site basinc sensoru bayat!");
    } else {
      clearAlarm("Site basinc sensoru bayat!");
    }

    // Madde 1: şebeke basınç sensörü 10 sn bayatsa işaretlenir (yalnızca ekran).
    if ((millis() - lastLineBarUpdate) > 10000UL) {
      lineSensorOK = false;
    }
    if (!lineSensorOK) {
      addAlarm("Sebeke basinc sensoru bayat!");
    } else {
      clearAlarm("Sebeke basinc sensoru bayat!");
    }

    // SYSTEM OFF
    if (!io.siteAlt || !io.fazOK || !sys_start || !siteSensorOK) {
      expected_inv = true;
      expected_yedek = true;
      pumpRunning = false;
      mismatchTimer = 0;
      recoveryActive = false;
    }
    else {

      // HYSTERESIS
      if (!pumpRunning && site_bar < site_alt_limit)
        pumpRunning = true;
      else if (pumpRunning && site_bar > site_ust_limit)
        pumpRunning = false;

      // OUTPUT SELECT
      if (pumpRunning) {
        if (io.inverterOK) {
          expected_inv = false;
          expected_yedek = true;
        } else {
          expected_inv = true;
          expected_yedek = false;
        }
      } else {
        expected_inv = true;
        expected_yedek = true;
      }
    }

    // WRITE ONLY IF CHANGED
    if (expected_inv != last_inv) {
      last_inv = expected_inv;
      mcp.digitalWrite(inv_en, expected_inv);
    }
    if (expected_yedek != last_yedek) {
      last_yedek = expected_yedek;
      mcp.digitalWrite(yedek_en, expected_yedek);
    }

    // READBACK 1 sn
    if (millis() - lastReadback >= 1000) {
      lastReadback = millis();

      bool inv_state = mcp.digitalRead(inv_en);
      bool yedek_state = mcp.digitalRead(yedek_en);

      bool mismatch =
        (inv_state != expected_inv) ||
        (yedek_state != expected_yedek);

      // Recovery sonrası kontrol
      if (recoveryActive) {
        if (millis() - recoveryStart >= 500) {
          recoveryActive = false;
          if (mismatch) {
            addAlarm("MCP OUTPUT FROZEN!");
            Rst_sys = true;
          } else {
            mismatchTimer = 0;
          }
        }
      }
      else {
        if (mismatch) {
          if (mismatchTimer == 0) {
            mismatchTimer = millis();
            addAlarm("in site control");
          }
          else if (millis() - mismatchTimer >= 10000) {
            clearAlarm("in site control");
            mcpSetup();
            recoveryActive = true;
            recoveryStart = millis();
          }
        } else {
          mismatchTimer = 0;
        }
      }
    }
  }
  else if (hard_test && io.fazOK) {
    expected_inv = !hard_inv;
    expected_yedek = !hard_yedek;

    if (expected_inv != last_inv) {
      last_inv = expected_inv;
      mcp.digitalWrite(inv_en, expected_inv);
    }
    if (expected_yedek != last_yedek) {
      last_yedek = expected_yedek;
      mcp.digitalWrite(yedek_en, expected_yedek);
    }
  }
}

// Kuyu pompası kontrolü: su/faz/başlatma koşulları ve yangın üst sensörüne göre
// çalıştırır; çıkış kilidini (MCP frozen) izleyip recovery uygular.
void kuyuControl() {
  static uint32_t mismatchTimer = 0;
  static uint32_t lastReadback = 0;
  static bool recoveryActive = false;
  static uint32_t recoveryStart = 0;
  // [DUZELTILDI] Kuyu: yangin tanki HEP DOLU tutulur.
  // fireUst (ust samandira) kuruysa kuyu calisir, islaksa (tank dolu) durur.
  // fireAlt'a bakilmaz; tank %100'den dusen su her seferinde tekrar doldurulur.
  // Kontak ziplamasi icin debounce readInputs() icinde mevcut (300 ms).
  // Guvenlik: fireUst arizali kalirsa 1 saat kesintisiz calismadan sonra pompa
  // durdurulur + alarm; kilit fireUst gorulunce veya KUYU_RETRY_MS bekleyis
  // sonrasi otomatik acilir (deadlock yok).
  // ZAMANLAMA: 1 saatlik sayac kuyu_start ACIK olduktan sonra, pompa fiilen
  // calismaya baslayinca (expected_kuyu false'a inince) baslar; pompa durdugu
  // an sifirlanir. kuyu_start kapaliyken asagidaki safety dal sayaci 0'lar —
  // sayac hicbir zaman kuyu starttan ONCE islemez.
  static uint32_t kuyuRunStart = 0;
  static bool kuyuRunLimitHit = false;
  static uint32_t kuyuRunStop = 0;
  const unsigned long KUYU_MAX_RUN_MS = 3600000UL;   // 1 saat
  const unsigned long KUYU_RETRY_MS = 600000UL;      // 10 dk bekleyis

  if (!hard_test) {

    // SYSTEM OFF / SAFETY STOP
    if (!io.kuyu || !io.fazOK || !kuyu_start) {
      expected_kuyu = true;
      kuyuRunLimitHit = false;
      kuyuRunStart = 0;
      kuyuRunStop = 0;
      clearAlarm("Kuyu pompasi max calisma suresi asildi!");
      mismatchTimer = 0;
      recoveryActive = false;
    }
    else {
      // Kilit aktifken acilma kosullarini kontrol et
      if (kuyuRunLimitHit) {
        if (io.fireUst) {
          // tank dolu → kilit kalksin
          kuyuRunLimitHit = false;
          kuyuRunStart = 0;
          kuyuRunStop = 0;
          clearAlarm("Kuyu pompasi max calisma suresi asildi!");
        } else if (millis() - kuyuRunStop >= KUYU_RETRY_MS) {
          // bekleyis bitti → yeniden dene
          kuyuRunLimitHit = false;
          kuyuRunStart = millis();
          clearAlarm("Kuyu pompasi max calisma suresi asildi!");
        }
      }

      if (!kuyuRunLimitHit) {
        if (!io.fireUst) {
          // tank dolu degil → calistir
          if (expected_kuyu) kuyuRunStart = millis();
          expected_kuyu = false;
          if (kuyuRunStart != 0 && (millis() - kuyuRunStart >= KUYU_MAX_RUN_MS)) {
            kuyuRunLimitHit = true;
            kuyuRunStop = millis();
            expected_kuyu = true;
            addAlarm("Kuyu pompasi max calisma suresi asildi!");
          }
        } else {
          // tank dolu → durdur
          expected_kuyu = true;
          kuyuRunStart = 0;
        }
      }
    }

    // WRITE ONLY IF CHANGED
    if (expected_kuyu != last_kuyu) {
      last_kuyu = expected_kuyu;
      mcp.digitalWrite(kuyu_en, expected_kuyu);
    }

    // READBACK 1 sn
    if (millis() - lastReadback >= 1000) {
      lastReadback = millis();

      bool kuyu_state = mcp.digitalRead(kuyu_en);
      bool mismatch = (kuyu_state != expected_kuyu);

      if (recoveryActive) {
        if (millis() - recoveryStart >= 500) {
          recoveryActive = false;
          if (mismatch) {
            addAlarm("MCP KUYU OUTPUT FROZEN!");
            Rst_sys = true;
          } else {
            mismatchTimer = 0;
          }
        }
      }
      else {
        if (mismatch) {
          if (mismatchTimer == 0) {
            mismatchTimer = millis();
            addAlarm("in kuyu control");
          }
          else if (millis() - mismatchTimer >= 10000) {
            clearAlarm("in kuyu control");
            mcpSetup();
            recoveryActive = true;
            recoveryStart = millis();
          }
        } else {
          mismatchTimer = 0;
        }
      }
    }
  }
  else if (hard_test && io.fazOK) {
    expected_kuyu = !hard_kuyu;

    if (expected_kuyu != last_kuyu) {
      last_kuyu = expected_kuyu;
      mcp.digitalWrite(kuyu_en, expected_kuyu);
    }
  }
}

// Hidrant pompası kontrolü: basınç verisi geçerliyse histerezis, değilse 2 dk
// döngü (aç/kapa) modunda çalışır; MCP kilidini izler.
void hidrantControl() {
  static uint32_t hidrantTimer = 0;
  static bool hidrantCycle = false;
  static bool pumpRunning = false;
  static uint32_t mismatchTimer = 0;
  static uint32_t lastReadback = 0;
  static bool recoveryActive = false;
  static uint32_t recoveryStart = 0;

  if (!hard_test) {

    // SAFETY STOP
    if (!io.fireAlt || !io.fazOK || !hidr_start) {
      expected_hidr = true;
      hidrantTimer = millis();
      hidrantCycle = false;
      pumpRunning = false;
      mismatchTimer = 0;
      recoveryActive = false;
    }
    else {

      // PRESSURE VALID MODE
      // Madde 1: Upside yangın basınç sensörü bayat/ölüyse (firePressureOK=false)
      // basınç verisi geçerli sayılmaz → zaman tabanlı 2/2 döngüsüne düşülür.
      bool basincGecerli = ((millis() - lastYanBasincUpdate) < 20000UL) && remoteDataOK && rxFirePressureOK;

      if (basincGecerli) {
        // Histerezis
        if (yan_basinc < 2.0f)
          pumpRunning = true;
        else if (yan_basinc >= 2.5f)
          pumpRunning = false;
        expected_hidr = pumpRunning ? false : true;
        hidrantTimer = millis();
        hidrantCycle = false;
      }
      else if (rxFire1Start) {
        // [EKLENDI 07/08] Madde 3: Gerçek yangın (fire1_start) aktifken upside yangın basınç sensörü
        // ölü/bayatsa hidrant tank bitene kadar kesintisiz çalışır (röle ON). Tank
        // fireAlt'ın altına düşünce güvenlik STOP zaten devreye girer. fire1_start
        // kalkınca veya sensör geri gelince normal kontrolüne dönülür.
        expected_hidr = false;
        hidrantTimer = millis();
        hidrantCycle = false;
      }
      else {
        // TIME BASED MODE
        uint32_t elapsed = millis() - hidrantTimer;
        if (!hidrantCycle) {
          if (elapsed >= 120000UL) {
            hidrantCycle = true;
            hidrantTimer = millis();
            expected_hidr = false;
          }
          else {
            expected_hidr = true;
          }
        }
        else {
          if (elapsed >= 120000UL) {
            hidrantCycle = false;
            hidrantTimer = millis();
            expected_hidr = true;
          }
          else {
            expected_hidr = false;
          }
        }
      }
    }

    // WRITE ONLY IF CHANGED
    if (expected_hidr != last_hidr) {
      last_hidr = expected_hidr;
      mcp.digitalWrite(hidr_en, expected_hidr);
    }

    // READBACK 1 sn
    if (millis() - lastReadback >= 1000UL) {
      lastReadback = millis();
      bool hidr_state = mcp.digitalRead(hidr_en);
      bool mismatch = (hidr_state != expected_hidr);

      if (recoveryActive) {
        if (millis() - recoveryStart >= 500UL) {
          recoveryActive = false;
          if (mismatch) {
            addAlarm("MCP HIDRANT OUTPUT FROZEN!");
            Rst_sys = true;
          } else {
            mismatchTimer = 0;
          }
        }
      }
      else {
        if (mismatch) {
          if (mismatchTimer == 0) {
            mismatchTimer = millis();
            addAlarm("in hidr control");
          }
          else if (millis() - mismatchTimer >= 10000UL) {
            clearAlarm("in hidr control");
            mcpSetup();
            recoveryActive = true;
            recoveryStart = millis();
          }
        } else {
          mismatchTimer = 0;
        }
      }
    }
  }
  else if (hard_test && io.fazOK) {
    expected_hidr = !hard_hidr;
    if (expected_hidr != last_hidr) {
      last_hidr = expected_hidr;
      mcp.digitalWrite(hidr_en, expected_hidr);
    }
  }
}

// 3 kere üst üste I2C ping'ine cevap gelmezse sistem resetlenir.
void i2cPing() {
  static uint8_t mcpFail = 0;
  static uint8_t adsFail = 0;

  Wire.beginTransmission(MCP_ADDR);
  if (Wire.endTransmission() == 0) {
    mcpFail = 0;
  } else {
    mcpFail++;
    if (mcpFail >= 3) {
      addAlarm("MCP I2C Koptu!");
      Rst_sys = true;
    }
  }

  Wire.beginTransmission(ADS_ADDR);
  if (Wire.endTransmission() == 0) {
    adsFail = 0;
  } else {
    adsFail++;
    if (adsFail >= 3) {
      addAlarm("ADS I2C Koptu!");
      Rst_sys = true;
    }
  }
}

// UDP broadcast paketi gönderir (hedef UPSIDE; resetUpside bayrağı ile sıfırlama istenebilir).
void udpSend(bool resetUpside) {
  if (!udpStarted) return;

  static uint32_t lastSend = 0;
  if (millis() - lastSend < 1000) return;
  lastSend = millis();

  txPacket.deviceID = DEVICE_ID;
  txPacket.reset_upside = resetUpside;
  txPacket.counter = udpCounter++;
  txPacket.crc16 = crc16((const uint8_t*)&txPacket, offsetof(UdptxPacket, crc16));

  udp.beginPacket(broadcastIP, udpPort);
  udp.write((uint8_t*)&txPacket, sizeof(txPacket));
  udp.endPacket();
}

// UDP paketlerini dinler: UPSIDE'ın tüm verisini alır ve global'lere kaydeder.
// CRC doğrulaması başarısız paketler yok sayılır; sensör OK bayrakları da
// buradan alınır (firePressureOK → hidrant basınç kapısı).
// 5 sn görülmezse veri eskir (remoteDataOK=false, yan_basinc sıfırlanır).
void udpReceive() {
  int packetSize = udp.parsePacket();
  if (packetSize == sizeof(rxPacket)) {
    udp.read((uint8_t*)&rxPacket, sizeof(rxPacket));
    if (rxPacket.deviceID == 1) {
      uint16_t calc = crc16((const uint8_t*)&rxPacket, offsetof(UdprxPacket, crc16));
      if (rxPacket.crc16 == calc) {
        remoteDataOK = true;
        upsideLastSeen = millis();
        rxSitePressureOK = rxPacket.sitePressureOK;
        rxFirePressureOK = rxPacket.firePressureOK;
        rxFire1Start = rxPacket.fire1Start;   // [EKLENDI 07/08] gerçek yangın bayrağı
        if (fabs(yan_basinc - rxPacket.firePressure) >= 0.1f) {
          yan_basinc = rxPacket.firePressure;
        }
        lastYanBasincUpdate = millis();
        remoteLastSeen = millis();
      }
    }
  }

  if (millis() - remoteLastSeen > 5000) {
    remoteDataOK = false;
    rxSitePressureOK = false;
    rxFirePressureOK = false;
    rxFire1Start = false;   // [EKLENDI 07/08]
    if (yan_basinc != 0.0f) {
      yan_basinc = 0.0f;
    }
  }
}

// Ana döngü: web sunucusunu besler, cloud günceller, LED, IO/ADC okur,
// Nextion, kontroller, UDP ve gerektiğinde reset işlemlerini çalıştırır.
void loop() {
  /* ===================================================== */
  /* WEB HOST EKİ (BLOK E) — loop başı                  */
  /* downside_web_ek.txt dosyasından gömüldü.            */
  /* ===================================================== */
  if (WiFi.status() == WL_CONNECTED && !webStarted) {
    webStarted = true;
    webSrv.begin();
  }
  webSrv.handleClient();
  /* ===================================================== */

  static bool cloudInited = false;
  static bool ledState = false;
  static uint32_t lastLedToggle = 0;

  // LED 500ms aralıklarla yanıp söner
  if (millis() - lastLedToggle >= 500) {
    lastLedToggle = millis();
    ledState = !ledState;
    digitalWrite(LED_BUILTIN, ledState);
  }

  /* ===== CLOUD (ilk 3 saniyeden sonra başlat) ===== */
  if (!cloudInited && millis() > 3000) {
    cloudInited = true;
    ArduinoCloud.begin(ArduinoIoTPreferredConnection);
    setDebugMessageLevel(2);
  }

  // Kütüphane her döngüde update() çağrılmasını ister; bağlantı kurulamazsa
  // bu çağrı yeniden bağlanmayı da yönetir.
  if (cloudInited) {
    ArduinoCloud.update();

    // Messenger akışı: kayıtlıları 500 ms arayla tek tek gönderir
    if (webAkisAktif && (uint32_t)(millis() - webAkisSon) >= WEB_AKIS_BEKLE_MS) {
      webAkisAdim();
    }

    if (!cloudStarted && ArduinoCloud.connected()) {
      cloudStarted = true;
      configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET_SEC, NTP_SERVER);
    }
  }

  /* ===== RESET (cloud'dan bağımsız çalışır) ===== */
  if (Rst_sys) {
    Rst_sys = false;
    rcnt++;
    if (hardreset) rcnt = 0;
    hardreset = false;
    prefs.begin("settings", false);
    prefs.putLong("rcnt", rcnt);
    prefs.end();
    delay(1000);
    ESP.restart();
  }

  /* ===== IO POLL (100ms) ===== */
  if (millis() - tIo > 100) {
    tIo = millis();

    static uint32_t tI2cPing = 0;
    if (millis() - tI2cPing >= 5000) {  // Her 5 saniyede bir
      tI2cPing = millis();
      i2cPing();
    }

    readInputs();

    inv_stat   = expected_inv;
    yedek_stat = expected_yedek;
    kuyu_stat  = expected_kuyu;
    hidr_stat  = expected_hidr;
    faz_stat   = io.fazOK;
  }

  /* ===== ADC (500ms, non-blocking zamanlayıcılı) ===== */
  static uint32_t tAdc = 0;
  if (millis() - tAdc >= 500) {
    tAdc = millis();

    int16_t adc0 = ads.readADC_SingleEnded(1);
    int16_t adc1 = ads.readADC_SingleEnded(3);

    float V0 = adc0 * 0.125f / 1000.0f;
    float V1 = adc1 * 0.125f / 1000.0f;

    line_bar = voltsToBar(V0, V4_CH0, V20_CH0);
    site_bar = voltsToBar(V1, V4_CH1, V20_CH1);

    // Madde 1: 4-20mA alt eşik (V4=0.8V). 0.5V altı = kablo kopması/transdüser
    // altı → site sensörü anında geçersiz. Geçerli örnekte zaman damgası tazelenir.
    if (V1 >= 0.5f) {
      siteSensorOK = true;
      lastSiteBarUpdate = millis();
    } else {
      siteSensorOK = false;
    }

    // Madde 1: şebeke sensörü (V0) için aynı alt eşik + zaman damgası.
    if (V0 >= 0.5f) {
      lineSensorOK = true;
      lastLineBarUpdate = millis();
    } else {
      lineSensorOK = false;
    }

    if (fabs(line_bar - last_line_bar) >= 0.5) {
      line_basinc = line_bar;
      last_line_bar = line_bar;
    }
    if (fabs(site_bar - last_site_bar) >= 0.5) {
      site_basinc = site_bar;
      last_site_bar = site_bar;
    }
  }

  /* ===== CLOUD STATUS VAR SYNC (1s) ===== */
  static uint32_t tCloud = 0;
  if (cloudStarted && (millis() - tCloud > 1000)) {
    tCloud = millis();

    if (site_su_mevcut != io.siteAlt) site_su_mevcut = io.siteAlt;
    if (fire_alt != io.fireAlt) fire_alt = io.fireAlt;
    if (fire_ust != io.fireUst) fire_ust = io.fireUst;

    if (inv_stat != inv_run) inv_run = inv_stat;
    if (yedek_stat != yedek_run) yedek_run = yedek_stat;
    if (kuyu_stat != kuyu_run) kuyu_run = kuyu_stat;
    if (hidr_stat != hidr_run) hidr_run = hidr_stat;
    if (faz_stat != enerji_ok) enerji_ok = faz_stat;

    if (millis() - lastliveUpdate > 3000) {  // 3 saniyede bir
      lastliveUpdate = millis();
      isLive = !isLive;
    }
  }

  /* ===== NEXTION ===== */
  nextionUpdate();
  updateStatusDisplay();
  readNextion();

  /* ===== KONTROL ===== */
  udpReceive();
  siteControl();
  kuyuControl();
  hidrantControl();
  saveToPrefsDelayed();

  /* ===== WIFI / UDP BAŞLAT ===== */
  if (WiFi.status() == WL_CONNECTED) {
    IPAddress ip = WiFi.localIP();
    if (ip != IPAddress(0, 0, 0, 0)) {
      String newIP = ip.toString();
      if (local_ip != newIP) {
        local_ip = newIP;
      }
      if (!udpStarted) {
        udp.begin(udpPort);
        broadcastIP = ip;
        broadcastIP[3] = 255;
        udpStarted = true;
      }
    }
  }

  if (udp_send_need) {
    udp_send_need = false;
    bool doReset = up_reset;
    up_reset = false;
    udpSend(doReset);
  }
}

// Cloud'dan talep aksiyonu gelir: 1 = onayla, 2 = reddet (sıradaki ilk bekleyen talep).
// Onayda hane defterine işlenir (degisim ise eski numara yenisiyle değiştirilir).
// Talepler düşmediği için onaylanan kullanıcı sonradan token'ı ile şifresini alır.
// Dashboard için: "talep var" bayrağı, onay bekleyen kalmadığında düşer; bilgi alanı
// her aksiyondan sonra sıradaki bekleyen talebi gösterir.
void onWebTalepAksiyonChange() {
  int ilk = -1;
  for (int i = 0; i < MAX_TALEP; i++)
    if (webTalep[i].durum == WT_BEKLIYOR) { ilk = i; break; }
  if (web_talep_aksiyon == 1 && ilk >= 0) {
    if (webTalep[ilk].degisim) {
      webHaneSil(String(webTalep[ilk].hane), String(webTalep[ilk].eski));
      webHaneEkle(String(webTalep[ilk].hane), String(webTalep[ilk].ad), String(webTalep[ilk].tel));
      webOturumTelSil(String(webTalep[ilk].eski));
    } else {
      webHaneEkle(String(webTalep[ilk].hane), String(webTalep[ilk].ad), String(webTalep[ilk].tel));
    }
    webTalep[ilk].durum = WT_ONAYLANDI;
  } else if (web_talep_aksiyon == 2 && ilk >= 0) {
    webTalep[ilk].durum = WT_RED;
  }
  web_talep_aksiyon = 0;
  webTalepKaydet();
  webTalepGuncelle();
  ArduinoCloud.update();
}

// Messenger akışını başlatır: "liste" komutu gelince kayıtlıları 500 ms arayla
// tek tek mesaj olarak gönderir (webAkisAdim loop() içinde çalışır).
void webAkisBasla() {
  String s = webKayitOku();
  int n = 0, pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) n++;
    if (e < 0) break;
    pos = e + 1;
  }
  webAkisToplam = n;
  webAkisGonderilen = 0;
  webAkisDefter = s;
  webAkisAktif = true;
  webAkisHam = false;
  webAkisBitti = false;
  webAkisSon = millis();
  if (n == 0) {
    web_kullanici_liste = (s.length() > 0)
      ? "KAYITLI YOK (ham veri: " + s + ")"
      : "KAYITLI KULLANICI YOK (defter bos)";
    webAkisAktif = false;
  } else {
    web_kullanici_liste = "KAYITLI (" + String(n) + ") - listeleniyor";
  }
  ArduinoCloud.update();
}

  ArduinoCloud.update();
}

// Akışın bir adımını gönderir (loop() her 500 ms'de çağırır).
void webAkisAdim() {
  webAkisSon = millis();
  if (webAkisBitti) {   // son kayıt gönderildi; TAMAM mesajı ayrı adımda gönderilir
    web_kullanici_liste = "TAMAM - " + String(webAkisToplam) + " kayit goruntulendi";
    webAkisBitti = false;
    webAkisAktif = false;
    ArduinoCloud.update();
    return;
  }
  int e = webAkisDefter.indexOf('|');
  String kayit = (e < 0) ? webAkisDefter : webAkisDefter.substring(0, e);
  webAkisDefter = (e < 0) ? "" : webAkisDefter.substring(e + 1);
  if (kayit.length() > 0) {
    if (webAkisHam) {
      web_kullanici_liste = String(webAkisGonderilen + 1) + "/" + String(webAkisToplam) + " | " + kayit;
    } else {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a) {
        web_kullanici_liste = String(webAkisGonderilen + 1) + "/" + String(webAkisToplam) + " | " +
                              kayit.substring(0, a) + " | " + kayit.substring(a + 1, b) + " | " +
                              webTelGoster(kayit.substring(b + 1));
      }
    }
  }
  webAkisGonderilen++;
  if (webAkisGonderilen >= webAkisToplam || webAkisDefter.length() == 0)
    webAkisBitti = true;   // son kayıt gönderildi → TAMAM'ı bir sonraki adımda gönder
  ArduinoCloud.update();
}

void onWebKullaniciListeChange() {
  String ham = web_kullanici_liste;
  ham.trim();
  // Cihaz akış mesajları ("KAYITLI (", "1/3 |", "TAMAM -" vb.) cloud'dan geri
  // yansıyınca bu callback de tetiklenir. Bunlar komut değildir → akışı bozmadan yok say.
  bool komut =
       ham == "liste" || ham.startsWith("liste ") || ham == "listeS" ||
       ham.startsWith("sil ") ||
       ham == "aktif" || ham.startsWith("aktif ") ||
       ham == "ham" || ham.startsWith("ham ");
  if (!komut) return;
  webAkisAktif = false;   // yeni komut geldi → devam eden akışı durdur
  if (ham == "liste" || ham.startsWith("liste ")) {
    if (ham.startsWith("liste ")) {
      String hane = webNorm(ham.substring(6));
      if (hane.length() >= 1 && hane.length() <= 4) {
        web_kullanici_liste = webMetinSinirla(webHaneKayitListesi(hane), 190);   // hane bazlı
        ArduinoCloud.update();
        return;
      }
    }
    webAkisBasla();   // liste → Messenger akışı (her kullanıcı ayrı mesaj)
    return;
  }
  if (ham == "listeS") {
    web_kullanici_liste = webMetinSinirla(webKayitlilarListesi(), 190);
    ArduinoCloud.update();
    return;
  }
  if (ham.startsWith("sil ")) {
    String tel = webNorm(ham.substring(4));
    if (tel.length() != 11) {
      web_kullanici_liste = "Hatali numara (11 hane girin)";
      ArduinoCloud.update();
      return;
    }
    if (!webTelKayitli(tel)) {
      web_kullanici_liste = "Kayit yok";
      ArduinoCloud.update();
      return;
    }
    webHaneTelSil(tel);
    webOturumTelSil(tel);
    web_kullanici_liste = webTelGoster(tel) + " silindi";
    ArduinoCloud.update();
    return;
  }
  if (ham == "aktif" || ham.startsWith("aktif ")) {
    String hane = "";
    if (ham.startsWith("aktif ")) hane = webNorm(ham.substring(6));
    if (hane.length() >= 1 && hane.length() <= 4)
      web_kullanici_liste = webMetinSinirla(webAktiflerListesi(hane), 190);   // hane bazlı
    else
      web_kullanici_liste = webMetinSinirla(webAktiflerListesi(), 190);
    ArduinoCloud.update();
    return;
  }
  if (ham == "ham" || ham.startsWith("ham ")) {
    String hane = "";
    if (ham.startsWith("ham ")) hane = webNorm(ham.substring(4));
    if (hane.length() >= 1 && hane.length() <= 4) {
      web_kullanici_liste = webMetinSinirla(webHamHaneListesi(hane), 190);   // hane bazlı
      ArduinoCloud.update();
    } else {
      webAkisBaslaHam();   // ham → Messenger akışı (ham kayıtlar tek tek, kırpma yok)
    }
    return;
  }
  if (ham.length() < 5) return;
  int a = ham.indexOf(';');
  int b = ham.indexOf(';', a + 1);
  if (a > 0 && b > a) {
    String hane = webNorm(ham.substring(0, a));
    String ad = ham.substring(a + 1, b);
    ad.replace(';', ' '); ad.replace('|', ' '); ad.trim();
    String tel = webNorm(ham.substring(b + 1));
    if (hane.length() >= 1 && ad.length() >= 2 && tel.length() == 11) {
      if (webHaneEkle(hane, ad, tel)) {
        web_kullanici_liste = webMetinSinirla(webKayitlilarListesi(), 190);
      } else {
        web_kullanici_liste = "Hane dolu !";
      }
    } else {
      return;
    }
  } else {
    return;
  }
  ArduinoCloud.update();
}

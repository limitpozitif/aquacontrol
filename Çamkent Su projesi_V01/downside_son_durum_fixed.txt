void addAlarm(const char* msg) {
  for (int i = 0; i < alarmCount; i++) {
    if (strcmp(alarmList[i], msg) == 0)
      return;
  }
  if (alarmCount < MAX_ALARM) {
    strncpy(alarmList[alarmCount], msg, 99);
    alarmList[alarmCount][99] = '\0';
    alarmCount++;
  }
}

void clearAlarm(const char* msg) {
  for (int i = 0; i < alarmCount; i++) {
    if (strcmp(alarmList[i], msg) == 0) {
      for (int j = i; j < alarmCount - 1; j++)
        strcpy(alarmList[j], alarmList[j + 1]);
      alarmCount--;
      return;
    }
  }
}

float clampf(float x, float a, float b) {
  if (x < a) return a;
  if (x > b) return b;
  return x;
}

float voltsToBar(float V, float V4, float V20) {
  if (fabs(V20 - V4) < 0.0001f)
    return 0;
  float ratio = (V - V4) / (V20 - V4);
  float bar = ratio * SENSOR_MAX_BAR;
  return clampf(bar, 0, SENSOR_MAX_BAR);
}

int wifiToPercent(int rssi) {
  if (rssi <= -100) return 0;
  if (rssi >= -50)  return 100;
  return 2 * (rssi + 100);
}

// Nextion'dan gelen veri akışı: 0xFF (end marker/gürültü) ve 0x65
// (dokunmatik olay, 7 byte) yok sayılır, tanınan kontrol ID'leri
// (0x01..0x06) için 4 byte little-endian değer okunur.
void readNextion() {
  static uint32_t lastRead = 0;
  if (millis() - lastRead < 50) return;
  lastRead = millis();

  while (nex.available() > 0) {
    int first = nex.peek();
    if (first < 0) break;

    if (first == 0xFF) {         // end marker / gürültü
      nex.read();
      continue;
    }

    if (first == 0x65) {         // dokunmatik olay → 7 byte atla
      if (nex.available() < 7) break;
      for (int i = 0; i < 7; i++) nex.read();
      continue;
    }

    if (first < 0x01 || first > 0x06) {  // tanınmayan byte → yeniden senkron
      nex.read();
      continue;
    }

    if (nex.available() < 5) break;      // eksik frame, bir sonraki tura bırak

    uint8_t controlId = nex.read();
    uint32_t value = 0;
    for (int i = 0; i < 4; i++) {
      value |= ((uint32_t)nex.read()) << (i * 8);
    }

    switch (controlId) {
      case 0x01:  // site_alt_limit progress bar
        site_alt_limit = value / 10.0f;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x02:  // site_ust_limit progress bar
        site_ust_limit = value / 10.0f;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x03:  // page1 yüklendi → değerleri gönder
        buffer1[0] = '\0';
        buffer2[0] = '\0';
        snprintf(buffer1, sizeof(buffer1), "page1.x0.val=%d", int(site_alt_limit * 10.0f));
        snprintf(buffer2, sizeof(buffer2), "page1.x1.val=%d", int(site_ust_limit * 10.0f));
        nextionSend(buffer1);
        nextionSend(buffer2);
        buffer1[0] = '\0';
        buffer2[0] = '\0';
        snprintf(buffer1, sizeof(buffer1), "page1.h0.val=%d", int(site_alt_limit * 10.0f));
        snprintf(buffer2, sizeof(buffer2), "page1.h1.val=%d", int(site_ust_limit * 10.0f));
        nextionSend(buffer1);
        nextionSend(buffer2);
        nextionSend(kuyu_start ? "page1.c0.val=1" : "page1.c0.val=0");
        nextionSend(hidr_start ? "page1.c1.val=1" : "page1.c1.val=0");
        break;

      case 0x04:
        sys_start = bool(value);
        sistem_start = sys_start;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x05:
        kuyu_start = bool(value);
        ckuyu_start = kuyu_start;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      case 0x06:
        hidr_start = bool(value);
        chidr_start = hidr_start;
        prefsNeedsSave = true;
        lastPrefsChange = millis();
        break;

      default:
        break;
    }
  }
}

void nextionUpdate() {
  static uint32_t last = 0;

  if (millis() - last < 1000)
    return;
  last = millis();

  /* ===== WIFI ===== */
  static int lastWifiPercent = -1;
  static int lastCloudWifiPercent = -1;
  if (WiFi.status() == WL_CONNECTED) {
    nextionSetPco("page0.ccomm_ok", 2016);
    nextionSetBco("page0.ccomm_ok", 0);
    wifiRssi = WiFi.RSSI();
    wifiPercent = wifiToPercent(wifiRssi);

    if (wifiPercent != lastWifiPercent) {
      lastWifiPercent = wifiPercent;
      char buffer[30];
      snprintf(buffer, sizeof(buffer), "page0.jwifi.val=%d", wifiPercent);
      nextionSend(buffer);
    }
    if (cloudStarted && abs(wifiPercent - lastCloudWifiPercent) >= 5) {
      wifi_guc = wifiPercent;
      lastCloudWifiPercent = wifiPercent;
    }
    clearAlarm("Wifi Baglanti Yok");
  } else {
    if (lastWifiPercent != 0) {
      lastWifiPercent = 0;
      nextionSend("page0.jwifi.val=0");
    }
    nextionSetPco("page0.ccomm_ok", 63488);
    nextionSetBco("page0.ccomm_ok", 65504);
    addAlarm("Wifi Baglanti Yok");
  }

  s = (int)(site_bar * 10);
  l = (int)(line_bar * 10);

  buffer1[0] = '\0';
  buffer2[0] = '\0';
  snprintf(buffer1, sizeof(buffer1), "page0.xsite_bar.val=%d", s);
  nextionSend(buffer1);
  snprintf(buffer2, sizeof(buffer2), "page0.xline_bar.val=%d", l);
  nextionSend(buffer2);
  snprintf(buffer1, sizeof(buffer1), "page0.xfire_bar.val=%d", int(yan_basinc * 10));
  nextionSend(buffer1);

  /* ===== FAZ ===== */
  if (io.fazOK) {
    nextionSetPco("page0.cfaz_ok", 2016);
    nextionSetBco("page0.cfaz_ok", 0);
    clearAlarm("Faz Hatasi ! Sistem calisamaz...");
  } else {
    nextionSetPco("page0.cfaz_ok", 63488);
    nextionSetBco("page0.cfaz_ok", 65504);
    addAlarm("Faz Hatasi ! Sistem calisamaz...");

    addAlarm("Yangin tanki bos !");

  } else {
    clearAlarm("Yangin tanki bos !");
    nextionSetPco("page0.jfire_tank", 1024);
    nextionSetPco("page0.cfired_sens", 2016);
    nextionSetBco("page0.cfired_sens", 0);
    if (io.fireUst == 0) {
      nextionSend("page0.jfire_tank.val=50");
      nextionSetPco("page0.cfireu_sens", 63488);
      nextionSetBco("page0.cfireu_sens", 65504);
    } else {
      nextionSend("page0.jfire_tank.val=100");
      nextionSetPco("page0.cfireu_sens", 2016);

    }
  }

  /* ===== SITE TANKI ===== */
  if (io.siteAlt) {
    nextionSetPco("page0.jsite_tank", 1024);
    nextionSetPco("page0.cs_sens", 2016);
    nextionSetBco("page0.cs_sens", 0);

    nextionSetPco("page0.cs_sens", 63488);

    nextionSetBco("page0.cinv_ok", 0);

    nextionSetPco("page0.cinv_ok", 63488);

void readInputs() {
  io.fireAlt    = !mcp.digitalRead(0);
  io.fireUst    = !mcp.digitalRead(1);
  io.siteAlt    = !mcp.digitalRead(2);
  io.fazOK      = !mcp.digitalRead(3);
  io.inverterOK = !mcp.digitalRead(4);
  io.kuyu       = !mcp.digitalRead(5);
}

// Tüm çıkışları beklenen duruma göre zorla yazar. mcpSetup recovery
// sırasında çağrıldığında diğer kanalların (kuyu/hidrant) OFF'ta
// kilitlenmesini önler.
void forceWriteOutputs() {
  mcp.digitalWrite(inv_en, expected_inv);
  mcp.digitalWrite(yedek_en, expected_yedek);
  mcp.digitalWrite(kuyu_en, expected_kuyu);
  mcp.digitalWrite(hidr_en, expected_hidr);
  last_inv   = expected_inv;
  last_yedek = expected_yedek;
  last_kuyu  = expected_kuyu;
  last_hidr  = expected_hidr;
}

void mcpSetup() {
  if (!mcp.begin_I2C(MCP_ADDR)) {
    addAlarm("MCP Baglanti Hatasi");
    return;
  }

  for (int i = 0; i <= 5; i++) {
    mcp.pinMode(i, INPUT);
  }

  mcp.pinMode(inv_en, OUTPUT);
  mcp.pinMode(yedek_en, OUTPUT);
  mcp.pinMode(kuyu_en, OUTPUT);
  mcp.pinMode(hidr_en, OUTPUT);

  // Önce güvenli durum (hepsi kapalı)
  mcp.digitalWrite(inv_en, HIGH);
  mcp.digitalWrite(yedek_en, HIGH);
  mcp.digitalWrite(kuyu_en, HIGH);
  mcp.digitalWrite(hidr_en, HIGH);

  forceWriteOutputs();
}

void siteControl() {
  static bool pumpRunning = false;
  static uint32_t mismatchTimer = 0;
  static uint32_t lastReadback = 0;
  static bool recoveryActive = false;
  static uint32_t recoveryStart = 0;

  if (!hard_test) {

    // SYSTEM OFF
    if (!io.siteAlt || !io.fazOK || !sys_start) {
      expected_inv = true;
      expected_yedek = true;
      pumpRunning = false;
      mismatchTimer = 0;
      recoveryActive = false;
    }
    else {

      // HYSTERESIS
      if (!pumpRunning && site_bar < site_alt_limit)
        pumpRunning = true;
      else if (pumpRunning && site_bar > site_ust_limit)
        pumpRunning = false;

      // OUTPUT SELECT
      if (pumpRunning) {
        if (io.inverterOK) {
          expected_inv = false;
          expected_yedek = true;
        } else {
          expected_inv = true;
          expected_yedek = false;
        }
      } else {
        expected_inv = true;
        expected_yedek = true;
      }
    }

    // WRITE ONLY IF CHANGED
    if (expected_inv != last_inv) {
      last_inv = expected_inv;
      mcp.digitalWrite(inv_en, expected_inv);
    }
    if (expected_yedek != last_yedek) {
      last_yedek = expected_yedek;
      mcp.digitalWrite(yedek_en, expected_yedek);
    }

    // READBACK 1 sn
    if (millis() - lastReadback >= 1000) {
      lastReadback = millis();

      bool inv_state = mcp.digitalRead(inv_en);
      bool yedek_state = mcp.digitalRead(yedek_en);

      bool mismatch =
        (inv_state != expected_inv) ||
        (yedek_state != expected_yedek);

      // Recovery sonrası kontrol
      if (recoveryActive) {
        if (millis() - recoveryStart >= 500) {
          recoveryActive = false;
          if (mismatch) {
            addAlarm("MCP OUTPUT FROZEN!");
            Rst_sys = true;
          } else {
            mismatchTimer = 0;
          }
        }
      }
      else {
        if (mismatch) {
          if (mismatchTimer == 0) {
            mismatchTimer = millis();
            addAlarm("in site control");
          }
          else if (millis() - mismatchTimer >= 10000) {
            clearAlarm("in site control");
            mcpSetup();
            recoveryActive = true;
            recoveryStart = millis();
          }
        } else {
          mismatchTimer = 0;
        }
      }
    }
  }
  else if (hard_test && io.fazOK) {
    expected_inv = !hard_inv;
    expected_yedek = !hard_yedek;

    if (expected_inv != last_inv) {
      last_inv = expected_inv;
      mcp.digitalWrite(inv_en, expected_inv);
    }
    if (expected_yedek != last_yedek) {
      last_yedek = expected_yedek;
      mcp.digitalWrite(yedek_en, expected_yedek);
    }
  }
}

void kuyuControl() {
  static uint32_t mismatchTimer = 0;
  static uint32_t lastReadback = 0;
  static bool recoveryActive = false;
  static uint32_t recoveryStart = 0;

  if (!hard_test) {

    // SYSTEM OFF / SAFETY STOP
    if (!io.kuyu || !io.fazOK || !kuyu_start) {
      expected_kuyu = true;
      mismatchTimer = 0;
      recoveryActive = false;
    }
    else {
      // CONTROL LOGIC
      if (!io.fireUst && io.fazOK && kuyu_start) {
        expected_kuyu = false;
      } else {
        expected_kuyu = true;
      }
    }

    // WRITE ONLY IF CHANGED
    if (expected_kuyu != last_kuyu) {
      last_kuyu = expected_kuyu;
      mcp.digitalWrite(kuyu_en, expected_kuyu);
    }

    // READBACK 1 sn
    if (millis() - lastReadback >= 1000) {
      lastReadback = millis();

      bool kuyu_state = mcp.digitalRead(kuyu_en);
      bool mismatch = (kuyu_state != expected_kuyu);

      if (recoveryActive) {
        if (millis() - recoveryStart >= 500) {
          recoveryActive = false;
          if (mismatch) {
            addAlarm("MCP KUYU OUTPUT FROZEN!");
            Rst_sys = true;
          } else {
            mismatchTimer = 0;
          }
        }
      }
      else {
        if (mismatch) {
          if (mismatchTimer == 0) {
            mismatchTimer = millis();
            addAlarm("in kuyu control");
          }
          else if (millis() - mismatchTimer >= 10000) {
            clearAlarm("in kuyu control");
            mcpSetup();
            recoveryActive = true;
            recoveryStart = millis();
          }
        } else {
          mismatchTimer = 0;
        }
      }
    }
  }
  else if (hard_test && io.fazOK) {
    expected_kuyu = !hard_kuyu;

    if (expected_kuyu != last_kuyu) {
      last_kuyu = expected_kuyu;
      mcp.digitalWrite(kuyu_en, expected_kuyu);
    }
  }
}

void hidrantControl() {
  static uint32_t hidrantTimer = 0;
  static bool hidrantCycle = false;
  static bool pumpRunning = false;
  static uint32_t mismatchTimer = 0;
  static uint32_t lastReadback = 0;
  static bool recoveryActive = false;
  static uint32_t recoveryStart = 0;

  if (!hard_test) {

    // SAFETY STOP
    if (!io.fireAlt || !io.fazOK || !hidr_start) {
      expected_hidr = true;
      hidrantTimer = millis();
      hidrantCycle = false;
      pumpRunning = false;
      mismatchTimer = 0;
      recoveryActive = false;
    }
    else {

      // PRESSURE VALID MODE
      bool basincGecerli = ((millis() - lastYanBasincUpdate) < 20000UL) && remoteDataOK;

      if (basincGecerli) {
        // Histerezis
        if (yan_basinc < 2.0f)
          pumpRunning = true;
        else if (yan_basinc >= 2.5f)
          pumpRunning = false;
        expected_hidr = pumpRunning ? false : true;
        hidrantTimer = millis();
        hidrantCycle = false;
      }
      else {
        // TIME BASED MODE
        uint32_t elapsed = millis() - hidrantTimer;
        if (!hidrantCycle) {
          if (elapsed >= 120000UL) {
            hidrantCycle = true;
            hidrantTimer = millis();
            expected_hidr = false;
          }
          else {
            expected_hidr = true;
          }
        }
        else {
          if (elapsed >= 120000UL) {
            hidrantCycle = false;
            hidrantTimer = millis();
            expected_hidr = true;
          }
          else {
            expected_hidr = false;
          }
        }
      }
    }

    // WRITE ONLY IF CHANGED
    if (expected_hidr != last_hidr) {
      last_hidr = expected_hidr;
      mcp.digitalWrite(hidr_en, expected_hidr);
    }

    // READBACK 1 sn
    if (millis() - lastReadback >= 1000UL) {
      lastReadback = millis();
      bool hidr_state = mcp.digitalRead(hidr_en);
      bool mismatch = (hidr_state != expected_hidr);

      if (recoveryActive) {
        if (millis() - recoveryStart >= 500UL) {
          recoveryActive = false;
          if (mismatch) {
            addAlarm("MCP HIDRANT OUTPUT FROZEN!");
            Rst_sys = true;
          } else {
            mismatchTimer = 0;
          }
        }
      }
      else {
        if (mismatch) {
          if (mismatchTimer == 0) {
            mismatchTimer = millis();
            addAlarm("in hidr control");
          }
          else if (millis() - mismatchTimer >= 10000UL) {
            clearAlarm("in hidr control");
            mcpSetup();
            recoveryActive = true;
            recoveryStart = millis();
          }
        } else {
          mismatchTimer = 0;
        }
      }
    }
  }
  else if (hard_test && io.fazOK) {
    expected_hidr = !hard_hidr;
    if (expected_hidr != last_hidr) {
      last_hidr = expected_hidr;
      mcp.digitalWrite(hidr_en, expected_hidr);
    }
  }
}

// 3 kere üst üste I2C ping'ine cevap gelmezse sistem resetlenir.
void i2cPing() {
  static uint8_t mcpFail = 0;
  static uint8_t adsFail = 0;

  Wire.beginTransmission(MCP_ADDR);
  if (Wire.endTransmission() == 0) {
    mcpFail = 0;
  } else {
    mcpFail++;
    if (mcpFail >= 3) {
      addAlarm("MCP I2C Koptu!");
      Rst_sys = true;
    }
  }

  Wire.beginTransmission(ADS_ADDR);
  if (Wire.endTransmission() == 0) {
    adsFail = 0;
  } else {
    adsFail++;
    if (adsFail >= 3) {
      addAlarm("ADS I2C Koptu!");
      Rst_sys = true;
    }
  }
}

void udpSend(bool resetUpside) {
  if (!udpStarted) return;

  static uint32_t lastSend = 0;
  if (millis() - lastSend < 1000) return;
  lastSend = millis();

  txPacket.deviceID = DEVICE_ID;
  txPacket.reset_upside = resetUpside;
  txPacket.counter = udpCounter++;

  udp.beginPacket(broadcastIP, udpPort);
  udp.write((uint8_t*)&txPacket, sizeof(txPacket));
  udp.endPacket();
}

void udpReceive() {
  int packetSize = udp.parsePacket();
  if (packetSize == sizeof(rxPacket)) {
    udp.read((uint8_t*)&rxPacket, sizeof(rxPacket));
    if (rxPacket.deviceID == 1) {
      remoteDataOK = true;
      if (fabs(yan_basinc - rxPacket.firePressure) >= 0.1f) {
        yan_basinc = rxPacket.firePressure;
      }
      lastYanBasincUpdate = millis();
      remoteLastSeen = millis();
    }
  }

  if (millis() - remoteLastSeen > 5000) {
    remoteDataOK = false;
    if (yan_basinc != 0.0f) {
      yan_basinc = 0.0f;
    }
  }
}

void setup() {
  Serial.begin(9600);
  delay(50);

  Wire.setClock(100000);
  Wire.setTimeOut(100);
  Wire.begin(21, 22);
  delay(50);

  /* ---- NEXTION ---- */
  nex.begin(NEXTION_BAUD, SERIAL_8N1, NEXTION_RX, NEXTION_TX);
  delay(500);
  while (nex.available()) nex.read();  // Buffer temizle

  pinMode(LED_BUILTIN, OUTPUT);
  digitalWrite(LED_BUILTIN, LOW);

  mcpSetup();

  /* ---- ADS ---- */
  if (!ads.begin(ADS_ADDR)) {
    addAlarm("ADS Baglanti Hatasi");
  } else {
    ads.setGain(GAIN_ONE);
  }

  /* ---- RS485 ---- */
  Serial2.begin(9600, SERIAL_8N1, 16, 17);
  delay(100);

  /* ---- CLOUD / PREFERENCES ---- */
  initProperties();
  loadFromPrefs();
  delay(50);

  Rst_sys = false;
}

/* ===================================================== */
/* LOOP */
/* ===================================================== */

void loop() {
  static bool cloudInited = false;
  static bool ledState = false;
  static uint32_t lastLedToggle = 0;

  // LED 500ms aralıklarla yanıp söner
  if (millis() - lastLedToggle >= 500) {
    lastLedToggle = millis();
    ledState = !ledState;
    digitalWrite(LED_BUILTIN, ledState);
  }

  /* ===== CLOUD (ilk 3 saniyeden sonra başlat) ===== */
  if (!cloudInited && millis() > 3000) {
    cloudInited = true;
    ArduinoCloud.begin(ArduinoIoTPreferredConnection);
    setDebugMessageLevel(2);
  }

  // Kütüphane her döngüde update() çağrılmasını ister; bağlantı kurulamazsa
  // bu çağrı yeniden bağlanmayı da yönetir.
  if (cloudInited) {
    ArduinoCloud.update();

    if (!cloudStarted && ArduinoCloud.connected()) {
      cloudStarted = true;
      configTime(GMT_OFFSET_SEC, DAYLIGHT_OFFSET_SEC, NTP_SERVER);
    }
  }

  /* ===== RESET (cloud'dan bağımsız çalışır) ===== */
  if (Rst_sys) {
    Rst_sys = false;
    rcnt++;
    if (hardreset) rcnt = 0;
    hardreset = false;
    prefs.begin("settings", false);
    prefs.putLong("rcnt", rcnt);
    prefs.end();
    delay(1000);
    ESP.restart();
  }

  /* ===== IO POLL (100ms) ===== */
  if (millis() - tIo > 100) {
    tIo = millis();

    static uint32_t tI2cPing = 0;
    if (millis() - tI2cPing >= 5000) {  // Her 5 saniyede bir
      tI2cPing = millis();
      i2cPing();
    }

    readInputs();

    inv_stat   = expected_inv;
    yedek_stat = expected_yedek;
    kuyu_stat  = expected_kuyu;
    hidr_stat  = expected_hidr;
    faz_stat   = io.fazOK;
  }

  /* ===== ADC (500ms, non-blocking zamanlayıcılı) ===== */
  static uint32_t tAdc = 0;
  if (millis() - tAdc >= 500) {
    tAdc = millis();

    int16_t adc0 = ads.readADC_SingleEnded(1);
    int16_t adc1 = ads.readADC_SingleEnded(3);

    float V0 = adc0 * 0.125f / 1000.0f;
    float V1 = adc1 * 0.125f / 1000.0f;

    line_bar = voltsToBar(V0, V4_CH0, V20_CH0);
    site_bar = voltsToBar(V1, V4_CH1, V20_CH1);

    if (fabs(line_bar - last_line_bar) >= 0.5) {
      line_basinc = line_bar;
      last_line_bar = line_bar;
    }
    if (fabs(site_bar - last_site_bar) >= 0.5) {
      site_basinc = site_bar;
      last_site_bar = site_bar;
    }
  }

  /* ===== CLOUD STATUS VAR SYNC (1s) ===== */
  static uint32_t tCloud = 0;
  if (cloudStarted && (millis() - tCloud > 1000)) {
    tCloud = millis();

    if (site_su_mevcut != io.siteAlt) site_su_mevcut = io.siteAlt;
    if (fire_alt != io.fireAlt) fire_alt = io.fireAlt;
    if (fire_ust != io.fireUst) fire_ust = io.fireUst;

    if (inv_stat != inv_run) inv_run = inv_stat;
    if (yedek_stat != yedek_run) yedek_run = yedek_stat;
    if (kuyu_stat != kuyu_run) kuyu_run = kuyu_stat;
    if (hidr_stat != hidr_run) hidr_run = hidr_stat;
    if (faz_stat != enerji_ok) enerji_ok = faz_stat;

    if (millis() - lastliveUpdate > 3000) {  // 3 saniyede bir
      lastliveUpdate = millis();
      isLive = !isLive;
    }
  }

  /* ===== NEXTION ===== */
  nextionUpdate();
  updateStatusDisplay();
  readNextion();

  /* ===== KONTROL ===== */
  udpReceive();
  siteControl();
  kuyuControl();
  hidrantControl();
  saveToPrefsDelayed();

  /* ===== WIFI / UDP BAŞLAT ===== */
  if (WiFi.status() == WL_CONNECTED) {
    IPAddress ip = WiFi.localIP();
    if (ip != IPAddress(0, 0, 0, 0)) {
      String newIP = ip.toString();
      if (local_ip != newIP) {
        local_ip = newIP;
      }
      if (!udpStarted) {
        udp.begin(udpPort);
        broadcastIP = ip;
        broadcastIP[3] = 255;
        udpStarted = true;
      }
    }
  }

  if (udp_send_need) {
    udp_send_need = false;
    bool doReset = up_reset;
    up_reset = false;
    udpSend(doReset);
  }
}

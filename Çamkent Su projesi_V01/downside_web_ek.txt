================================================================================
Bu dosya DOWNSIDE'a eklenecek YENİ koddur. downside_son_durum_fixed.txt
DEĞİŞTİRİLMEZ; aşağıdaki bloklar kopyalanıp ilgili yerlere yapıştırılır.

GÖREVLERİ
- Panel dosyalarını (LittleFS) servis eder
- Giriş kapısı: telefon no + şifre (Model A, her cihazdan)
  şifre = HMAC-SHA256(gizli anahtar, "camkent:"+numara) → ilk 6 hane
- Engelli numara listesi (cloud: web_engelli)
- Toplu iptal (cloud: web_sifirla → anahtar sıfırlanır)
- Şifre üretimi (cloud: web_sifre_sor → web_sifre_cevap)
- /data: kendi verisi + UPSIDE'ın UDP broadcast'inden gelen veriyi birleştirir

YENİ BAĞIMLILIKLAR (Arduino IDE Kütüphane Yöneticisi'nden kurulur)
- ArduinoJson (benoitblanchon/ArduinoJson) — sürüm 6.x
- WebServer, LittleFS, HTTPClient, mbedtls: ESP32 çekirdeğiyle gelir
  (HTTPClient yalnızca derleme sırası için kalır — UPSIDE verisi UDP ile alınır,
   HTTP ile çekim YOKTUR)

CLOUD'DA EKLENECEK 4 DEĞİŞKEN (DOWNSIDE Thing'ine)
- web_sifre_sor   String   WRITE + callback
- web_sifre_cevap String   READ
- web_engelli     String   WRITE (virgülle ayrılmış numaralar)
- web_sifirla     bool     WRITE + callback
  (Arduino Cloud Editörü'nde her değişkeni "callback" aktif ekleyin,
   thingProperties.h yeniden üretilir; ona dokunmazsınız)

YAPILANDIRMA (aşağıda doldurulacaklar)
- WEB_PORT       : 8080 (modemde tek port buraya yönlendirilir)

NOT: UPSIDE'ın IP adresi YAPILANDIRILMAZ — veri UDP broadcast (port 4210) ile
gelir, downside kendi WiFi'ında UPSIDE paketini otomatik bulur. UPSIDE_IP sabiti
KALDIRILDI (HTTP çekimi yok).

UDP ALICI YAPISI (upside'ın UdptxPacket'iyle BİREBİR AYNI olmalıdır — sıra ve
tipler DEĞİŞTİRİLMEZ, iki taraf da packed DEĞİL):

{

  uint16_t crc16;         // Madde 5: paket bütünlük CRC'si (son eleman!)

CRC (Madde 5): crc16 alanı struct'ın SON elemanıdır. Gönderen (upside) CRC'yi
offsetof(UdptxPacket,crc16) kadar olan baytlardan hesaplar; alıcı (downside)
aynı şekilde doğrular. CRC tutmayan paketler yok sayılır. udpReceive'de
rxFirePressureOK = rxPacket.firePressureOK; rxSitePressureOK = rxPacket.sitePressureOK;
alınır; 5 sn paket gelmezse her ikisi de false yapılır. basincGecerli koşuluna
&& rxFirePressureOK eklenmiştir (hidrant, upside yangın sensörü ölüyken zaman
tabanlı 2/2 döngüsüne düşer).

GEREKLİ GLOBAL (udpReceive'de kullanılır):
uint32_t upsideLastSeen = 0;   // son başarılı UDP paketinin zamanı (upOnline için)

udpReceive() devreye eklenir: gelen paket rxPacket'i doldurur,
upsideLastSeen = millis(); 5 sn görülmezse remoteDataOK=false.

String webNorm(const String &s) {          // numarayı sadece rakama çevirir
  String r;
  for (unsigned i = 0; i < s.length(); i++) {
    char c = s[i];
    if (c >= '0' && c <= '9') r += c;
  }
  return r;
}

void webSecretInit() {                     // gizli anahtar: ilk açılışta üret
  prefs.begin("settings", false);
  if (!prefs.isKey("web_secret")) {
    char hex[33];
    for (int i = 0; i < 16; i++) snprintf(hex + i * 2, 3, "%02x", (uint8_t)esp_random());
    prefs.putString("web_secret", hex);
  }
  webSecret = prefs.getString("web_secret", "");
  prefs.end();
}

void webSecretSifirla() {                  // tüm şifreleri geçersiz kılar
  prefs.begin("settings", false);

  prefs.putString("web_secret", hex);

  prefs.end();
  for (int i = 0; i < MAX_SES; i++) webOturum[i].sid[0] = 0;
  webSesKaydet();
}

String webSifreUret(const String &telNorm) {
  if (webSecret.length() == 0) return "";
  String msg = "camkent:" + telNorm;
  unsigned char out[32] = {0};
  mbedtls_md_context_t ctx;
  const mbedtls_md_info_t *info = mbedtls_md_info_from_type(MBEDTLS_MD_SHA256);
  mbedtls_md_init(&ctx);
  if (mbedtls_md_setup(&ctx, info, 1) != 0) { mbedtls_md_free(&ctx); return ""; }
  mbedtls_md_hmac_starts(&ctx, (const unsigned char *)webSecret.c_str(), webSecret.length());
  mbedtls_md_hmac_update(&ctx, (const unsigned char *)msg.c_str(), msg.length());
  mbedtls_md_hmac_finish(&ctx, out);
  mbedtls_md_free(&ctx);
  uint32_t v = ((uint32_t)out[0] << 16) | ((uint32_t)out[1] << 8) | out[2];
  char buf[8];
  snprintf(buf, sizeof(buf), "%06u", (unsigned)(v % 1000000));
  return String(buf);
}

bool webEngelliMi(const String &telNorm) { // web_engelli: "0532...,0533..." (virgülle)
  String list = web_engelli;
  int start = 0;
  while (start <= (int)list.length()) {
    int i = list.indexOf(',', start);
    String one = (i < 0) ? list.substring(start) : list.substring(start, i);
    if (webNorm(one) == telNorm) return true;
    if (i < 0) break;
    start = i + 1;
  }
  return false;
}

// Numara herhangi bir hanede kayıtlı mı (silinen/taşınan kullanıcı girişini engeller).
bool webTelKayitli(const String &telNorm) {
  String s = webKayitOku();
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a && kayit.substring(b + 1) == telNorm) return true;
    }
    if (e < 0) break;
    pos = e + 1;
  }
  return false;
}

// Kayıtlı (onaylı) kullanıcıları SAYFALAR halinde döndürür (cloud ~256 bayt sınırı için).
// Her sayfa en fazla KAYITLI_PAGE_ADET kullanıcı içerir; 40-50 kullanıcıda bile
// cevap 190 baytı aşmaz. Sayfalar: liste → 1, liste2 → 2, liste3 → 3, ...
String webKayitlilarSayfa(int sayfa) {
  #define KAYITLI_PAGE_ADET 5
  String s = webKayitOku();
  String kayitlar[96];
  int n = 0;
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0 && n < 96) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a) kayitlar[n++] = kayit;
    }
    if (e < 0) break;
    pos = e + 1;
  }
  if (n == 0) {
    if (s.length() > 0)
      return "KAYITLI YOK (ham veri: " + s + ")";  // tanı: defter içeriğini göster
    return "KAYITLI KULLANICI YOK (defter bos)";
  }
  int toplam = (n + KAYITLI_PAGE_ADET - 1) / KAYITLI_PAGE_ADET;
  if (sayfa < 1) sayfa = 1;
  if (sayfa > toplam) sayfa = toplam;
  int bas = (sayfa - 1) * KAYITLI_PAGE_ADET;
  int bit = bas + KAYITLI_PAGE_ADET;
  if (bit > n) bit = n;
  String satir = "KAYITLI (" + String(n) + ") sayfa " + String(sayfa) + "/" + String(toplam) + "\r\n";
  for (int i = bas; i < bit; i++) {
    String kayit = kayitlar[i];
    int a = kayit.indexOf(';');
    int b = kayit.indexOf(';', a + 1);
    satir += kayit.substring(0, a) + " | " + kayit.substring(a + 1, b) + " | " +
             webTelGoster(kayit.substring(b + 1)) + "\r\n";
  }
  return satir;
}

// Belirli bir hanedeki kayıtları "hane | ad | tel" satırlarıyla döndürür.

String webHaneKayitListesi(const String &hane) {
  String s = webKayitOku();
  String satir = "";
  int n = 0;
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a && kayit.substring(0, a) == hane) {
        n++;
        satir += kayit.substring(0, a) + " | " +
                 kayit.substring(a + 1, b) + " | " + webTelGoster(kayit.substring(b + 1)) + "\r\n";
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
  if (n == 0) return "HANE " + hane + ": kayit yok";
  return "HANE " + hane + " (" + String(n) + " kayit)\r\n" + satir;
}

// Şu an paneli açık olan (anlık aktif) kullanıcıların listesi.
// Panel her 2 sn'de /data çeker ve her istekte webOturumVar() "son"u tazeler;
// son etkinlik 10 sn içindeyse o kullanıcı ANLIK AKTİF sayılır.
// haneFiltre doluysa yalnızca o haneye kayıtlı aktif numaralar gösterilir ("aktif 27").
String webAktiflerListesi(const String &haneFiltre = "") {
  #define AKTIF_ESIK_MS 10000UL
  String unique[16];
  bool online[16];
  uint8_t uc = 0;
  uint32_t now = millis();
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0 || webOturum[i].tel.length() < 7) continue;
    bool var = false;
    for (uint8_t j = 0; j < uc; j++) if (unique[j] == webOturum[i].tel) { var = true; break; }
    if (!var && uc < 16) { unique[uc] = webOturum[i].tel; online[uc] = false; uc++; }
  }
  int onSay = 0;
  for (uint8_t j = 0; j < uc; j++) {
    uint32_t sonGiris = 0;
    for (int i = 0; i < MAX_SES; i++)
      if (webOturum[i].sid[0] != 0 && webOturum[i].tel == unique[j] && webOturum[i].son > sonGiris)
        sonGiris = webOturum[i].son;
    online[j] = ((uint32_t)(now - sonGiris) <= AKTIF_ESIK_MS);
    if (online[j] && (haneFiltre.length() == 0 || webTelHane(unique[j]) == haneFiltre)) onSay++;
  }
  if (onSay == 0) {
    if (haneFiltre.length() > 0) return "HANE " + haneFiltre + ": AKTIF KULLANICI YOK";
    return "SU AN AKTIF KULLANICI YOK";
  }
  String r = "ANLIK AKTIF KULLANICILAR (" + String(onSay) + ")\r\n";
  uint8_t no = 0;
  for (uint8_t j = 0; j < uc; j++)
    if (online[j] && (haneFiltre.length() == 0 || webTelHane(unique[j]) == haneFiltre)) {
      no++;
      r += String(no) + ". " + webTelGoster(unique[j]) + " - ONLINE\r\n";
    }
  return r;
}

// Telefon numarasının kayıtlı olduğu hane; kayıtlı değilse boş döner.
String webTelHane(const String &tel) {
  String s = webKayitOku();

  while (pos <= (int)s.length()) {

    if (kayit.length() > 0) {

}

String webYeniSid(const String &telNorm) { // 16 hanelik rastgele oturum kimliği
  static uint32_t seq = 0;
  char id[17];
  for (int i = 0; i < 16; i++) {
    uint32_t r = esp_random();
    id[i] = "0123456789abcdef"[r % 16];
  }
  id[16] = 0;
  int slot = -1;
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0) { slot = i; break; }
  }
  if (slot < 0) { slot = (int)(seq++ % MAX_SES); } // doluysa en eskiyi bul
  memcpy(webOturum[slot].sid, id, 17);
  webOturum[slot].tel = telNorm;
  webOturum[slot].son = millis();
  webSesKaydet();
  return String(id);
}

String webSidFromCookie() {                // Cookie başlığından sid çeker
  String c = webSrv.header("Cookie");
  int p = c.indexOf("sid=");
  if (p < 0) return "";
  p += 4;
  int e = c.indexOf(';', p);
  String v = (e < 0) ? c.substring(p) : c.substring(p, e);
  v.trim();
  return v;
}

bool webOturumVar() {                      // geçerli oturum var mı
  String sid = webSidFromCookie();
  if (sid.length() != 16) return false;
  uint32_t now = millis();
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] != 0 && strcmp(webOturum[i].sid, sid.c_str()) == 0) {
      if (now - webOturum[i].son > SES_TTL_MS) { webOturum[i].sid[0] = 0; return false; }
      webOturum[i].son = now;
      return true;
    }
  }
  return false;
}

// Oturum tablosunu NVS'e kaydeder (reboot sonrası oturumlar korunur).
void webSesKaydet() {
  String s = "";
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0) continue;
    if (s.length() > 0) s += "|";
    s += String(webOturum[i].sid) + ";" + webOturum[i].tel;
  }
  Preferences p;
  p.begin("settings", false);
  p.putString("web_ses", s);
  p.end();
}

// Kayıtlı oturumları geri yükler (son etkinlik şimdi kabul edilir; TTL yine işler).
void webSesYukle() {
  Preferences p;
  p.begin("settings", true);
  String s = p.getString("web_ses", "");
  p.end();
  int pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    int a = kayit.indexOf(';');
    if (kayit.length() > 16 && a == 16) {
      String sid = kayit.substring(0, 16);
      String tel = kayit.substring(a + 1);
      for (int i = 0; i < MAX_SES; i++) {
        if (webOturum[i].sid[0] == 0) {
          memcpy(webOturum[i].sid, sid.c_str(), 17);
          webOturum[i].tel = tel;
          webOturum[i].son = millis();
          break;
        }
      }
    }
    if (e < 0) break;
    pos = e + 1;
  }
}

// Bir numaranın tüm aktif oturumlarını düşürür (kullanıcı silme/engelleme).
void webOturumTelSil(const String &telNorm) {
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] != 0 && webOturum[i].tel == telNorm) {
      webOturum[i].sid[0] = 0;
      webOturum[i].tel = "";
    }
  }
  webSesKaydet();
}

void webKilitTemiz(uint32_t ip) {
  for (int i = 0; i < MAX_KAZAN; i++) if (webKilit[i].ip == ip) { webKilit[i].fail = 0; webKilit[i].kilid = 0; }
}

bool webKilitli(uint32_t ip) {
  for (int i = 0; i < MAX_KAZAN; i++) {
    if (webKilit[i].ip == ip) return (millis() - webKilit[i].kilid < LOCK_SURESI);
  }
  return false;
}

void webKilitArttir(uint32_t ip) {
  for (int i = 0; i < MAX_KAZAN; i++) {
    if (webKilit[i].ip == ip) {
      if (millis() - webKilit[i].kilid >= LOCK_SURESI) webKilit[i].fail = 0;
      webKilit[i].fail++;
      if (webKilit[i].fail >= LOCK_SINIR) webKilit[i].kilid = millis();
      return;
    }
  }
  for (int i = 0; i < MAX_KAZAN; i++) {
    if (webKilit[i].ip == 0) { webKilit[i].ip = ip; webKilit[i].fail = 1; return; }
  }
}

void webSrvFile(const char *path, const char *type) {
  if (!LittleFS.exists(path)) { webSrv.send(404, "text/plain", "not found"); return; }
  File f = LittleFS.open(path, "r");
  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.streamFile(f, type);
  f.close();
}

void webLogin() {
  uint32_t ip = (uint32_t)webSrv.client().remoteIP();
  if (webKilitli(ip)) {
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(429, "application/json", "{\"ok\":false,\"msg\":\"Cok fazla deneme. 10 dk sonra tekrar deneyin.\"}");
    return;
  }
  String tel = webNorm(webSrv.arg("tel"));
  String sif = webSrv.arg("sifre");
  if (tel.length() != 11 || sif.length() < 4) {
    webKilitArttir(ip);
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(403, "application/json", "{\"ok\":false,\"msg\":\"Hatali giris\"}");
    return;
  }
  bool engelli = webEngelliMi(tel);
  bool kayitli = webTelKayitli(tel);
  bool dogru   = (webSifreUret(tel) == sif);
  if (engelli || !kayitli || !dogru) {
    webKilitArttir(ip);
    webSrv.sendHeader("Cache-Control", "no-store");
    webSrv.send(403, "application/json", "{\"ok\":false,\"msg\":\"Hatali giris\"}");
    return;
  }
  webKilitTemiz(ip);
  String sid = webYeniSid(tel);
  webSrv.sendHeader("Cache-Control", "no-store");
  webSrv.sendHeader("Set-Cookie", "sid=" + sid + "; Path=/; Max-Age=2592000");
  webSrv.send(200, "application/json", "{\"ok\":true}");
}

}

// Kayıt defterindeki toplam kullanıcı (kayıt) sayısı.
int webKayitSayisi() {
  String s = webKayitOku();
  int n = 0, pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) n++;
    if (e < 0) break;
    pos = e + 1;
  }
  return n;
}

// Şu an aktif (10 sn içinde paneli kullanan) benzersiz kullanıcı sayısı.
int webAktifSayisi() {
  const uint32_t ESIK = 10000UL;
  String unique[16];
  uint8_t uc = 0;
  uint32_t now = millis();
  for (int i = 0; i < MAX_SES; i++) {
    if (webOturum[i].sid[0] == 0 || webOturum[i].tel.length() < 7) continue;
    bool var = false;
    for (uint8_t j = 0; j < uc; j++) if (unique[j] == webOturum[i].tel) { var = true; break; }
    if (!var && uc < 16) { unique[uc] = webOturum[i].tel; uc++; }
  }
  int onSay = 0;
  for (uint8_t j = 0; j < uc; j++) {
    uint32_t sonGiris = 0;
    for (int i = 0; i < MAX_SES; i++)
      if (webOturum[i].sid[0] != 0 && webOturum[i].tel == unique[j] && webOturum[i].son > sonGiris)
        sonGiris = webOturum[i].son;
    if ((uint32_t)(now - sonGiris) <= ESIK) onSay++;
  }
  return onSay;
}

// Dashboard'dan doğrudan kullanıcı ekleme: web_kayit_ekle = "hane;ad;tel".
// Messenger akışı: "liste"/"ham" komutları kayıtları tek tek mesaj olarak gönderir.
// (loop() içinde ArduinoCloud.update() ile birlikte:
//   if (webAkisAktif && (uint32_t)(millis() - webAkisSon) >= WEB_AKIS_BEKLE_MS) webAkisAdim();)
void webAkisBasla() {
  String s = webKayitOku();
  int n = 0, pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) n++;
    if (e < 0) break;
    pos = e + 1;
  }
  webAkisToplam = n;
  webAkisGonderilen = 0;
  webAkisDefter = s;
  webAkisAktif = true;
  webAkisHam = false;
  webAkisBitti = false;
  webAkisSon = millis();
  if (n == 0) {
    web_kayit_ekle = (s.length() > 0)
      ? "KAYITLI YOK (ham veri: " + s + ")"
      : "KAYITLI KULLANICI YOK (defter bos)";
    webAkisAktif = false;
  } else {
    web_kayit_ekle = "KAYITLI (" + String(n) + ") - listeleniyor";
  }
  ArduinoCloud.update();
}

void webAkisBaslaHam() {
  String s = webKayitOku();
  int n = 0, pos = 0;
  while (pos <= (int)s.length()) {
    int e = s.indexOf('|', pos);
    String kayit = (e < 0) ? s.substring(pos) : s.substring(pos, e);
    if (kayit.length() > 0) n++;
    if (e < 0) break;
    pos = e + 1;
  }
  webAkisToplam = n;
  webAkisGonderilen = 0;
  webAkisDefter = s;
  webAkisAktif = true;
  webAkisHam = true;
  webAkisBitti = false;
  webAkisSon = millis();
  if (n == 0) {
    web_kayit_ekle = (s.length() > 0)
      ? "KAYITLI YOK (ham veri: " + s + ")"
      : "KAYITLI KULLANICI YOK (defter bos)";
    webAkisAktif = false;
  } else {
    web_kayit_ekle = "NVS webkayit ham (" + String(n) + " kayit) - listeleniyor";
  }
  ArduinoCloud.update();
}

void webAkisAdim() {
  webAkisSon = millis();
  if (webAkisBitti) {   // son kayıt gönderildi; TAMAM mesajı ayrı adımda gönderilir
    web_kayit_ekle = "TAMAM - " + String(webAkisToplam) + " kayit goruntulendi";
    webAkisBitti = false;
    webAkisAktif = false;
    ArduinoCloud.update();
    return;
  }
  int e = webAkisDefter.indexOf('|');
  String kayit = (e < 0) ? webAkisDefter : webAkisDefter.substring(0, e);
  webAkisDefter = (e < 0) ? "" : webAkisDefter.substring(e + 1);
  if (kayit.length() > 0) {
    if (webAkisHam) {
      web_kayit_ekle = String(webAkisGonderilen + 1) + "/" + String(webAkisToplam) + " | " + kayit;
    } else {
      int a = kayit.indexOf(';');
      int b = kayit.indexOf(';', a + 1);
      if (a > 0 && b > a) {
        web_kayit_ekle = String(webAkisGonderilen + 1) + "/" + String(webAkisToplam) + " | " +
                         kayit.substring(0, a) + " | " + kayit.substring(a + 1, b) + " | " +
                         webTelGoster(kayit.substring(b + 1));
      }
    }
  }
  webAkisGonderilen++;
  if (webAkisGonderilen >= webAkisToplam || webAkisDefter.length() == 0)
    webAkisBitti = true;   // son kayıt gönderildi → TAMAM'ı bir sonraki adımda gönder
  ArduinoCloud.update();
}

void onWebKayitEkleChange() {
  String ham = web_kayit_ekle;
  ham.trim();
  if (ham.startsWith("liste") && !ham.startsWith("liste ")) {   // liste / liste2 / liste3 ... → sayfalı kayıtlılar
    int sayfa = 1;
    String sn = ham.substring(5);
    sn.trim();
    if (sn.length() > 0 && sn.length() < 4) {
      int p = sn.toInt();
      if (p > 0) sayfa = p;
    }
    web_kayit_ekle = webMetinSinirla(webKayitlilarSayfa(sayfa), 190);
    ArduinoCloud.update();
    return;
  }
  if (ham.startsWith("liste ")) {            // "liste 27" → o hanedeki kayıtlar (kırpma yok)
    String hane = webNorm(ham.substring(6));
    if (hane.length() >= 1 && hane.length() <= 4) {
      web_kayit_ekle = webHaneKayitListesi(hane);
      ArduinoCloud.update();
      return;
    }
  }
  if (ham == "aktif" || ham.startsWith("aktif ")) {   // aktif / aktif 27 → aktif kullanıcılar
    String hane = "";
    if (ham.startsWith("aktif ")) hane = webNorm(ham.substring(6));
    if (hane.length() >= 1 && hane.length() <= 4)
      web_kayit_ekle = webAktiflerListesi(hane);       // hane bazlı: tamamı, kırpma yok
    else
      web_kayit_ekle = webMetinSinirla(webAktiflerListesi(), 190);
    ArduinoCloud.update();
    return;
  }
  if (ham == "ham" || ham.startsWith("ham ")) {        // ham / ham 27 → ham defter
    String hane = "";
    if (ham.startsWith("ham ")) hane = webNorm(ham.substring(4));
    if (hane.length() >= 1 && hane.length() <= 4) {
      web_kayit_ekle = webHamHaneListesi(hane);        // hane bazlı: tamamı, kırpma yok
      ArduinoCloud.update();
    } else {
      webAkisBaslaHam();                               // ham → Messenger akışı (kırpma yok)
    }
    return;
  }
  if (ham.startsWith("sil ")) {              // "sil 053322233444" → kaydı sil
    String tel = webNorm(ham.substring(4));
    if (tel.length() != 11) {
      web_kayit_ekle = "Hatali numara (11 hane girin)";
      ArduinoCloud.update();
      return;
    }
    if (!webTelKayitli(tel)) {
      web_kayit_ekle = "Kayit yok";
      ArduinoCloud.update();
      return;
    }
    webHaneTelSil(tel);
    webOturumTelSil(tel);
    web_kayit_ekle = webTelGoster(tel) + " silindi";
    ArduinoCloud.update();
    return;
  }
  if (ham.length() < 5) return;              // yarım yazım sırasında sessiz kal
  int a = ham.indexOf(';');
  int b = ham.indexOf(';', a + 1);
  if (a > 0 && b > a) {
    String hane = webNorm(ham.substring(0, a));
    String ad = ham.substring(a + 1, b);
    ad.replace(';', ' '); ad.replace('|', ' '); ad.trim();
    String tel = webNorm(ham.substring(b + 1));
    if (hane.length() >= 1 && ad.length() >= 2 && tel.length() == 11) {
      if (webHaneEkle(hane, ad, tel)) {
        web_kayit_ekle = "Eklendi";
        web_kullanici_liste = webMetinSinirla(webKayitlilarSayfa(1), 190);
      } else {
        web_kayit_ekle = "Hane dolu !";
      }
    } else {
      return;                                // eksik/hatalı ad veya tel → sessiz kal
    }
  } else {
    return;                                  // ';' yok → sessiz kal
  }
  ArduinoCloud.update();
}

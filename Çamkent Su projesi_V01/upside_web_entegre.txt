// CRC16 (Modbus polysiyle aynı, A001/FFFF) — paket bütünlük doğrulaması için.
// crc16 alanı struct'ın SON elemanı olduğundan, CRC hesabı crc16 alanına kadar
// (offsetof kadar) yapılır ve paket reddedilmeden önce karşılaştırılır.
uint16_t crc16(const uint8_t* data, size_t len) {
  uint16_t crc = 0xFFFF;
  for (size_t i = 0; i < len; i++) {
    crc ^= data[i];
    for (uint8_t j = 0; j < 8; j++) {
      if (crc & 1) crc = (crc >> 1) ^ 0xA001;
      else crc >>= 1;
    }
  }
  return crc;
}

// UDP paketlerini dinler: DOWNSIDE'ın sıfırlama isteğini alır ve uygular;
// 3 sn paket gelmezse bağlantı kopuk sayılır. CRC doğrulaması başarısız
// paketler yok sayılır; sıfırlama isteği TEK pakette uygulanır.
void udpReceive(){
    int packetSize = udp.parsePacket();
    if(packetSize == sizeof(rxPacket)){
        udp.read((uint8_t*)&rxPacket,sizeof(rxPacket));


        // aşağıdan veri varsa al
        if(rxPacket.deviceID == 2){
            uint16_t calc = crc16((const uint8_t*)&rxPacket, offsetof(UdprxPacket, crc16));
            if(rxPacket.crc16 == calc){
                if(rxPacket.reset_upside){
                    ESP.restart();
                }
                remoteDataOK = true;
                remoteLastSeen = millis();
            }
        }
    }
    // bağlantı kontrolü
    if(millis() - remoteLastSeen > 3000)
    {
        remoteDataOK = false;
    }
}

// Frame durum kodunu okunabilir metne çevirir (OK/INVALID/TIMEOUT/UNKNOWN).
const char* frameStatusText(uint8_t status)
{
  switch (status)
  {
    case FS_OK:       return "OK";
    case FS_INVALID:   return "INVALID";
    case FS_TIMEOUT:   return "TIMEOUT";
    default:           return "UNKNOWN";
  }
}

// Yazma (0x06) cevabının slave, register ve değerle birebir + CRC doğruluğunu kontrol eder.
bool is_valid_write_response(uint8_t *buf, uint8_t len, uint8_t slave, uint16_t reg, uint16_t value)
{
  if (len < 8) return false;
  if (buf[0] != slave) return false;
  if (buf[1] != 0x06) return false;

  uint16_t regRx = ((uint16_t)buf[2] << 8) | buf[3];
  uint16_t valRx = ((uint16_t)buf[4] << 8) | buf[5];

  if (regRx != reg) return false;
  if (valRx != value) return false;

  uint16_t crcCalc = modbus_crc16(buf, len - 2);
  uint16_t crcRx = buf[len - 2] | (buf[len - 1] << 8);

  return (crcCalc == crcRx);
}

// =========================
// CRC
// =========================
// Modbus CRC-16 (A001 polinomu) hesabı yapar.
uint16_t modbus_crc16(uint8_t *buf, uint16_t len)
{
  uint16_t crc = 0xFFFF;
  for (uint16_t i = 0; i < len; i++)
  {
    crc ^= buf[i];
    for (uint8_t j = 0; j < 8; j++)
    {
      if (crc & 1)
      {
        crc >>= 1;
        crc ^= 0xA001;
      }
      else crc >>= 1;
    }
  }
  return crc;
}

// =========================
// REQUEST FRAME
// =========================
// RS485 üzerinden Modbus okuma isteği gönderir (DE/RE pinini yönlendirerek).
void modbusReadRequest(uint8_t slave, uint8_t func, uint16_t reg, uint16_t qty)
{
  while (RS485Serial.available()){
    RS485Serial.read();
  }
  uint8_t frame[8];

  frame[0] = slave;
  frame[1] = func;
  frame[2] = highByte(reg);
  frame[3] = lowByte(reg);
  frame[4] = highByte(qty);
  frame[5] = lowByte(qty);

  uint16_t crc = modbus_crc16(frame, 6);

  frame[6] = lowByte(crc);
  frame[7] = highByte(crc);

  digitalWrite(MAX485_DE_RE, HIGH);
  delayMicroseconds(80);

  RS485Serial.write(frame, 8);
  RS485Serial.flush();

  delayMicroseconds(200);

  digitalWrite(MAX485_DE_RE, LOW);
}

// RS485 üzerinden tek register yazma (0x06) isteği gönderir.
void modbusWriteSingleRequest(uint8_t slave, uint16_t reg, uint16_t value)
{
  while (RS485Serial.available()){
    RS485Serial.read();
  }
  uint8_t frame[8];

  frame[0] = slave;
  frame[1] = 0x06;
  frame[2] = highByte(reg);
  frame[3] = lowByte(reg);
  frame[4] = highByte(value);
  frame[5] = lowByte(value);

  uint16_t crc = modbus_crc16(frame, 6);

  frame[6] = lowByte(crc);
  frame[7] = highByte(crc);

  digitalWrite(MAX485_DE_RE, HIGH);
  delayMicroseconds(80);

  RS485Serial.write(frame, 8);
  RS485Serial.flush();

  delayMicroseconds(200);

  digitalWrite(MAX485_DE_RE, LOW);
}

// =========================
// NON-BLOCKING RX FIX
// =========================
// RS485'ten gelen cevabı non-blocking toplar; 20 ms sessizlikte frame tamamlandı sayar.
bool modbus_read_response(uint8_t *buf, uint8_t &len)
{
    static uint32_t lastByteTime = 0;
    while (RS485Serial.available() && len < 32)
    {
        buf[len++] = RS485Serial.read();
        lastByteTime = millis();
    }

    if (len == 0)    {
        return false;
    }

    if (millis() - lastByteTime > 20)    {
        if (len < 5)        {
            len = 0;
            return false;
        }
        return true;
    }
    return false;
}

// =========================
// VALIDATION
// =========================
// Okuma cevabının slave + fonksiyon + CRC doğruluğunu kontrol eder.
bool is_valid_response(uint8_t *buf, uint8_t len, uint8_t slave, uint8_t func)
{
  if (len < 7) return false;
  if (buf[0] != slave) return false;
  if (buf[1] != func) return false;

  uint16_t crcCalc = modbus_crc16(buf, len - 2);
  uint16_t crcRx = buf[len - 2] | (buf[len - 1] << 8);

  return (crcCalc == crcRx);
}

// TANK VOLUME
// =========================
// Tank seviye sensörü değerini tonaja çevirir (silindir hacmi).
float calcTankVolumeTon(uint8_t id, float level_mm)
{
  if (id < 1 || id > 5) return 0;

  TankParametre t = tanklar[id];

  float h = t.yukseklik_mm - (level_mm - t.sensor_yukseklik_mm);
  if (h < 0) h = 0;
  if (h > t.yukseklik_mm) h = t.yukseklik_mm;

  float r = (t.cap_mm / 1000.0) / 2.0;
  float hm = h / 1000.0;

  return 3.14159 * r * r * hm;
}

// Tank seviyesini yüzde doluluk oranına çevirir.
float calcTankPercent(uint8_t id, float level_mm)
{
    if (id < 1 || id > 5) return 0;
    TankParametre t = tanklar[id];
    float h = t.yukseklik_mm - (level_mm - t.sensor_yukseklik_mm);
    if (h < 0) h = 0;
    if (h > t.yukseklik_mm) h = t.yukseklik_mm;
    return (h * 100.0f) / t.yukseklik_mm;
}

// Basınç sensörü voltajını (0.5..4.5V) 0..16 bar aralığına çevirir.
float voltageToBar(float v){
  const float V_MIN = 0.5;
  const float V_MAX = 4.5; 
  float bar = (v - V_MIN) * (16.0 / (V_MAX - V_MIN));
  if (bar < 0) bar = 0;
  if (bar > 16) bar = 16;
  return bar;
}

// Basınç sensörlerini okur; 3 örneğin ortalamasını alır ve site/yangın basıncını günceller.
// Kablo kopması / 4mA altı (0.5V'un altı, kablosuz 0V) okumada ilgili kanalın
// OK bayrağı hemen düşürülür; geçerli örnek gelince tekrar set edilir.
void readPressureSensors()
{
    static unsigned long lastRead = 0;
    static float sumSite = 0.0f;
    static float sumYangin = 0.0f;
    static uint8_t sampleCount = 0;
    static bool siteValid = true;
    static bool fireValid = true;

    unsigned long now = millis();
    if (now - lastRead < 250) return;
    lastRead = now;

    if (!ads1115Ok) return;

    int16_t adc0 = ads1115.readADC_SingleEnded(0);
    float voltage0 = adc0 * 0.1875 / 1000.0;
    if (voltage0 < 0.3f) siteValid = false;
    sumSite += voltageToBar(voltage0);

    int16_t adc1 = ads1115.readADC_SingleEnded(1);
    float voltage1 = adc1 * 0.1875 / 1000.0;
    if (voltage1 < 0.3f) fireValid = false;
    sumYangin += voltageToBar(voltage1);

    sampleCount++;

    if(sampleCount >= 3){
        if (siteValid) {
            siteBasinc = sumSite / 3.0f;
            if(siteBasinc != site_basinc) site_basinc = siteBasinc;
            siteSensorOK = true;
            lastSiteSensorUpdate = now;
        } else {
            siteSensorOK = false;
        }
        if (fireValid) {
            yanginBasinc = sumYangin / 3.0f;
            if(yanginBasinc != yangin_basinc) yangin_basinc = yanginBasinc;
            fireSensorOK = true;
            lastFireSensorUpdate = now;
        } else {
            fireSensorOK = false;
        }
        pressureReady = siteSensorOK && fireSensorOK; 
        
        char buf[80];
        snprintf(buf, sizeof(buf), "Site: %.1f Fire: %.1f", siteBasinc, yanginBasinc);
        pressures = buf;
        
        sumSite = 0.0f;
        sumYangin = 0.0f;
        sampleCount = 0;
        siteValid = true;
        fireValid = true;
    }
}

// Sensör sağlık kontrolü (Madde 1): 10 sn içinde geçerli örnek gelmemişse
// kanal bayat sayılır; kontrol kapıları (siteControl/fireControl) bu bayrağa
// göre pompayı durdurur. fire1_start ekrandan (fire1_local) geliyorsa yangın
// basıncı bayatken büyük yangın pompası sabit 40 Hz blind sürülür — bu durum
// yalnızca alarm mesajı üretir, kapama kodu fire1Control içindedir.
void sensorSaglikKontrol() {
  static bool siteAlarmOn = false;
  static bool fireAlarmOn = false;
  unsigned long now = millis();

  if (!ads1115Ok || (now - lastSiteSensorUpdate) > 10000UL) {
    siteSensorOK = false;
    if (!siteAlarmOn) {
      siteAlarmOn = true;
      messages = "UYARI: Site sensoru bayat!";
    }
  } else {
    siteAlarmOn = false;
  }

  if (!ads1115Ok || (now - lastFireSensorUpdate) > 10000UL) {
    fireSensorOK = false;
    if (!fireAlarmOn) {
      fireAlarmOn = true;
      if (fire1_local && fire1_start) {
        messages = "UYARI: Sensor bayat! Acil 40Hz";
      } else {
        messages = "UYARI: Yangin sensoru bayat!";
      }
    }
  } else {
    fireAlarmOn = false;
  }

  pressureReady = siteSensorOK && fireSensorOK;
}

// =================================================
// INVERTER WRITE FUNCTIONS (STRUCT İÇİ DEĞİŞKENLERLE)
// =================================================
// İnverter frekansını ayarlamak için yazma kuyruğuna register/değer ekler.
void inverterSetFrequency(uint8_t slave, float hz)
{
    uint16_t newValue = (uint16_t)(hz * 10.0f + 0.5f);

    if (inverterState[slave].writePending || (inverterState[slave].lastFreqReg == newValue && !inverterState[slave].first))
        return;
    
    inverterState[slave].writeReg    = 0x2136;
    inverterState[slave].writeValue  = newValue;
    inverterState[slave].writePending = true;
}

// İnverteri çalıştırmak için RUN komutunu (0x2135 = 0x000F) yazma kuyruğuna ekler.
void inverterRun(uint8_t slave)
{
    if(inverterState[slave].writePending || (inverterState[slave].running == true))
        return;
   
    inverterState[slave].writeReg = 0x2135;
    inverterState[slave].writeValue = 0x000F;
    inverterState[slave].writePending = true; 
}

// =================================================
// MAIN SCHEDULER (TÜM DEĞİŞKENLER DÜZELTİLDİ)
// =================================================
// Ana Modbus zamanlayıcısı: sıfırlama (öncelik 1), yazma (öncelik 2) ve okuma
// işlemlerini sırayla başlatır; cevap/timed-out durumunu yönetir.
void ModbusScheduler()
{
    switch (mbState)
    {
        // =================================================
        // IDLE STATE
        // =================================================
        case MB_IDLE:
        {
            // 1) RESET KONTROLÜ (Öncelik 1)
            for (uint8_t i = 6; i <= 8; i++) {
                if (inverterState[i].in_reset && !inverterState[i].writePending) {
                    
                    if (inverterState[i].resetStep == RESET_STOP) {
                        inverterState[i].resetExpectedValue = 0x0006;
                        inverterState[i].resetted = false;
                        inverterState[i].writeAckOk = false;
                        inverterState[i].timeOut = false;
                        modbusWriteSingleRequest(i, 0x2135, inverterState[i].resetExpectedValue);
                        
                        inverterState[i].resetTimer = millis();
                        mbResponseStart = millis();
                        mbLen = 0;
                        mbJob = JOB_RESET;
                        activeSlave = i;
                        mbState = MB_WAIT_RESPONSE;
                        inverterState[i].resetStep = RESET_ON;
                        return;
                    }
                    else if (inverterState[i].resetStep == RESET_ON && (millis() - inverterState[i].resetTimer > 200)) {
                        inverterState[i].resetExpectedValue = 0x0080;
                        inverterState[i].writeAckOk = false;
                        inverterState[i].timeOut = false;
                        modbusWriteSingleRequest(i, 0x2135, inverterState[i].resetExpectedValue);
                        
                        inverterState[i].resetTimer = millis();
                        mbResponseStart = millis();
                        mbLen = 0;
                        mbJob = JOB_RESET;
                        activeSlave = i;
                        mbState = MB_WAIT_RESPONSE;
                        inverterState[i].resetStep = RESET_OFF;
                        return;
                    }
                    else if (inverterState[i].resetStep == RESET_OFF && (millis() - inverterState[i].resetTimer > 200)) {
                        inverterState[i].resetExpectedValue = 0x0006;
                        inverterState[i].writeAckOk = false;
                        inverterState[i].timeOut = false;
                        modbusWriteSingleRequest(i, 0x2135, inverterState[i].resetExpectedValue);
                        
                        inverterState[i].resetTimer = millis();
                        mbResponseStart = millis();
                        mbLen = 0;
                        mbJob = JOB_RESET;
                        activeSlave = i;
                        mbState = MB_WAIT_RESPONSE;
                        inverterState[i].resetFinalCommand = true;
                        return;
                    }
                }
            }

            // 2) YAZMA KONTROLÜ (Öncelik 2)
            for (uint8_t i = 6; i <= 8; i++) {
                if (inverterState[i].writePending && !inverterState[i].in_reset) {
                    inverterState[i].writeAckOk = false;
                    inverterState[i].timeOut = false;
                    
                    modbusWriteSingleRequest(i, inverterState[i].writeReg, inverterState[i].writeValue);
                    
                    mbResponseStart = millis();
                    mbLen = 0;
                    mbJob = JOB_WRITE;
                    activeSlave = i;
                    mbState = MB_WAIT_RESPONSE;
                    return; 
                }
            }

            // 3) NORMAL OKUMA İŞLEMLERİ (Öncelik 3)
            if (mbJob == JOB_NONE && (millis() - mbLastRequest > 100)) {
                mbLastRequest = millis();
                
                if (phase == PHASE_TANK) {
                    modbusReadRequest(tankIndex, 0x04, 0x0000, 1);
                } else {
                    if (inverterIndex < 6 || inverterIndex > 8) {
                        nextInverter();
                        return;
                    }
                    
                    if (inverterState[inverterIndex].in_reset) {
                        nextInverter();
                        return;
                    }

                    switch(invStep) {
                        case INV_STATUS:     modbusReadRequest(inverterIndex, 0x03, 0x0C81, 1); break;
                        case INV_FAULT:      modbusReadRequest(inverterIndex, 0x03, 0x1BD1, 1); break;
                        case INV_FREQUENCY:  modbusReadRequest(inverterIndex, 0x03, 0x0C82, 1); break;
                        case INV_CURRENT:    modbusReadRequest(inverterIndex, 0x03, 0x0C84, 1); break;
                        case INV_VOLTAGE:    modbusReadRequest(inverterIndex, 0x03, 0x0C87, 1); break;
                    }
                }

                mbResponseStart = millis();
                mbLen = 0;
                mbJob = JOB_READ;
                mbState = MB_WAIT_RESPONSE;
            }
            break;
        }

        // =================================================
        // RESPONSE STATE
        // =================================================
        case MB_WAIT_RESPONSE:
        {
            if (modbus_read_response(mbBuffer, mbLen)) {
                bool ok = false;
                frameStatus = FS_INVALID;
                
                // --- RESET CEVABI ---
                if (mbJob == JOB_RESET) {
                    uint8_t rs = activeSlave;
                    if (mbLen >= 8 && mbBuffer[0] == rs && mbBuffer[1] == 0x06) {
                        ok = is_valid_write_response(mbBuffer, mbLen, rs, 0x2135, inverterState[rs].resetExpectedValue);
                        if (ok) {
                            inverterState[rs].writeAckOk = true;
                            if (inverterState[rs].resetFinalCommand) {
                                inverterState[rs].first = true;
                                ArduinoCloud.update();
                                inverterState[rs].running = false;
                                inverterState[rs].in_reset = false;
                                inverterState[rs].resetted = true;
                                inverterState[rs].lastFreqReg = 0;
                                
                                rstmsg = "RESET COMPLETE SLAVE " + String(rs);
                                inverterState[rs].resetFinalCommand = false;
                                inverterState[rs].resetStep = RESET_IDLE;
                                messages = rstmsg;
                            }
                        } else {
                            inverterState[rs].writeAckOk = false;
                            inverterState[rs].in_reset = false;
                            inverterState[rs].resetStep = RESET_IDLE;
                            inverterState[rs].resetFinalCommand = false;
                            
                            rstmsg = "RESET FAIL SLAVE " + String(rs);
                            //messages = rstmsg;
                        }
                    } else {
                        inverterState[rs].writeAckOk = false;
                        inverterState[rs].in_reset = false;
                        inverterState[rs].resetStep = RESET_IDLE;
                        inverterState[rs].resetFinalCommand = false;
                        
                        rstmsg = "RESET CORRUPT RESP FROM SLAVE " + String(rs);
                        //messages = rstmsg;
                    }
                    
                    mbJob = JOB_NONE;
                    mbState = MB_IDLE;
                    break; 
                }

                // --- WRITE CEVABI ---
                if (mbJob == JOB_WRITE) {
                    uint8_t ws = activeSlave;
                    if (mbLen >= 8 && mbBuffer[0] == ws && mbBuffer[1] == 0x06) {
                        ok = is_valid_write_response(mbBuffer, mbLen, ws, inverterState[ws].writeReg, inverterState[ws].writeValue);
                        inverterState[ws].writeAckOk = ok; 
                        if (ok) {
                            if (inverterState[ws].writeReg == 0x2136) {
                                inverterState[ws].lastFreqReg = inverterState[ws].writeValue;
                            }
                            if (inverterState[ws].writeReg == 0x2135 && inverterState[ws].writeValue == 0x000F) {
                                inverterState[ws].running = true;
                                //writemsg = "INV " + String(ws) + " RUN OK";
                            }  
                        } else {
                            //writemsg = "INV " + String(ws) + " " + String(inverterState[ws].writeReg, HEX) + " JOB_WRITE FAIL";
                        }
                    } else {
                        inverterState[ws].writeAckOk = false;
                        //writemsg = "WRITE CORRUPT RESP FROM SLAVE " + String(ws);
                    }
                    
                    //messages = writemsg;
                    inverterState[ws].writePending = false; 
                    mbJob = JOB_NONE;
                    mbState = MB_IDLE;
                    break;
                }
              
                // --- TANK VE INVERTER READ CEVABI ---
                if (phase == PHASE_TANK){
                      ok = is_valid_response(mbBuffer, mbLen, tankIndex, 0x04);
                      if (ok){
                            frameStatus = FS_OK;
                            uint16_t level = ((uint16_t)mbBuffer[3] << 8) | mbBuffer[4];
                            if (!tankLevels[tankIndex].filtInit) {
                                  tankLevels[tankIndex].tankLevelSm = level;
                                  tankLevels[tankIndex].filtInit = true;
                            } else {
                                  tankLevels[tankIndex].tankLevelSm = (uint16_t)(tankLevels[tankIndex].tankLevelSm * 0.88f + level * 0.12f);
                            }
                            uint16_t filt = tankLevels[tankIndex].tankLevelSm;
                            tankLevels[tankIndex].tankLevel  = level;
                            tankLevels[tankIndex].tankTonaj  = round(calcTankVolumeTon(tankIndex, filt) * 10.0f) / 10.0f;
                            tankLevels[tankIndex].tankYuzde  = round(calcTankPercent(tankIndex, filt) * 10.0f) / 10.0f;
                      }else{
                            frameStatus = FS_INVALID;
                            tankLevels[tankIndex].tankLevel = 65535; 
                      }
                      nextTank();
                      if (tankCycleDone){
                            uint32_t toplam = 0;
                            uint8_t adet = 0;
                            // Önce geçerli tankları topla
                            for (uint8_t i = 2; i <= 5; i++){
                                  if (tankLevels[i].tankLevel != 65535){
                                        toplam += tankLevels[i].tankLevel;
                                        adet++;
                                  }
                            }
                            // En az bir tank okunmuşsa, okunamayanları ortalama ile doldur
                            if (adet > 0 && adet < 4) {
                                  uint16_t ortalama = toplam / adet;
                                  for (uint8_t i = 2; i <= 5; i++){
                                        if (tankLevels[i].tankLevel == 65535){
                                              tankLevels[i].tankLevel   = ortalama;
                                              tankLevels[i].tankLevelSm = ortalama;
                                              tankLevels[i].filtInit    = true;
                                              tankLevels[i].tankTonaj  = round(calcTankVolumeTon(i, ortalama) * 10.0f) / 10.0f;
                                              tankLevels[i].tankYuzde  = round(calcTankPercent(i, ortalama) * 10.0f) / 10.0f;
                                        }
                                  }
                            }
                            phase = PHASE_INVERTER;
                            tankCycleDone = false;
                            loopTime = millis();
                      }
                } else {   
                    ok = is_valid_response(mbBuffer, mbLen, inverterIndex, 0x03);
                    if (ok) {
                          frameStatus = FS_OK;
                          uint16_t value = ((uint16_t)mbBuffer[3] << 8) | mbBuffer[4];
                      
                          switch(invStep) {
                                case INV_STATUS:
                                      inverterState[inverterIndex].status=value;
                                      inverterState[inverterIndex].drv_ready = value & 0x0001;
                                      
                                      invStep = INV_FAULT;
                                break;
                                case INV_FAULT:
                                      inverterState[inverterIndex].fault=value;
                                      invStep = INV_FREQUENCY;
                                break;
                                case INV_FREQUENCY:
                                      inverterState[inverterIndex].frequency=value;
                                      invStep = INV_CURRENT;
                                break;
                                case INV_CURRENT:
                                      inverterState[inverterIndex].current=value;
                                      invStep = INV_VOLTAGE;
                                break;
                                case INV_VOLTAGE:
                                      inverterState[inverterIndex].voltage=value;
                                      invStep = INV_STATUS;
                                      nextInverter();
                                break;
                          }
                    } else {
                          frameStatus = FS_INVALID;
                          invStep = INV_STATUS;
                          nextInverter();
                    }
                }

                mbJob = JOB_NONE;
                mbState = MB_IDLE;
            }
            // --- TIMEOUT KONTROLÜ ---
            else if (millis() - mbResponseStart > 600) {
                while (RS485Serial.available()) {
                    RS485Serial.read();
                }
                mbLen = 0;
                
                if (mbJob == JOB_RESET) {
                      uint8_t rs = activeSlave;
                      inverterState[rs].timeOut = true;
                      inverterState[rs].writeAckOk = false;
                      inverterState[rs].in_reset = false; 
                      inverterState[rs].resetStep = RESET_IDLE;
                      inverterState[rs].resetFinalCommand = false;
                      
                      rstmsg = "RESET TIMEOUT SLAVE " + String(rs);
                } else if (mbJob == JOB_WRITE) {
                      uint8_t ws = activeSlave;
                      inverterState[ws].timeOut = true;
                      inverterState[ws].writeAckOk = false;
                      inverterState[ws].writePending = false; 
                      
                      //writemsg = "WRITE TIMEOUT " + String(ws);  
                } else {
                      frameStatus = FS_TIMEOUT;
                      if(phase == PHASE_TANK) {
                          tankLevels[tankIndex].tankLevel = 65535;
                          nextTank();
                      } else {
                          invStep = INV_STATUS;
                          nextInverter();
                      }
                }
                
                if (mbJob != JOB_READ) {
                    //messages = writemsg;
                }
                
                mbJob = JOB_NONE;
                mbState = MB_IDLE;
            }
            break; 
        }
    } 
}

// POMPA KONTROL SCHEDULER

String sitemsg ="";
// Site pompa kontrolü: koşullar uygunsa inverteri başlatır/frekans ayarlar,
// aksi halde resetler (1 saniyede bir çağrılır).
void siteControl()
{
    static uint32_t lastControl = 0;
    if (millis() - lastControl < 1000) return;
    lastControl = millis();

    if(site_start && siteSensorOK && ads1115Ok && !inverterState[6].in_reset && !inv6_fault){ 
          float hz = 0;
          if(inverterState[6].running && !inverterState[6].first){
               hz = site_pompa_kontrol_hesapla();
          }
              
          if (!inverterState[6].running && inverterState[6].resetted && inverterState[6].first && !inverterState[6].writePending) {
                sitemsg = "ilk frekans seti\r\n";
                inverterSetFrequency(6, 0.0f); 
                inverterState[6].first = false;
                return;
          } else if(inverterState[6].running && !inverterState[6].writePending) {
               sitemsg += "normal frekans\r\n"; 
               inverterSetFrequency(6, hz);
              return;
          } else if(!inverterState[6].running && inverterState[6].resetted && !inverterState[6].writePending){
               sitemsg += "run\r\n";
               inverterRun(6);
               return;
          } else {
               // Madde 2 (ilk açılış kilidi): Açılışta start zaten açıksa buraya düşülür.
               // Sürücü henüz resetlenmedi (resetted=false) → reset'e alınır. Böylece her
               // açılışta "reset → 0 Hz ilk frekans → run → rampa" sırası garanti olur;
               // sürücü EEPROM'daki son frekanstan kalkamaz ve fault/kilitliyse takılmaz.
               if (!inverterState[6].writePending) {
                    inverterReset(6);
                    resetSitePumpVariables = true;
               }
               return;
          }
    } else if(inverterState[6].running || inv6_fault || (!inverterState[6].resetted && !inverterState[6].in_reset)){
          sitemsg = "resete girdi\r\n";
          inverterReset(6);
          resetSitePumpVariables = true;
          return;      
    }
}

// Site pompa frekansını hesaplar: basınca göre hedef hız, adım adım rampa ile
// uygulanır; tank boş/uyku modunda 0 döner.
float site_pompa_kontrol_hesapla() {
    if (!ads1115Ok) return 0.0f; 
    if (site_min < 0.5f || site_max < 0.5f) return 0.0f; 

    const float MIN_PRESS = site_min;        
    const float MAX_PRESS = site_max;        
    const float MID_PRESS = (MIN_PRESS + MAX_PRESS) * 0.5f;
    const float MIN_FREQ  = 35.0f;
    const float MAX_FREQ  = 50.0f; 
    const float STEP      = 0.5f;
    const float DP_DEC    = +0.15f;   
    const float DP_INC    = -0.15f;   
    const float TANK_STOP = 30.0f;     
    const unsigned long SLEEP_TIME = 5000;

    float basinc = site_basinc;
    float tank = tankLevels[1].tankYuzde;

    static float mevcutHiz = MIN_FREQ;
    static float hedefHiz  = MIN_FREQ;
    static float oncekiBasinc = basinc;
    static unsigned long stableStart = 0;
    static bool sleeping = false;

    if (resetSitePumpVariables) {
        mevcutHiz = MIN_FREQ;
        hedefHiz = MIN_FREQ;
        stableStart = 0;
        sleeping = false;
        oncekiBasinc = basinc;
        resetSitePumpVariables = false; 
    }

    unsigned long now = millis();

    if ((tank < TANK_STOP) || siteforceZero) {
        sleeping = true;
        return 0.0f;
    }

    if (sleeping) {
        if (basinc < MIN_PRESS && tank >= TANK_STOP) {
            sleeping = false;
            mevcutHiz = MIN_FREQ;
            hedefHiz  = MIN_FREQ;
            oncekiBasinc = basinc;
            stableStart = 0;
            return mevcutHiz;
        }
        return 0.0f;
    }

    if (basinc <= MIN_PRESS) {
        hedefHiz = MAX_FREQ;     
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (basinc >= MID_PRESS) {
        hedefHiz = MIN_FREQ;     
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    float dp = basinc - oncekiBasinc;

    if (dp < DP_INC) {   
        hedefHiz += STEP;
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (dp > DP_DEC) { 
        hedefHiz -= STEP;
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    oncekiBasinc = basinc;

    if (hedefHiz > mevcutHiz) {
        mevcutHiz += STEP;
        if (mevcutHiz > hedefHiz) mevcutHiz = hedefHiz;
    }
    else if (hedefHiz < mevcutHiz) {
        mevcutHiz -= STEP;
        if (mevcutHiz < hedefHiz) mevcutHiz = hedefHiz;
    }

    if (fabs(mevcutHiz - MIN_FREQ) < 0.1f) {
        if (stableStart == 0) stableStart = now;
        if (now - stableStart > SLEEP_TIME) {
            sleeping = true;
            return 0.0f;
        }
    } else {
        stableStart = 0;
    }

    return mevcutHiz;
}


// Yangın pompası (inv 7) kontrolü: site kontrolüyle aynı mantıkta çalışır.
void fireControl()
{
    static uint32_t lastControl = 0;
    if (millis() - lastControl < 1000) return;
    lastControl = millis();

    if(fire_start && fireSensorOK && ads1115Ok && !inverterState[7].in_reset && !inv7_fault){ 
          float hz = 0;
          if(inverterState[7].running && !inverterState[7].first){
               hz = fire_pompa_kontrol_hesapla();
          }
              
          if (!inverterState[7].running && inverterState[7].resetted && inverterState[7].first && !inverterState[7].writePending) {
                
                inverterSetFrequency(7, 0.0f); 
                inverterState[7].first = false;
                return;
          } else if(inverterState[7].running && !inverterState[7].writePending) {
               
               inverterSetFrequency(7, hz);
              return;
          } else if(!inverterState[7].running && inverterState[7].resetted && !inverterState[7].writePending){
               
               inverterRun(7);
               return;
          } else {
               // Madde 2 (ilk açılış kilidi): Açılışta start zaten açıksa buraya düşülür.
               // Sürücü henüz resetlenmedi (resetted=false) → reset'e alınır. Böylece her
               // açılışta "reset → 0 Hz ilk frekans → run → rampa" sırası garanti olur;
               // sürücü EEPROM'daki son frekanstan kalkamaz ve fault/kilitliyse takılmaz.
               if (!inverterState[7].writePending) {
                    inverterReset(7);
                    resetFirePumpVariables = true;
               }
               return;
          }
    } else if(inverterState[7].running || inv7_fault || (!inverterState[7].resetted && !inverterState[7].in_reset)){
          
          inverterReset(7);
          resetFirePumpVariables = true;
          return;      
    }
}

// Yangın pompası frekansını hesaplar (tank ortalaması ve oto-dolum durumuna duyarlı).
float fire_pompa_kontrol_hesapla() {
    if (!ads1115Ok) return 0.0f; 
    if (fire_min < 0.5f || fire_max < 0.5f) return 0.0f; 

    const float MIN_PRESS = fire_min;        
    const float MAX_PRESS = fire_max;        
    const float MID_PRESS = (MIN_PRESS + MAX_PRESS) * 0.5f;
    const float MIN_FREQ  = 35.0f;
    const float MAX_FREQ  = 50.0f; 
    const float STEP      = 0.5f;
    const float DP_DEC    = +0.15f;   
    const float DP_INC    = -0.15f;   
    const float TANK_STOP = 15.0f;     
    const unsigned long SLEEP_TIME = 5000;

    float basinc = yangin_basinc;
    float tank = (tankLevels[2].tankYuzde + tankLevels[3].tankYuzde + tankLevels[4].tankYuzde + tankLevels[5].tankYuzde) / 4;

    
  
    static float mevcutHiz = MIN_FREQ;
    static float hedefHiz  = MIN_FREQ;
    static float oncekiBasinc = basinc;
    static unsigned long stableStart = 0;
    static bool sleeping = false;
    // Otomatik dolum kontrolü
    

    // Dolum varsa pompayı durdur
    if(oto_dolum ){
        sleeping = true;
        return 0.0f;
    }

    if (resetFirePumpVariables) {
        mevcutHiz = MIN_FREQ;
        hedefHiz = MIN_FREQ;
        stableStart = 0;
        sleeping = false;
        oncekiBasinc = basinc;
        resetFirePumpVariables = false; 
    }

    unsigned long now = millis();

    

    if (sleeping) {
        if (basinc < MIN_PRESS && tank >= TANK_STOP ) {
            sleeping = false;
            mevcutHiz = MIN_FREQ;
            hedefHiz  = MIN_FREQ;
            oncekiBasinc = basinc;
            stableStart = 0;
            return mevcutHiz;
        }
        return 0.0f;
    }

    if (basinc <= MIN_PRESS) {
        hedefHiz = MAX_FREQ;     
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (basinc >= MID_PRESS) {
        hedefHiz = MIN_FREQ;     
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    float dp = basinc - oncekiBasinc;

    if (dp < DP_INC) {   
        hedefHiz += STEP;
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (dp > DP_DEC) { 
        hedefHiz -= STEP;
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    oncekiBasinc = basinc;

    if (hedefHiz > mevcutHiz) {
        mevcutHiz += STEP;
        if (mevcutHiz > hedefHiz) mevcutHiz = hedefHiz;
    }
    else if (hedefHiz < mevcutHiz) {
        mevcutHiz -= STEP;
        if (mevcutHiz < hedefHiz) mevcutHiz = hedefHiz;
    }

    if (fabs(mevcutHiz - MIN_FREQ) < 0.1f) {
        if (stableStart == 0) stableStart = now;
        if (now - stableStart > SLEEP_TIME) {
            sleeping = true;
            return 0.0f;
        }
    } else {
        stableStart = 0;
    }

    return mevcutHiz;
}



// Yangın 2. pompası (inv 8) kontrolü; yangın pompası mantığının aynısıdır.
void fire1Control()
{
    static uint32_t lastControl = 0;
    if (millis() - lastControl < 1000) return;
    lastControl = millis();

    // Madde 1: Ekrandan (fire1_local) ve gerçek yangın anında başlatıldıysa,
    // yangın basınç sensörü bayat/ölü olsa bile büyük yangın pompası SABİT 40 Hz
    // blind çalışmaya devam eder. Buluttan (fire1_local=false) başlatıldıysa
    // sensör bayatken pompa DURDURULUR (else dalı reset'e girer).
    bool fire1Blind = fire1_start && fire1_local && (!fireSensorOK || !ads1115Ok);

    if(fire1_start && (fireSensorOK || fire1Blind) && !inverterState[8].in_reset && !inv8_fault ){ 
          float hz = 0;
          if(inverterState[8].running && !inverterState[8].first){
               if (fire1Blind) {
                    hz = 40.0f;
               } else {
                    hz = fire1_pompa_kontrol_hesapla();
               }
          }
              
          if (!inverterState[8].running && inverterState[8].resetted && inverterState[8].first && !inverterState[8].writePending) {
                
                inverterSetFrequency(8, 0.0f); 
                inverterState[8].first = false;
                return;
          } else if(inverterState[8].running && !inverterState[8].writePending) {
               
               inverterSetFrequency(8, hz);
              return;
          } else if(!inverterState[8].running && inverterState[8].resetted && !inverterState[8].writePending){
               
               inverterRun(8);
               return;
          } else {
               // Madde 2 (ilk açılış kilidi): Açılışta start zaten açıksa buraya düşülür.
               // Sürücü henüz resetlenmedi (resetted=false) → reset'e alınır. Böylece her
               // açılışta "reset → 0 Hz ilk frekans → run → rampa" sırası garanti olur;
               // sürücü EEPROM'daki son frekanstan kalkamaz ve fault/kilitliyse takılmaz.
               if (!inverterState[8].writePending) {
                    inverterReset(8);
                    resetFire1PumpVariables = true;
               }
               return;
          }
    } else if(inverterState[8].running || inv8_fault || (!inverterState[8].resetted && !inverterState[8].in_reset)){
          
          inverterReset(8);
          resetFire1PumpVariables = true;
          return;      
    }
}

// Yangın 2. pompa frekansını hesaplar (alt frekans aralığıyla çalışır).
float fire1_pompa_kontrol_hesapla() {
    if (!ads1115Ok) return 0.0f; 
    if (fire_min < 0.5f || fire_max < 0.5f) return 0.0f; 

    const float MIN_PRESS = fire_min;        
    const float MAX_PRESS = fire_max;        
    const float MID_PRESS = (MIN_PRESS + MAX_PRESS) * 0.5f;
    const float MIN_FREQ  = 25.0f;
    const float MAX_FREQ  = 50.0f; 
    const float STEP      = 0.5f;
    const float DP_DEC    = +0.15f;   
    const float DP_INC    = -0.15f;   
    const float TANK_STOP = 15.0f;     
    const unsigned long SLEEP_TIME = 5000;

    float basinc = yangin_basinc;
    float tank = (tankLevels[2].tankYuzde + tankLevels[3].tankYuzde + tankLevels[4].tankYuzde + tankLevels[5].tankYuzde) / 4;
    
    
  
    static float mevcutHiz = MIN_FREQ;
    static float hedefHiz  = MIN_FREQ;
    static float oncekiBasinc = basinc;
    static unsigned long stableStart = 0;
    static bool sleeping = false;

    
  
    if (oto_dolum ) {
        sleeping = true; // Pompanın rampa değerlerini sıfırlaması için uyku moduna alıyoruz
        return 0.0f;     // Dolum bitene kadar fire pompası çalışmayacak
    }
  
    if (resetFire1PumpVariables) {
        mevcutHiz = MIN_FREQ;
        hedefHiz = MIN_FREQ;
        stableStart = 0;
        sleeping = false;
        oncekiBasinc = basinc;
        resetFire1PumpVariables = false; 
    }

    unsigned long now = millis();

    

    if (sleeping) {
        if (basinc < MIN_PRESS && tank >= TANK_STOP ) {
            sleeping = false;
            mevcutHiz = MIN_FREQ;
            hedefHiz  = MIN_FREQ;
            oncekiBasinc = basinc;
            stableStart = 0;
            return mevcutHiz;
        }
        return 0.0f;
    }

    if (basinc <= MIN_PRESS) {
        hedefHiz = MAX_FREQ;     
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (basinc >= MID_PRESS) {
        hedefHiz = MIN_FREQ;     
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    float dp = basinc - oncekiBasinc;

    if (dp < DP_INC) {   
        hedefHiz += STEP;
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (dp > DP_DEC) { 
        hedefHiz -= STEP;
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    oncekiBasinc = basinc;

    if (hedefHiz > mevcutHiz) {
        mevcutHiz += STEP;
        if (mevcutHiz > hedefHiz) mevcutHiz = hedefHiz;
    }
    else if (hedefHiz < mevcutHiz) {
        mevcutHiz -= STEP;
        if (mevcutHiz < hedefHiz) mevcutHiz = hedefHiz;
    }

    if (fabs(mevcutHiz - MIN_FREQ) < 0.1f) {
        if (stableStart == 0) stableStart = now;
        if (now - stableStart > SLEEP_TIME) {
            sleeping = true;
            return 0.0f;
        }
    } else {
        stableStart = 0;
    }

    return mevcutHiz;
}



// Otomatik dolum valfi kararını verir: yangın modunda %15 altı/60 üstü,
// normal modda %95 altı/99.5 üstü eşikleriyle oto_dolum bayrağını ayarlar.
void dolumControl()
{
    float tanks = (tankLevels[2].tankYuzde +
                   tankLevels[3].tankYuzde +
                   tankLevels[4].tankYuzde +
                   tankLevels[5].tankYuzde) / 4;

    if (fire1_start) {
        // Yangın modu: Yangın suyu tankları boşalırsa dolum başlat
        if (tanks < 15.0f && !oto_dolum) {
            oto_dolum = true;
        }
        else if (tanks >= 60.0f && oto_dolum) {
            oto_dolum = false;
        }
    }
    else {
        // Normal çalışma: Kullanma suyu tankına göre dolum
        if (tanks < 95.0f && !oto_dolum) {
            oto_dolum = true;
        }
        else if (tanks >= 99.5f && oto_dolum) {
            oto_dolum = false;
        }
    }
}

unsigned long lastSiteBtnTime = 0;

const unsigned int debounceDelay = 200;

// Nextion'da sayfa değiştirir (page komutu + end marker).
void sayfaDegistir(int sayfa) {
    char komut[20];
    snprintf(komut, sizeof(komut), "page %d", sayfa);
    strcat(komut, "\xFF\xFF\xFF");
    Nex.print(komut);
}

// --- Nextion buton callback'leri ---
// Nextion site butonu callback'i: debounce ile site pompasını başlatır/durdurur.
void sitebtn() {
    unsigned long now = millis();
    if (now - lastSiteBtnTime < debounceDelay) return;
    lastSiteBtnTime = now;
    site_start = siteButton.value();
    messages = site_start ? "Site pompa calistirildi (ekran)" : "Site pompa durduruldu (ekran)";
}

// Nextion yangın butonu callback'i: debounce ile GERÇEK YANGIN (fire1) pompasını
// başlatır/durdurur. Bu buton artık fire1_start'ı (büyük yangın pompası) kontrol
// eder; fire_start yalnızca buluttan açılır. fire1_local, butonun yerel (ekran)
// kaynaklı olduğunu işaretler (sensör bayatlığında 40 Hz blind modu için).
void firebtn() {
    unsigned long now = millis();
    if (now - lastFireBtnTime < debounceDelay) return;
    lastFireBtnTime = now;
    fire1_start = fireButton.value();
    fire1_local = fire1_start;
    saveBool("fire1_start", fire1_start);
    messages = fire1_start ? "Yangin (fire1) pompasi calistirildi (ekran)" : "Yangin (fire1) pompasi durduruldu (ekran)";
}

// Site basınç limitlerini ekrandan kaydeder (doğrulama yapıp Preferences'a yazar).
void siteSaveButtonCallback() {
    float minVal = siteMinP.value() / 10;
    float maxVal = siteMaxP.value() / 10;
    if (isnan(minVal) || isnan(maxVal) || minVal < 0 || maxVal < 0 ||
        maxVal < minVal || minVal > maxVal || maxVal > 12.0) {
        messages = "Gecersiz site basinc degeri!";
        sayfaDegistir(0);
        return;
    }
    site_min = minVal;
    site_max = maxVal;
    saveFloat("site_min", site_min);
    saveFloat("site_max", site_max);
    messages = "Site basinc limitleri kaydedildi!";
    sayfaDegistir(0);
}

// --- Ekran guncelleme (degisen degerler yazilir) ---
// Nextion ekranını 500 ms'de bir günceller: tanklar, pompa LED'leri, arıza/status,
// voltaj/akım/frekans, basınç, limitler, butonlar ve durum mesajı.
void updateDisplay() {
    static uint32_t lastRun = 0;
    if (millis() - lastRun < 500) return;
    lastRun = millis();

    char buf[64];

    static int16_t  lastBarVal[6] = { -1, -1, -1, -1, -1, -1 };
    static uint16_t lastBarColor[6] = { 0, 0, 0, 0, 0, 0 };
    static uint16_t lastTxtColor[6] = { 0, 0, 0, 0, 0, 0 };
    static char     lastTxt[6][32] = { "" };

    static NextionComponent* const bars[6] = {
        nullptr, &tank1, &tank2, &tank3, &tank4, &tank5
    };

    static NextionComponent* const lvls[6] = {
        nullptr, &level1, &level2, &level3, &level4, &level5
    };

    for (uint8_t id = 1; id <= 5; id++) {

        bool timeout = (tankLevels[id].tankLevel == 65535);
        float pct = tankLevels[id].tankYuzde;
        float ton = tankLevels[id].tankTonaj;

        uint16_t barColor = (pct < 20.0f) ? 63488 : 1024;
        uint16_t txtColor = (pct < 20.0f) ? 63488 : 0;

        if (timeout) {
            barColor = 1048;
            txtColor = 1048;
        }

        int16_t v = timeout ? 0 : (int16_t)(pct + 0.5f);

        if (v != lastBarVal[id]) {
            bars[id]->value(v);
            lastBarVal[id] = v;
        }

        if (barColor != lastBarColor[id]) {
            bars[id]->attribute("pco", barColor);
            lastBarColor[id] = barColor;
        }

        if (txtColor != lastTxtColor[id]) {
            lvls[id]->attribute("pco", txtColor);
            lastTxtColor[id] = txtColor;
        }

        if (timeout)
            snprintf(buf, sizeof(buf), "N/A - Timeout");
        else
            snprintf(buf, sizeof(buf), "%.1f%% - %.2f ton", pct, ton);

        if (strcmp(lastTxt[id], buf) != 0) {
            strcpy(lastTxt[id], buf);
            lvls[id]->text(buf);
        }
    }

    static bool lastSiteRun = true, lastFireRun = true;

    if (inverterState[6].running != lastSiteRun) {
        lastSiteRun = inverterState[6].running;
        uint16_t c = lastSiteRun ? 2016 : 63488;
        site_pump_led.attribute("bco", c);
        site_pump_led.attribute("pco", c);
    }

    if (inverterState[7].running != lastFireRun) {
        lastFireRun = inverterState[7].running;
        uint16_t c = lastFireRun ? 2016 : 63488;
        fire_pump_led.attribute("bco", c);
        fire_pump_led.attribute("pco", c);
    }

    static uint16_t lastSiteFault = 0xFFFF, lastFireFault = 0xFFFF;

    if (inverterState[6].fault != lastSiteFault) {
        lastSiteFault = inverterState[6].fault;
        snprintf(buf, sizeof(buf), "0x%04X", lastSiteFault);
        siteFault.text(buf);
    }

    if (inverterState[7].fault != lastFireFault) {
        lastFireFault = inverterState[7].fault;
        snprintf(buf, sizeof(buf), "0x%04X", lastFireFault);
        fireFault.text(buf);
    }

    static uint16_t lastSiteStat = 0xFFFF, lastFireStat = 0xFFFF;

    if (inverterState[6].status != lastSiteStat) {
        lastSiteStat = inverterState[6].status;
        snprintf(buf, sizeof(buf), "0x%04X", lastSiteStat);
        siteStat.text(buf);
    }

    if (inverterState[7].status != lastFireStat) {
        lastFireStat = inverterState[7].status;
        snprintf(buf, sizeof(buf), "0x%04X", lastFireStat);
        fireStat.text(buf);
    }

    static uint16_t lastSiteVolt = 0xFFFF, lastFireVolt = 0xFFFF;

    if (inverterState[6].voltage != lastSiteVolt) {
        lastSiteVolt = inverterState[6].voltage;
        snprintf(buf, sizeof(buf), "%.1f V", lastSiteVolt * 0.1f);
        siteLineVoltage.text(buf);
    }

    if (inverterState[7].voltage != lastFireVolt) {
        lastFireVolt = inverterState[7].voltage;
        snprintf(buf, sizeof(buf), "%.1f V", lastFireVolt * 0.1f);
        fireLineVoltage.text(buf);
    }

    static uint16_t lastSiteCur = 0xFFFF, lastFireCur = 0xFFFF;

    if (inverterState[6].current != lastSiteCur) {
        lastSiteCur = inverterState[6].current;
        snprintf(buf, sizeof(buf), "%.1f A", lastSiteCur * 0.1f);
        siteCurrent.text(buf);
    }

    if (inverterState[7].current != lastFireCur) {
        lastFireCur = inverterState[7].current;
        snprintf(buf, sizeof(buf), "%.1f A", lastFireCur * 0.1f);
        fireCurrent.text(buf);
    }

    static uint16_t lastSiteFreq = 0xFFFF, lastFireFreq = 0xFFFF;

    if (inverterState[6].frequency != lastSiteFreq) {
        lastSiteFreq = inverterState[6].frequency;
        snprintf(buf, sizeof(buf), "%.1f Hz", lastSiteFreq * 0.1f);
        siteOutFreq.text(buf);
    }

    if (inverterState[7].frequency != lastFireFreq) {
        lastFireFreq = inverterState[7].frequency;
        snprintf(buf, sizeof(buf), "%.1f Hz", lastFireFreq * 0.1f);
        fireOutFreq.text(buf);
    }

    static int lastSiteP = -1, lastFireP = -1;

    int siteP10 = (int)(siteBasinc * 10.0f + 0.5f);
    int fireP10 = (int)(yanginBasinc * 10.0f + 0.5f);

    if (siteP10 != lastSiteP) {
        lastSiteP = siteP10;
        siteP.value(siteP10);
    }

    if (fireP10 != lastFireP) {
        lastFireP = fireP10;
        fireP.value(fireP10);
    }

    static int lastSiteMin = -1, lastSiteMax = -1;
    static int lastFireMin = -1, lastFireMax = -1;

    int siteMin10 = (int)(site_min * 10.0f + 0.5f);
    int siteMax10 = (int)(site_max * 10.0f + 0.5f);
    int fireMin10 = (int)(fire_min * 10.0f + 0.5f);
    int fireMax10 = (int)(fire_max * 10.0f + 0.5f);

    if (siteMin10 != lastSiteMin) {
        lastSiteMin = siteMin10;
        siteMinP.value(siteMin10);
    }

    if (siteMax10 != lastSiteMax) {
        lastSiteMax = siteMax10;
        siteMaxP.value(siteMax10);
    }

    if (fireMin10 != lastFireMin) {
        lastFireMin = fireMin10;
        fireMinP.value(fireMin10);
    }

    if (fireMax10 != lastFireMax) {
        lastFireMax = fireMax10;
        fireMaxP.value(fireMax10);
    }

    static bool lastSiteBtn = false, lastFireBtn = false;

    if (site_start != lastSiteBtn) {
        lastSiteBtn = site_start;
        siteButton.value(site_start);
    }

    if (fire1_start != lastFireBtn) {
        lastFireBtn = fire1_start;
        fireButton.value(fire1_start);
    }

    static char lastMsg[32] = "";
    char msgBuf[32];

    snprintf(msgBuf, sizeof(msgBuf), "%s", messages.c_str());

    if (strcmp(lastMsg, msgBuf) != 0) {
        strcpy(lastMsg, msgBuf);
        t0.text(msgBuf);
    }
}

// =========================
// LOOP
// =========================
// Ana döngü: cloud, ekran, boot gecikmesi, sensör okuma,
// UDP ve tüm pompa/modbus kontrollerini çalıştırır.
void loop()
{
  ArduinoCloud.update();
  Ekran.update();
  updateDisplay();
  if (!bootDelayDone) {
        if (millis() >= 15000) {
            bootDelayDone = true; 
            messages = "System Started";
            ArduinoCloud.update();
            delay(500);
        } else {
            return; 
        }
  }
  if(stop_sys) return;

  if (WiFi.status() == WL_CONNECTED){
      IPAddress ip = WiFi.localIP();
      if (ip != IPAddress(0,0,0,0)){
          String newIP = ip.toString();
          if (local_ip != newIP){
              local_ip = newIP;
          }
          if(!udpStarted){
              udp.begin(udpPort);
              broadcastIP = ip;
              broadcastIP[3] = 255;
              udpStarted = true;
          }
      }
    }
  
  static String newStatus;

  newStatus = String(inverterState[6].status, HEX) + " " +
            String(inverterState[7].status, HEX) + " " +
            String(inverterState[8].status, HEX);

  if (inv_status != newStatus)
  {
    inv_status = newStatus;
  }
  
  if(tankLevels[1].tankYuzde != tank1_yuzde) tank1_yuzde = tankLevels[1].tankYuzde; 
  if((tankLevels[2].tankYuzde + tankLevels[3].tankYuzde + tankLevels[4].tankYuzde + tankLevels[5].tankYuzde) / 4 != fire_yuzde)
             fire_yuzde =  (tankLevels[2].tankYuzde + tankLevels[3].tankYuzde + tankLevels[4].tankYuzde + tankLevels[5].tankYuzde) / 4;

  if(inverterState[6].current != site_curr) site_curr = inverterState[6].current / 10.0f;
  if(inverterState[7].current != fire_curr) fire_curr = inverterState[7].current / 10.0f;
  if(inverterState[8].current != fire1_curr) fire1_curr = inverterState[8].current / 10.0f;

  if(inverterState[6].frequency / 10.0f != aktif_site_freq) aktif_site_freq = inverterState[6].frequency / 10.0f;
  if(inverterState[7].frequency / 10.0f != aktif_fire_freq) aktif_fire_freq = inverterState[7].frequency / 10.0f;
  if(inverterState[8].frequency / 10.0f != aktif_fire1_freq) aktif_fire1_freq = inverterState[8].frequency / 10.0f;

  bool fault;
  fault = (inverterState[6].status & 0x0008) != 0;
  if (inv6_fault != fault) inv6_fault = fault;

  fault = (inverterState[7].status & 0x0008) != 0;
  if (inv7_fault != fault) inv7_fault = fault;
  
  fault = (inverterState[8].status & 0x0008) != 0;
  if (inv8_fault != fault) inv8_fault = fault;
  
  readPressureSensors();
  sensorSaglikKontrol();
  udpSend();
  
  
  siteControl();
  fireControl();
  fire1Control();
  ModbusScheduler();
  dolumControl();
  udpReceive();
  
  //otomatik/manuel üst yangın tankları dolum valfi

  if(dolum != oto_dolum){
      dolum = oto_dolum;
      digitalWrite(RELAY1, !oto_dolum);
      messages = oto_dolum ? "Otomatik dolum basladi" : "Otomatik dolum bitti";
  }  
}

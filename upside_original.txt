const char* resetReasonToStr(esp_reset_reason_t r) {
  switch (r) {
    case ESP_RST_POWERON:   return "POWERON";
    case ESP_RST_EXT:       return "EXT_RESET";
    case ESP_RST_SW:        return "SW_RESET";
    case ESP_RST_PANIC:     return "PANIC";
    case ESP_RST_INT_WDT:   return "INT_WDT";
    case ESP_RST_TASK_WDT:  return "TASK_WDT";
    case ESP_RST_WDT:       return "WDT";
    case ESP_RST_DEEPSLEEP: return "DEEPSLEEP";
    case ESP_RST_BROWNOUT:  return "BROWNOUT";
    case ESP_RST_SDIO:      return "SDIO";
    default:                return "UNKNOWN";
  }
}

void inverterleriResetle() {
    nodeInverter.begin(6, RS485Serial);
    uint8_t result;
    result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Stop
    SAFE_DELAY(200);

    result = nodeInverter.writeSingleRegister(0x2135, 0x0080); // Reset (bit 7 = 1)
    SAFE_DELAY(200);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Reset (bit 7 = 0)
    SAFE_DELAY(200);

    nodeInverter.begin(7, RS485Serial);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Stop
    SAFE_DELAY(200);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0080); // Reset (bit 7 = 1)
    SAFE_DELAY(200);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Reset (bit 7 = 0)
    SAFE_DELAY(200);

    nodeInverter.begin(8, RS485Serial);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Stop
    delay(200);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0080); // Reset (bit 7 = 1)
    SAFE_DELAY(200);
    result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Reset (bit 7 = 0)
    SAFE_DELAY(200);
  
    invSiteLastOkMs = invFireLastOkMs = invFire1LastOkMs = millis();
}

void site_Stop() {
  
    if (!siteMotorRunning) {
        Serial.println("[DEBUG] Site_Stop: Motor zaten durmuş, komut gönderilmiyor");
        return; // Motor zaten durmuşsa komut gönderme
    }
    
    Serial.println("[DEBUG] Site_Stop: 0x0007 komutu gönderiliyor...");
    //if (currentInverterID != 6) {
        nodeInverter.begin(6, RS485Serial);
        currentInverterID = 6;
        SAFE_DELAY(50);
    //}
    uint8_t result = nodeInverter.writeSingleRegister(0x2135, 0x0007);
    if (result != nodeInverter.ku8MBSuccess){
      messages = "[ERROR] Site inverter 0x2135 STOP (0x0007) yazma hatası!";
    }
    if (result == nodeInverter.ku8MBSuccess) {
        messages = "[DEBUG] Site_Stop: Başarılı!";
        siteMotorRunning = false; // Motor durmuş olarak işaretle
        siteStopRequested = true;
      
    }
    
}

float tankHacmiTon(const TankParametre& t) {
  float hacim_mm3 = PI * pow(t.cap_mm / 2.0, 2) * t.yukseklik_mm;
  float hacim_litre = hacim_mm3 / 1000000.0;
  return hacim_litre / 1000.0; // ton
}

/**********************  Yangın Tankları için Konsensus Destekli Okuma  **********************
 * - Modbus ID (1..5) ile C/C++ dizi indeksi (0..4) karıştırılmasın: her yerde id-1 kullanıldı.
 * - Tank-1 (site suyu) direkt güncellenir.
 * - Tank-2..5 (yangın) için okunan seviye_mm cache'lenir; timeout/outlier varsa UI'de median (konsensus) gösterilir.
 * - Ham değer ASLA değiştirilmez; sadece UI'de konsensus uygulanır.
 * - "Uçuk fark" eşiği mm cinsinden, iletişimli kap fiziğine uygundur (ABS_MM).
 ********************************************************************************************/

// === Global/Static cache (bu dosyada bir kere tanımlı olsun) ===
static float g_fireLevel_mm[4] = {NAN, NAN, NAN, NAN}; // tank 2..5
static bool  g_fireOk[4]       = {false,false,false,false};
static bool  g_fireSeen[4]     = {false,false,false,false};

// Basit median (2–4 eleman için yeterli)
static float _medianOf(std::vector<float> v) {
  if (v.empty()) return NAN;
  std::sort(v.begin(), v.end());
  size_t n = v.size();
  return (n % 2) ? v[n/2] : 0.5f*(v[n/2 - 1] + v[n/2]);
}

//08-11-2025
static void _EqualizeFireTanks_UI() {
  const int   MIN_VALID = 1;
  const float ABS_MM    = 50.0f;

  // HEAP-FREE: geçerli okumaları sabit diziye topla (vector/sort yok)
  float vals_arr[4]; 
  int   vals_n = 0;
  for (int i = 0; i < 4; i++) {
    if (g_fireOk[i] && !std::isnan(g_fireLevel_mm[i])) { vals_arr[vals_n++] = g_fireLevel_mm[i]; }
  }
  
  allFireNok = (vals_n == 0);
  if (!allFireNok) fireLastOkMs = millis();

  float total_fire_ton = 0.0f; // 2..5 toplam su (ton)

  if (vals_n < MIN_VALID) {
    // Konsensus yok → geçerli olanlardan toplamı hesapla ve timeout'ları işaretle
    for (int i = 0; i < 4; i++) {
      if (g_fireSeen[i] && !g_fireOk[i]) {
        int id = i + 2;
        switch (id) {
          case 2: level2.text("N/A - Timeout"); tank2.attribute("pco", 1048); break;
          case 3: level3.text("N/A - Timeout"); tank3.attribute("pco", 1048); break;
          case 4: level4.text("N/A - Timeout"); tank4.attribute("pco", 1048); break;
          case 5: level5.text("N/A - Timeout"); tank5.attribute("pco", 1048); break;
        }
      }

      // Geçerli okumalardan toplamı topla
      if (g_fireOk[i] && !std::isnan(g_fireLevel_mm[i])) {
        int id = i + 2;                  // 2..5
        const auto &t = tanklar[id];     // 1..5 indeksli
        if (t.yukseklik_mm > 0.0f) {
          float tonaj = (g_fireLevel_mm[i] / t.yukseklik_mm) * tankHacmiTon(t);
          total_fire_ton += max(0.0f, tonaj);
        }
      }
    }

    // Toplamı yaz ve çık
    total_fire_water_level = total_fire_ton;
    total_fire_water_levely = (g_total_fire_capacity_ton > 0.0f) 
                              ? (total_fire_ton / g_total_fire_capacity_ton) * 100.0f 
                              : 0.0f;

    // üst tanklar set edilen seviyenin altında ise su talebini açalım
    if ((total_fire_water_levely <= min_seviye) && !dolum_hata && !fire && (dolum == 0) /*&& !su_bitti*/) {
       dolum = 1;
    }
    return;
  }

  // Konsensus var → median (dinamik bellek kullanmadan)
  float cons;
  {
    float t[4]; 
    int   n = vals_n; 
    if (n > 4) n = 4; 
    for (int i = 0; i < n; i++) t[i] = vals_arr[i];
    // küçük selection sort
    for (int i = 0; i < n - 1; i++) {
      int m = i; 
      for (int j = i + 1; j < n; j++) if (t[j] < t[m]) m = j;
      float x = t[i]; t[i] = t[m]; t[m] = x;
    }
    cons = (n & 1) ? t[n/2] : 0.5f * (t[n/2 - 1] + t[n/2]);
  }

  for (int i = 0; i < 4; i++) {
    if (!g_fireSeen[i]) continue;
    int id = i + 2;                    // 2..5
    const auto &t = tanklar[id];       // 1..5 indeksli

    bool timeout = !g_fireOk[i];
    bool outlier = (!timeout && !std::isnan(g_fireLevel_mm[i]))
                   ? fabsf(g_fireLevel_mm[i] - cons) > ABS_MM
                   : false;

    float use_mm = (timeout || outlier || std::isnan(g_fireLevel_mm[i])) ? cons : g_fireLevel_mm[i];

    float pct   = (t.yukseklik_mm > 0.0f) ? (use_mm / t.yukseklik_mm) * 100.0f : 0.0f;
    pct = constrain(pct, 0.0f, 100.0f);
    float tonaj = (t.yukseklik_mm > 0.0f) ? (use_mm / t.yukseklik_mm) * tankHacmiTon(t) : 0.0f;

    // Toplama ekle
    total_fire_ton += max(0.0f, tonaj);

    const uint16_t RENK_KIRMIZI = 63488, RENK_YESIL = 1024, RENK_MAVI = 1048;
    bool kritik = pct < 20.0f;
    uint16_t barColor  = (timeout || outlier) ? RENK_MAVI : (kritik ? RENK_KIRMIZI : RENK_YESIL);
    uint16_t textColor = (timeout || outlier) ? RENK_MAVI : (kritik ? RENK_KIRMIZI : 0);

    char sonuc[64]; 
    snprintf(sonuc, sizeof(sonuc), "%.1f%% - %.2f ton", pct, tonaj);

    switch (id) {
      case 2: tank2_seviye = tonaj; tank2_yuzde = pct;
              tank2.attribute("pco", barColor); level2.attribute("pco", textColor);
              tank2.value((int)lroundf(pct)); level2.text((timeout || outlier) ? "Konsensus" : sonuc); break;
      case 3: tank3_seviye = tonaj; tank3_yuzde = pct;
              tank3.attribute("pco", barColor); level3.attribute("pco", textColor);
              tank3.value((int)lroundf(pct)); level3.text((timeout || outlier) ? "Konsensus" : sonuc); break;
      case 4: tank4_seviye = tonaj; tank4_yuzde = pct;
              tank4.attribute("pco", barColor); level4.attribute("pco", textColor);
              tank4.value((int)lroundf(pct)); level4.text((timeout || outlier) ? "Konsensus" : sonuc); break;
      case 5: tank5_seviye = tonaj; tank5_yuzde = pct;
              tank5.attribute("pco", barColor); level5.attribute("pco", textColor);
              tank5.value((int)lroundf(pct)); level5.text((timeout || outlier) ? "Konsensus" : sonuc); break;
    }
    
  }

  // Konsensuslu nihai değerlerle toplamı yaz
  total_fire_water_level = total_fire_ton;
  total_fire_water_levely = (g_total_fire_capacity_ton > 0.0f) 
                            ? (total_fire_ton / g_total_fire_capacity_ton) * 100.0f 
                            : 0.0f;
  if ((total_fire_water_levely <= min_seviye) && !dolum_hata && !fire && (dolum == 0) /*&& !su_bitti*/) {
     dolum = 1;
  }
}





void tankSeviyeOkuVeGuncelle(uint8_t id) {
  if (id < 1 || id > 5) return;

  // ✅ Static değişken: son kullanılan ID
  static uint8_t lastSensorId = 1;

  // --- Sabitler (yalnızca bu fonksiyon için) ---
  static const uint32_t NOK_GRACE_MS = 20000;   // 20 sn: sürekli NOK ise 0'a zorla

  const int idx = id - 1;              // 0..4
  const auto &t = tanklar[id];         // tank parametreleri 1..5 indeksli

  // ✅ SADECE ID değiştiğinde begin() çağır
  if (lastSensorId != id) {
    nodeSensor.begin(id, RS485Serial);
    lastSensorId = id;
    SAFE_DELAY(50); // Kısa stabilizasyon
  }

  unsigned long t0 = millis();
  uint8_t result = nodeSensor.readInputRegisters(0, 1);
  (void)t0; // t0 kullanmıyorsan uyarıyı sustur

  bool  ok  = (result == nodeSensor.ku8MBSuccess);
  float val = ok ? nodeSensor.getResponseBuffer(0) : NAN;

  // Geçerli aralık kontrolü (sensörün beklenen mm aralığı)
  if (ok && !(val >= 200.0f && val <= 4500.0f)) ok = false;

  // Ani sıçrama filtresi (sadece ok=true iken)
  if (ok) {
    float prev = oncekiTankSeviye[idx];
    if (prev >= 0 && fabsf(val - prev) > SICRAMA_ESIK) {
      aniSicramaSayaci[idx]++;
      if (aniSicramaSayaci[idx] < SICRAMA_TOLERANS) {
        val = prev; // tek atımlık anomaliyi yut
      }
    } else {
      aniSicramaSayaci[idx] = 0;
    }
    oncekiTankSeviye[idx] = val;
  }

  // Seviye hesabı
  float seviye_mm = NAN;
  if (ok) {
    seviye_mm = t.yukseklik_mm - (val - t.sensor_yukseklik_mm);
    seviye_mm = constrain(seviye_mm, 0.0f, t.yukseklik_mm);
  }

  if (id == 1) { // Site tankı
    if (ok && (t.yukseklik_mm > 0.0f)) {
      float tonaj = (seviye_mm / t.yukseklik_mm) * tankHacmiTon(t);
      float pct   = (seviye_mm / t.yukseklik_mm) * 100.0f;
      pct = constrain(pct, 0.0f, 100.0f);

      // SADECE ok=true iken iç durumu ve göstergeyi güncelle
      tank1_seviye = tonaj;
      tank1_yuzde  = pct;

      tank1.attribute("pco", (pct < 20.0f) ? 63488 : 1024);
      level1.attribute("pco", (pct < 20.0f) ? 63488 : 0);
      tank1.value((int)lroundf(pct));

      char sonuc[64];
      snprintf(sonuc, sizeof(sonuc), "%.1f%% - %.2f ton", pct, tonaj);
      level1.text(sonuc);

      siteNok = false;
      siteLastOkMs = millis();
    } else {
      // ok=false ise: 20 sn'yi geçtiyse 0'a zorla, aksi halde hold-last-valid
      forceZero = (millis() - siteLastOkMs >= NOK_GRACE_MS);

      if (forceZero) {
        tank1_seviye = 0.0f;
        tank1_yuzde  = 0.0f;
        tank1.value(0);
        tank1.attribute("pco", 63488);
        level1.attribute("pco", 63488);

        char s[64];
        snprintf(s, sizeof(s), "0.0%% - 0.00 ton (timeout>%us)", (unsigned)(NOK_GRACE_MS/1000));
        level1.text(s);
      } else {
        // GRACE: İç durumu DEĞİŞTİRME, sadece UI'da timeout göster
        siteLastOkMs = millis();  // timeout sayacını tazele
        level1.text("N/A - Timeout");
        tank1.attribute("pco", 50712);
        level1.attribute("pco", 50712);
        // tank1_yuzde / tank1_seviye / tank1.value() aynen kalır
      }

      // Motoru gereksiz kapatmamak için sadece gerçekten forceZero iken NOK say
      siteNok = forceZero;
    }

    // _msg_acc_1 için tahsissiz tek atama
    char acc[24];
    snprintf(acc, sizeof(acc), "%u: %s  ", id, ok ? "OK" : "NOK");
    _msg_acc_1 = acc; // _msg_acc_1 String ise güvenli (kopyalar)

  } else {
    // 2..5 (yangın tankları): konsensus katmanı için ham veriyi/flag'ı bırak
    int fi = id - 2; // 2..5 -> 0..3
    if (fi >= 0 && fi < 4) {
      g_fireSeen[fi]     = true;
      g_fireOk[fi]       = ok;
      g_fireLevel_mm[fi] = ok ? seviye_mm : NAN;
    }
  }
}






void readPressureSensors() {
  static bool adsInit = false;
  static bool adsOk = true;
  if (!adsInit) {
    adsOk = ads1115.begin();
    adsInit = true;
    if (!adsOk) {
      Serial.println("ADS1115 bulunamadı! Okuma yapılmayacak.");
      return;
    }
  }
  if (!adsOk) return;

  const int sampleCount = 5;
  float sumSite = 0.0, sumYangin = 0.0;

  for (int i = 0; i < sampleCount; i++) {
    int16_t adc0 = ads1115.readADC_SingleEnded(0); // A0: Site basınç
    int16_t adc1 = ads1115.readADC_SingleEnded(1); // A1: Yangın basınç

    float voltage0 = adc0 * 0.1875 / 1000.0; // mV to V
    float voltage1 = adc1 * 0.1875 / 1000.0; // mV to V

    // 0 bar = 0.6V, 9 bar = 2.2770V
    float barSite = (voltage0 - 0.5) * (16.0 / (3.3 - 0.5));  // = (voltage - 0.5) * 5.714
    float barYangin = (voltage1 - 0.5) * (16.0 / (3.3 - 0.5));  // = (voltage - 0.5) * 5.714
    
    sumSite += barSite;
    sumYangin += barYangin;

    SAFE_DELAY(5); // Gürültüye karşı kısa bekleme
  }

  site_basinc = sumSite / sampleCount;
  yangin_basinc = sumYangin / sampleCount;
  siteP.value(site_basinc * 10);
  fireP.value(yangin_basinc * 10);
  /*
  Serial.print("Site Basınç: ");
  Serial.print(site_basinc);
  Serial.print(" bar | Yangın Basınç: ");
  Serial.print(yangin_basinc);
  Serial.println(" bar");
  */
}

float site_pompa_kontrol_hesapla() {

    // ===== BASINÇ EŞİKLERİ =====
    const float MIN_PRESS = site_min;        // ör: 2.5 bar
    const float MAX_PRESS = site_max;        // ör: 3.0 bar
    const float MID_PRESS = (MIN_PRESS + MAX_PRESS) * 0.5f;

    // ===== FREKANS EŞİKLERİ =====
    const float MIN_FREQ  = 35.0f;
    const float MAX_FREQ  = site_max_freq;
    const float STEP      = 0.5f;

    // ===== BASINÇ DEĞİŞİM ALGILAMA =====
    const float DP_DEC    = +0.15f;   // basınç bu kadar artarsa → hız düş
    const float DP_INC    = -0.15f;   // basınç bu kadar düşerse → tüketim var → hız artır
    const float JITTER    = 0.03f;

    // ===== TANK EŞİĞİ =====
    const float TANK_STOP = 15.0f;     // %15 altı → sistem durmalı

    // ===== UYKU SÜRESİ =====
    const unsigned long SLEEP_TIME = 5000;

    float basinc = site_basinc;
    float tank = tank1_yuzde;

    static float mevcutHiz = MIN_FREQ;
    static float hedefHiz  = MIN_FREQ;
    static float oncekiBasinc = basinc;

    static unsigned long stableStart = 0;
    static bool sleeping = false;

    unsigned long now = millis();

    // =======================================================
    // 0) SU YOKSA → DUR
    // =======================================================
    if ((tank < TANK_STOP) || forceZero) {
        sleeping = true;
        return 0.0f;
    }

    // =======================================================
    // 1) UYKU MODU
    // =======================================================
    if (sleeping) {

        // basınç min altına düştüyse → uyandır
        if (basinc < MIN_PRESS && tank >= TANK_STOP) {
            sleeping = false;
            mevcutHiz = MIN_FREQ;
            hedefHiz  = MIN_FREQ;
            oncekiBasinc = basinc;
            stableStart = 0;
            return mevcutHiz;
        }

        return 0.0f;
    }

    // =======================================================
    // 2) BASINCA GÖRE HEDEF HIZ (lineer YOK!)
    // =======================================================
    if (basinc <= MIN_PRESS) {
        hedefHiz = MAX_FREQ;     // basınç düşük → hız artır
    }
    else if (basinc >= MID_PRESS) {
        hedefHiz = MIN_FREQ;     // basınç yüksek → hız azalt (35 Hz)
    }
    else {
        // ORTA ZONDA → hedefi koru, lineer hesap yok
    }

    // =======================================================
    // 3) TÜKETİM ALGILAMA
    // =======================================================
    float dp = basinc - oncekiBasinc;

    if (dp < DP_INC) {   // basınç hızla düşüyorsa → tüketim var → artır
        hedefHiz += STEP;
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (dp > DP_DEC) { // basınç hızla yükseliyorsa → düşür
        hedefHiz -= STEP;
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }
    // dp jitter bölgesindeyse → hedef aynı

    oncekiBasinc = basinc;

    // =======================================================
    // 4) FREKANS RAMPA (±1 Hz)
    // =======================================================
    if (hedefHiz > mevcutHiz) {
        mevcutHiz += STEP;
        if (mevcutHiz > hedefHiz) mevcutHiz = hedefHiz;
    }
    else if (hedefHiz < mevcutHiz) {
        mevcutHiz -= STEP;
        if (mevcutHiz < hedefHiz) mevcutHiz = hedefHiz;
    }

    // =======================================================
    // 5) 35 Hz'de uzun kalırsa → UYKU
    // =======================================================
    if (fabs(mevcutHiz - MIN_FREQ) < 0.1f) {

        if (stableStart == 0)
            stableStart = now;

        if (now - stableStart > SLEEP_TIME) {
            sleeping = true;
            return 0.0f;
        }

    } else {
        stableStart = 0;
    }

    return mevcutHiz;
}



float yangin_pompa_kontrol_hesapla() {
    //fire1=true;
    //return 0.0f;
    if ((dolum > 0) && (dolum < 6)) return 0.0f;
  
    const float MIN_PRESS = fire_min;      
    const float MAX_PRESS = fire_max;      
    const float MID_PRESS = (MIN_PRESS + MAX_PRESS) * 0.5f;

    const float MIN_FREQ  = 35.0f;
    const float MAX_FREQ  = fire_max_freq;

    const float STEP      = 1.0f;

    const float DP_DEC    = +0.5f;
    const float DP_INC    = -0.5f;
    const float JITTER    = 0.1f;

    const float TANK_STOP = 10.0f;   // %10 altı → durma

    const unsigned long SLEEP_TIME = 5000;

    float basinc = yangin_basinc;
    float ort = (tank2_yuzde + tank3_yuzde + tank4_yuzde + tank5_yuzde) * 0.25f;

    static float mevcutHiz  = MIN_FREQ;
    static float hedefHiz   = MIN_FREQ;
    static float oncekiBasinc = basinc;
    static unsigned long stableStart = 0;
    static bool sleeping = false;

    unsigned long now = millis();

    // SU YOKSA → DUR
    if (ort < TANK_STOP) {
        sleeping = true;
        return 0.0f;
    }

    // UYKU MODU
    if (sleeping) {

        if (basinc < MIN_PRESS && ort >= TANK_STOP) {
            sleeping = false;
            mevcutHiz = MIN_FREQ;
            hedefHiz  = MIN_FREQ;
            oncekiBasinc = basinc;
            stableStart = 0;
            return mevcutHiz;
        }

        return 0.0f;
    }

    // BASINCA GÖRE KARAR (LINEER YOK!)
    if (basinc <= MIN_PRESS) {
        hedefHiz = MAX_FREQ;    // düşük basınç → hız artır
    }
    else if (basinc >= MID_PRESS) {
        hedefHiz = MIN_FREQ;    // orta nokta üstü → minimum hız
    }
    else {
        // ORTA BÖLGE → HEDEFİ SABİT TUT
        // Buraya lineer hesap YOK, dokunmuyoruz.
    }

    // TÜKETİM ALGILAMA
    float dp = basinc - oncekiBasinc;

    if (dp < DP_INC) {  
        hedefHiz += STEP;
        if (hedefHiz > MAX_FREQ) hedefHiz = MAX_FREQ;
    }
    else if (dp > DP_DEC) {
        hedefHiz -= STEP;
        if (hedefHiz < MIN_FREQ) hedefHiz = MIN_FREQ;
    }

    oncekiBasinc = basinc;

    // FREKANS RAMPA
    if (hedefHiz > mevcutHiz) {
        mevcutHiz += STEP;
        if (mevcutHiz > hedefHiz) mevcutHiz = hedefHiz;
        if((mevcutHiz >= 49) && fire){fire1 = true;} //yangın durumu varsa büyük pompayı aktif et
    }
    else if ((hedefHiz < mevcutHiz) && !fire1) {
        mevcutHiz -= STEP;
        if (mevcutHiz < hedefHiz) mevcutHiz = hedefHiz;
    }

    // UYKU
    if (fabs(mevcutHiz - MIN_FREQ) < 0.1f) {

        if (stableStart == 0)
            stableStart = now;

        if (now - stableStart > SLEEP_TIME) {
            sleeping = true;
            return 0.0f;
        }

    } else {
        stableStart = 0;
    }

    return mevcutHiz;
}

 void readHoldingReg(uint8_t id, uint16_t addr) {

    static uint8_t retryId = 0;
    static uint16_t retryAddr = 0;
    static bool retryPending = false;

    uint8_t result;

    // 🔹 Eğer retry varsa, SADECE onu dene
    if (retryPending) {
        nodeInverter.begin(retryId, RS485Serial);
        result = nodeInverter.readHoldingRegisters(retryAddr, 1);
        retryPending = false;
    } 
    // 🔹 Normal okuma
    else {
        nodeInverter.begin(id, RS485Serial);
        result = nodeInverter.readHoldingRegisters(addr, 1);

        if (result != nodeInverter.ku8MBSuccess) {
            retryId = id;
            retryAddr = addr;
            retryPending = true;
            return;   // 🚨 AYNI LOOP'TA İKİNCİ OKUMA YOK
        }
    }

    if (result == nodeInverter.ku8MBSuccess) {
        markInverterCommOk(id);
        guncelleInverterCloudVars(id, addr, nodeInverter.getResponseBuffer(0));
    } else {
        static char statusbuf[96];
        snprintf(statusbuf, sizeof(statusbuf), "INV Read Error id:%d", id);
        t0.text(statusbuf);
        //0messages = String(statusbuf);
    }
}

static char fireMsgBuf[100];

static void fireSpeedUpdate() {
    if (fireStopRequested) return;
    
    if (fire || sulama_rt) {

        if (!fireMotorRunning) {

            if (!fireFirstFreqSent) {
                float motorHizi = yangin_pompa_kontrol_hesapla(); // Hz
                if (motorHizi <= 0.0f) return;

                uint16_t regVal = (uint16_t)(motorHizi * 10.0f + 0.5f);
                nodeInverter.begin(7, RS485Serial);
                uint8_t r = nodeInverter.writeSingleRegister(0x2136, regVal);

                if (r == nodeInverter.ku8MBSuccess) {
                    fireFirstFreqSent = true;
                    lastFireHz = motorHizi;
                } else {
                    snprintf(fireMsgBuf, sizeof(fireMsgBuf),
                             "[MODBUS_WRITE] Fire İlk Frekans: HATA!");
                    t0.text(fireMsgBuf);
                    messages = fireMsgBuf;
                }

            } else if (!fireRunSent) {

                nodeInverter.begin(7, RS485Serial);
                uint8_t r = nodeInverter.writeSingleRegister(0x2135, 0x000F);

                if (r == nodeInverter.ku8MBSuccess) {
                    fireRunSent = true;
                } else {
                    snprintf(fireMsgBuf, sizeof(fireMsgBuf),
                             "[MODBUS_WRITE] Fire RUN: HATA!");
                    t0.text(fireMsgBuf);
                    messages = fireMsgBuf;
                }
            }

        } else if (fireRunSent) {

            float motorHizi = yangin_pompa_kontrol_hesapla();
            if (fabs(motorHizi - lastFireHz) > FIRE_HZ_EPS) {

                uint16_t regVal = (uint16_t)(motorHizi * 10.0f + 0.5f);
                nodeInverter.begin(7, RS485Serial);
                uint8_t r = nodeInverter.writeSingleRegister(0x2136, regVal);

                if (r == nodeInverter.ku8MBSuccess) {
                    lastFireHz = motorHizi;
                } else {
                    snprintf(fireMsgBuf, sizeof(fireMsgBuf),
                             "[MODBUS_WRITE] Fire Frekans: HATA!");
                    t0.text(fireMsgBuf);
                    messages = fireMsgBuf;
                }
            }
        }

    } else {
        if (fireMotorRunning) {
            fire_Stop();
        }
        fireFirstFreqSent = false;
        fireRunSent       = false;
        lastFireHz        = -1.0f;
    }
}

void loop() {


  if (WiFi.status() == WL_CONNECTED) {

    IPAddress ip = WiFi.localIP();

    if (ip != IPAddress(0,0,0,0)) {
      local_ip = ip.toString();
    }
  }

  /*
  if (ArduinoCloud.connected()) {
     ArduinoCloud.update(); 
  }
  */
    lt = millis();
    //Serial.print("[DEBUG] Loop #");
    //Serial.println(count_loop);
    count_loop = count_loop+1;
    
    static unsigned long lastCloud = 0;
    if (millis() - lastCloud > 200) {
        ArduinoCloud.update();
        lastCloud = millis();
    }
    
    Ekran.update();
    static uint8_t step = 0;
    static bool inverterlerResetlendi = false;
    static unsigned long lastLedToggle = 0;
    static bool ledState = false;
    if (millis() - lastLedToggle >= 500) {
        ledState = !ledState;
        digitalWrite(LED_BUILTIN, ledState ? HIGH : LOW);
        lastLedToggle = millis();
    }
    if (!inverterlerResetlendi) {
        inverterleriResetle();
        inverterlerResetlendi = true;
    }

    SAFE_DELAY(50);
  
    // --- State machine: inverter + tank okuma ---
    const uint8_t inverterStepCount = 27; // (yorumunu istersen güncelle)
    const uint8_t tankStepCount = sizeof(aktifTankler) / sizeof(aktifTankler[0]);
    const uint8_t totalStep = inverterStepCount + tankStepCount;

    if (step < inverterStepCount) {
        switch (step) {
            case 0:  readHoldingReg(6, 0x1BD1); break; // fault code
            case 1:  readHoldingReg(7, 0x1BD1); break;
            case 2:  readHoldingReg(8, 0x1BD1); break;
            case 3:  readHoldingReg(6, 0x0C84); break; // estimated motor current
            case 4:  readHoldingReg(7, 0x0C84); break;
            case 5:  readHoldingReg(8, 0x0C84); break;
            case 6:  readHoldingReg(6, 0x0C82); break; // estimated motor frequency
            case 7:  readHoldingReg(7, 0x0C82); break;
            case 8:  readHoldingReg(8, 0x0C82); break;
            case 9:  readHoldingReg(6, 0x0C87); break; // main voltage
            case 10: readHoldingReg(7, 0x0C87); break; // main voltage
            case 11: readHoldingReg(8, 0x0C87); break; // main voltage
            case 12: readHoldingReg(6, 0x2329); break; // ACC
            case 13: readHoldingReg(7, 0x2329); break; // ACC
            case 14: readHoldingReg(8, 0x2329); break; // ACC
            case 15: readHoldingReg(6, 0x232A); break; // DEC
            case 16: readHoldingReg(7, 0x232A); break; // DEC
            case 17: readHoldingReg(8, 0x232A); break; // DEC
            case 18: readHoldingReg(6, 0x2582); break; // nominal motor frequency
            case 19: readHoldingReg(7, 0x2582); break; // nominal motor frequency
            case 20: readHoldingReg(8, 0x2582); break; // nominal motor frequency
            case 21: readHoldingReg(6, 0x0C81); break; // status word
            case 22: readHoldingReg(7, 0x0C81); break; // status word
            case 23: readHoldingReg(8, 0x0C81); break; // status word
            case 24: readPressureSensors();fireSpeedUpdate(); break;
            case 25: siteSpeedUpdate(); break;
            case 26: fire1SpeedUpdate(); break;
        }
    } else {
        // --- Tank okuma ---
        uint8_t tankIdx = step - inverterStepCount;
        if (tankIdx < tankStepCount) {

            // 1) Tank fazının BAŞI: konsensus cache'ini sıfırla
            if (tankIdx == 0) {
                _msg_acc = "";
                _msg_acc_1 = "";
                for (int i = 0; i < 4; i++) {
                    g_fireSeen[i]     = false;
                    g_fireOk[i]       = false;
                    g_fireLevel_mm[i] = NAN;
                    
                }
              
            }

            uint8_t tid = aktifTankler[tankIdx];
            //Serial.print("[TANK_READ] Tank ID: "); Serial.println(tid);

            // Not: tankSeviyeOkuVeGuncelle() içinde indeks düzeltmesi (id-1) ve
            // 65535 / <200 / >4500 filtreleri zaten var.
            // Ayrıca bu fonksiyonun içinde _EqualizeFireTanks_UI() ÇAĞRISI OLMAMALI.
            tankSeviyeOkuVeGuncelle(tid);

            // 2) Bu tankın OK/NOK bilgisini ekleme: String biriktirmek yerine
            //    faz sonunda mevcut g_fireOk[] durumundan tek seferde metin oluşturacağız.

            // 2) Tank fazının SONU: tek seferde konsensus uygula (2..5 için)
            if (tankIdx == tankStepCount - 1) {
                _EqualizeFireTanks_UI();

                // String biriktirmek yerine mevcut durumdan tek seferde derle
                char tanksBuf[96];
                tanksBuf[0] = '\0';
                for (int id = 2; id <= 5; id++) {
                    int fi = id - 2;
                    char seg[16];
                    snprintf(seg, sizeof(seg), "%d: %s  ", id, (g_fireOk[fi] ? "OK" : "NOK"));
                    size_t curLen = strlen(tanksBuf);
                    size_t segLen = strlen(seg);
                    if (curLen + segLen < sizeof(tanksBuf)) {
                        memcpy(tanksBuf + curLen, seg, segLen + 1);
                    } else {
                        break; // taşmayı engelle
                    }
                }

                char buf_a[192];
                snprintf(buf_a, sizeof(buf_a), "%s%s%s", _msg_acc_1.c_str(), tanksBuf, last_reset_reason.c_str());

                // messages için "değiştiyse yaz" filtresi
                static char lastMsg[192] = "";
                if (strcmp(lastMsg, buf_a) != 0) {
                    //messages = buf_a; 
                    strncpy(lastMsg, buf_a, sizeof(lastMsg));
                    lastMsg[sizeof(lastMsg)-1] = '\0';
                }
                //messages = /*_msg_acc_1 + tanksBuf +*/ last_reset_reason;
            }
        }
    }
    step = (step + 1) % totalStep;
    //delay(10);


    if (millis() - fireLastOkMs > 20000) {
        tanks_readed = false;
    }else{
        tanks_readed = true;
    }

    // --- Site inverteri iki aşamalı stop işlemi ---
    if (siteStopRequested) {
        float freq = site_freq;
        if (freq == 0.0f) {
            if (siteStopTimer == 0) {
                siteStopTimer = millis();
                //Serial.println("[DEBUG] Site: İki aşamalı stop başlatıldı");
            }
            if (millis() - siteStopTimer >= 500) {
                //Serial.println("[DEBUG] Site: Final stop komutu gönderiliyor (0x0006)");
                nodeInverter.begin(6, RS485Serial);
                uint8_t result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Tam stop
              /*
                if (result != nodeInverter.ku8MBSuccess) Serial.println("[ERROR] Site inverter 0x2135 FINAL STOP (0x0006) yazma hatası!");*/
                siteStopRequested = false;
                siteStopTimer = 0;
            }
        } else {
            siteStopTimer = 0;
        }
    }
    if (fireStopRequested) {
        float freq = fire_freq;
        if (freq == 0.0f) {
            if (fireStopTimer == 0) {
                fireStopTimer = millis();
                //Serial.println("[DEBUG] Fire: İki aşamalı stop başlatıldı");
            }
            if (millis() - fireStopTimer >= 500) {
                //Serial.println("[DEBUG] Fire: Final stop komutu gönderiliyor (0x0006)");
                nodeInverter.begin(7, RS485Serial);
                uint8_t result = nodeInverter.writeSingleRegister(0x2135, 0x0006);
              /*
                if (result != nodeInverter.ku8MBSuccess) Serial.println("[ERROR] Fire inverter 0x2135 FINAL STOP (0x0006) yazma hatası!");*/
                fireStopRequested = false;
                fireStopTimer = 0;
            }
        } else {
            fireStopTimer = 0;
        }
    }


    if (fire1StopRequested) {
        float freq = fire1_freq;
        if (freq == 0.0f) {
            if (fire1StopTimer == 0) {
                fire1StopTimer = millis();
            }
            if (millis() - fire1StopTimer >= 500) {
                nodeInverter.begin(8, RS485Serial);
                uint8_t result = nodeInverter.writeSingleRegister(0x2135, 0x0006); // Tam stop
                fire1StopRequested = false;
                fire1StopTimer = 0;
            }
        } else {
            fire1StopTimer = 0;
        }
    }
    
    //readPressureSensors();

    switch (dolum) {
      case 0:
        // Yangın tankları istenen seviyede
      break;
      case 1: // Valfi aç
        dolum_count++;
        prefs.begin("tankcfg", false);
        prefs.putUInt("dolum_count",dolum_count);
        prefs.end();
        digitalWrite(RELAY1, LOW);      // valf aç
        wait_low_pressure_time = millis();
        wait_water_pressure = false;
        //ust_su_talep = true;
        dolum = 2;
      break;

      case 2: // 20 sn içinde hat basıncı düşmeli
        if (yangin_basinc <= 1.8f) {
          // Basınç düştü → valf açıldı kabul
          wait_water_pressure = true;
          wait_water_pressure_time = millis();
          dolum = 3;
        }else if (millis() - wait_low_pressure_time >= 20000UL) {
          messages = "Dolum basıncı beklenen sürede düşmedi! Valf açmamış olabilir.";
          dolum_hata = true;
          //ust_su_talep = false;
          digitalWrite(RELAY1, HIGH);
          dolum = 50;
        }
      break;

      case 3: // 25 sn içinde basınç 1.0-3.0 banda girmeli
        if (yangin_basinc >= 1.0f &&
            yangin_basinc < 3.0f) {
            messages = "Yangın tankları doldurulacak...";
            dolduruldu = false;
            dolum = 4;
        }else if (millis() - wait_water_pressure_time >= 25000UL) {
            if ((yangin_basinc < 1.0f) && !dolduruldu) {
                messages = "Alt tankta su bitti. Kuyudan doldurulacak.";
                //su_bitti = true;
                //ust_su_talep = false;
                digitalWrite(RELAY1, HIGH);
                kuyu_bekle = millis();
                dolum = 5;
            }else if ((yangin_basinc < 1.0f) && dolduruldu) {
                messages = "Alt tank kuyudan dolamıyor !";
                dolum_hata = true;
                dolum = 50;
            } else {
                messages = "Tanklar dolu olabilir ya da seviye hatası var.";
                dolum_hata = true;
                dolum = 50;
            }
        }
      break;

      case 4: // Dolum sürüyor
        if (fire) {
          //ust_su_talep = false;
          digitalWrite(RELAY1, HIGH);
          dolum = 0;
          break;
        }
        if ((yangin_basinc < 1.0f) /*&& ust_su_talep*/) {
          dolduruldu = false;
          //su_bitti = true;
          //ust_su_talep = false;
          digitalWrite(RELAY1, HIGH);
          messages = "Alt tankta su bitti. Kuyudan doldurulacak.";
          kuyu_bekle = millis();
          dolum = 5;
          break;
        }
        if ((yangin_basinc >= 2.0f /*&& ust_su_talep*/) ||
            total_fire_water_levely >= ust_talep_seviye) {
            //ust_su_talep = false;
            digitalWrite(RELAY1, HIGH);
            messages = "Yangın tankları doldu.";
            dolum = 0;
            break;
        }
        /*
        if (!ust_su_talep) {
            digitalWrite(RELAY1, HIGH);
            dolum = 0;
            break;
        }*/
        messages = "Yangın tankları doluyor...";
      break;

      case 5: // Kuyudan su doldurma
        //ust_su_talep = false;
        messages = "Kuyudan su çekiliyor...";
        if(fire_doldu){
        //if (millis() - kuyu_bekle >= 1800000UL) {
            dolduruldu = true;
            //su_bitti = false;
            dolum = 0;
        }
      break;

      case 50: // Hata
        //ust_su_talep = false;
        digitalWrite(RELAY1, HIGH);
        
          messages = "Dolumda hata oluştu";
          dolum = 51;
        
      break;

      case 51:
        // hata sonrası bekleme
      break;

}

    

    

    
  
    unsigned long _now = millis();
    invCommLostSite = (_now - invSiteLastOkMs) > INV_COMM_TIMEOUT_MS;
    invCommLostFire = (_now - invFireLastOkMs) > INV_COMM_TIMEOUT_MS;
    invCommLostFire1 = (_now - invFire1LastOkMs) > INV_COMM_TIMEOUT_MS;
    invCommLostAny  = (invCommLostSite || invCommLostFire || invCommLostFire1 );
    if(invCommLostAny){
        //rst_reason = "İnverterler resetleniyor";
        messages = "İnverterler resetleniyor";
        inverterleriResetle();
    }  
  loop_time = millis() - lt;
}
